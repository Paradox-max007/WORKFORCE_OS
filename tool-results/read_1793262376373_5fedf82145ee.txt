    65→    65→// ---------------------------------------------------------------------------
    66→    66→export function LeaveRequestPage() {
    67→    67→  const user = useAuthStore((s) => s.user);
    68→    68→  const isSuperAdmin = user?.role === 'super_admin';

    70→    70→

    73→    73→
    74→    74→  const [search, setSearch] = useState('');

    96→    96→        setRequests([]);
    97→    97→        setEmployees([]);
    98→    98→      }

   101→   101→  }, [version]);

   116→   116→      rejected: list.filter((r) => r.status === 'rejected').length,
   117→   117→    };
   118→   118→  }, [requests]);

   120→   120→  const filtered = useMemo(() => {

   127→   127→      return hay.includes(debouncedSearch);
   128→   128→    });
   129→   129→  }, [requests, debouncedSearch, statusFilter]);
   130→   130→
   131→   131→  const filtersActive = search.trim().length > 0 || statusFilter !== 'all';

   136→   136→    setSearch('');
   137→   137→    setStatusFilter('all');
   138→   138→  };

   140→   140→  // ---- review (super admin only, pending only) ---------------------------------

   145→   145→      const { req, approve } = reviewTarget;

   148→   148→        description: `${req.employee_name} — ${leaveTypeLabel(req)} (${req.total_days} day(s))`,

   150→   150→      setReviewTarget(null);
   151→   151→    } catch (err) {
   152→   152→      toast.error(err instanceof Error ? err.message : 'Failed to review the request');
   153→   153→    } finally {
   154→   154→      setReviewBusy(false);
   155→   155→    }
   156→   156→  };
   157→   157→
   158→   158→  const shareWhatsApp = (req: LeaveRequestVM) => {

   160→   160→  };

   166→   166→    { label: 'Rejected', value: stats.rejected, icon: <XCircle size={20} />, tone: 'rose' as const, hint: 'Declined applications' },
   167→   167→  ];
   168→   168→
   169→   169→  return (
   170→   170→    <div className="mx-auto max-w-7xl space-y-6">
   171→   171→      <PageHeader
   172→   172→        title="Leave Requests"
   173→   173→        description="Review and manage employee leave applications"
   174→   174→        actions={
   175→   175→          <Button
   176→   176→            onClick={() => setCreateOpen(true)}
   177→   177→            className="h-10 brand-gradient font-semibold text-white shadow-lg shadow-[var(--glow)] transition-transform active:scale-[0.98]"
   178→   178→          >
   179→   179→            <CalendarPlus size={16} /> New Leave Request
   180→   180→          </Button>
   181→   181→        }
   182→   182→      />
   183→   183→
   184→   184→      {/* ---- Stat cards ---- */}
   185→   185→      <div className="grid grid-cols-2 gap-3 md:grid-cols-4">
   186→   186→        {loading
   187→   187→          ? Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-24 rounded-xl" />)
   188→   188→          : statCards.map((s, i) => (
   189→   189→            <motion.div
   190→   190→              key={s.label}
   191→   191→              initial={{ opacity: 0, y: 12 }}
   192→   192→              animate={{ opacity: 1, y: 0 }}
   193→   193→              transition={{ delay: i * 0.04, duration: 0.35, ease: [0.22, 1, 0.36, 1] }}
   194→   194→            >
   195→   195→              <StatCard label={s.label} value={s.value} icon={s.icon} tone={s.tone} hint={s.hint} />
   196→   196→            </motion.div>

   232→   232→              <SelectItem value="rejected">Rejected</SelectItem>
   233→   233→            </SelectContent>

   245→   245→        {/* Content */}
   246→   246→        {loading ? (

   254→   254→              title={filtersActive ? 'No matching leave requests' : 'No leave requests yet'}

   257→   257→                : 'Create the first leave application — it will be reviewed by a super admin.'}
   258→   258→              action={!filtersActive ? (
   259→   259→                <Button
   260→   260→                  onClick={() => setCreateOpen(true)}
   261→   261→                  className="brand-gradient font-semibold text-white"
   262→   262→                >
   263→   263→                  <CalendarPlus size={16} /> New Leave Request
   264→   264→                </Button>
   265→   265→              ) : undefined}
   266→   266→            />
   267→   267→          </div>
   268→   268→        ) : (
   269→   269→          <>
   270→   270→            {/* ---- Desktop table ---- */}
   271→   271→            <div className="hidden overflow-x-auto p-2 md:block">
   272→   272→              <Table>
   273→   273→                <TableHeader>
   274→   274→                  <TableRow>
   275→   275→                    <TableHead className="px-3 py-3 text-xs uppercase tracking-wide text-muted-foreground">Employee</TableHead>
   276→   276→                    <TableHead className="px-3 py-3 text-xs uppercase tracking-wide text-muted-foreground">Leave Type</TableHead>
   277→   277→                    <TableHead className="px-3 py-3 text-xs uppercase tracking-wide text-muted-foreground">Duration</TableHead>
   278→   278→                    <TableHead className="px-3 py-3 text-xs uppercase tracking-wide text-muted-foreground">Reason</TableHead>
   279→   279→                    <TableHead className="px-3 py-3 text-xs uppercase tracking-wide text-muted-foreground">Status</TableHead>
   280→   280→                    <TableHead className="px-3 py-3 text-xs uppercase tracking-wide text-muted-foreground">Created</TableHead>
   281→   281→                    <TableHead className="px-3 py-3 text-right text-xs uppercase tracking-wide text-muted-foreground">Documents</TableHead>
   282→   282→                    {isSuperAdmin && (
   283→   283→                      <TableHead className="px-3 py-3 text-right text-xs uppercase tracking-wide text-muted-foreground">Review</TableHead>

   285→   285→                  </TableRow>
   286→   286→                </TableHeader>
   287→   287→                <TableBody>

   289→   289→                    <motion.tr

   291→   291→                      initial={{ opacity: 0, y: 12 }}
   292→   292→                      animate={{ opacity: 1, y: 0 }}
   293→   293→                      transition={{ delay: i * 0.04, duration: 0.3, ease: [0.22, 1, 0.36, 1] }}
   294→   294→                      className="border-b border-border/40 transition-colors last:border-0 hover:bg-muted/30"
   295→   295→                    >
   296→   296→                      <TableCell className="px-3 py-3">
   297→   297→                        <div className="flex items-center gap-2.5">
   298→   298→                          <span className="flex size-8 shrink-0 items-center justify-center rounded-full bg-brand/15 text-[11px] font-bold text-brand">
   299→   299→                            {initials(r.employee_name)}
   300→   300→                          </span>
   301→   301→                          <div className="min-w-0">
   302→   302→                            <p className="truncate text-sm font-medium">{r.employee_name}</p>
   303→   303→                            <p className="font-mono text-[11px] text-muted-foreground">{r.employee_code}</p>
   304→   304→                          </div>
   305→   305→                        </div>
   306→   306→                      </TableCell>
   307→   307→                      <TableCell className="px-3 py-3">
   308→   308→                        <span className={cn('inline-flex items-center rounded-full border px-2.5 py-0.5 text-[11px] font-medium', LEAVE_TYPE_CHIP[r.leave_type])}>
   309→   309→                          {leaveTypeLabel(r)}
   310→   310→                        </span>
   311→   311→                      </TableCell>
   312→   312→                      <TableCell className="px-3 py-3">
   313→   313→                        <p className="text-sm font-medium">{r.total_days} day{r.total_days === 1 ? '' : 's'}</p>

   345→   345→                            onClick={() => shareWhatsApp(r)}
   346→   346→                          >

   377→   377→                    </motion.tr>
   378→   378→                  ))}

   380→   380→              </Table>
   381→   381→            </div>
   382→   382→
   383→   383→            {/* ---- Mobile cards ---- */}
   384→   384→            <div className="space-y-3 p-4 md:hidden">
   385→   385→              {filtered.map((r, i) => (

   525→   525→function NewLeaveRequestDialog({
   526→   526→  open, onOpenChange, employees,
   527→   527→}: {
   528→   528→  open: boolean;
   529→   529→  onOpenChange: (open: boolean) => void;
   530→   530→  employees: EmployeeVM[];
   531→   531→}) {
   532→   532→  const [employeeId, setEmployeeId] = useState<string | null>(null);
   533→   533→  const [leaveType, setLeaveType] = useState<LeaveType | null>(null);
   534→   534→  const [otherTypeText, setOtherTypeText] = useState('');
   535→   535→  const [startDate, setStartDate] = useState('');
   536→   536→  const [endDate, setEndDate] = useState('');
   537→   537→  const [reason, setReason] = useState('');
   538→   538→  const [submitting, setSubmitting] = useState(false);
   539→   539→
   540→   540→  // Reset the form every time the dialog opens
   541→   541→  useEffect(() => {
   542→   542→    if (open) {
   543→   543→      setEmployeeId(null);
   544→   544→      setLeaveType(null);
   545→   545→      setOtherTypeText('');
   546→   546→      setStartDate('');
   547→   547→      setEndDate('');
   548→   548→      setReason('');
   549→   549→      setSubmitting(false);
   550→   550→    }
   551→   551→  }, [open]);
   552→   552→
   553→   553→  // Auto-calculated total days = date diff + 1 (minimum 1)
   554→   554→  const totalDays = useMemo(() => {
   555→   555→    if (!startDate || !endDate) return 0;
   556→   556→    const s = new Date(startDate);
   557→   557→    const e = new Date(endDate);
   558→   558→    if (Number.isNaN(s.getTime()) || Number.isNaN(e.getTime())) return 0;
   559→   559→    const diff = Math.round((e.getTime() - s.getTime()) / 86400000) + 1;
   560→   560→    return diff > 0 ? diff : 0;
   561→   561→  }, [startDate, endDate]);
   562→   562→
   563→   563→  const selectedEmployee = employees.find((e) => e.id === employeeId) ?? null;
   564→   564→
   565→   565→  const handleSubmit = async () => {
   566→   566→    if (!employeeId) { toast.error('Please select an employee'); return; }
   567→   567→    if (!leaveType) { toast.error('Please select a leave type'); return; }
   568→   568→    if (leaveType === 'other' && !otherTypeText.trim()) { toast.error('Please specify the leave type'); return; }
   569→   569→    if (!startDate || !endDate) { toast.error('Please select the start and end dates'); return; }
   570→   570→    if (totalDays < 1) { toast.error('The end date must be on or after the start date'); return; }
   571→   571→    if (!reason.trim()) { toast.error('Please provide a reason for the leave'); return; }
   572→   572→
   573→   573→    setSubmitting(true);
   574→   574→    try {
   575→   575→      await dataApi.createLeaveRequest({
   576→   576→        employee_id: employeeId,
   577→   577→        leave_type: leaveType,
   578→   578→        other_type_text: leaveType === 'other' ? otherTypeText.trim() : null,
   579→   579→        start_date: startDate,
   580→   580→        end_date: endDate,
   581→   581→        total_days: totalDays,
   582→   582→        reason: reason.trim(),
   583→   583→      });
   584→   584→      toast.success('Leave request submitted', {
   585→   585→        description: `${selectedEmployee?.full_name ?? 'Employee'} — ${totalDays} day(s) of leave, now pending review.`,
   586→   586→      });
   587→   587→      onOpenChange(false);
   588→   588→    } catch (err) {
   589→   589→      toast.error(err instanceof Error ? err.message : 'Failed to submit the leave request');
   590→   590→    } finally {
   591→   591→      setSubmitting(false);
   592→   592→    }
   593→   593→  };
   594→   594→
   595→   595→  return (
   596→   596→    <Dialog open={open} onOpenChange={onOpenChange}>
   597→   597→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-lg">
   598→   598→        <DialogHeader>
   599→   599→          <DialogTitle>New Leave Request</DialogTitle>
   600→   600→          <DialogDescription>
   601→   601→            Submit a leave application on behalf of an employee. A super admin will review it.
   602→   602→          </DialogDescription>
   603→   603→        </DialogHeader>
   604→   604→
   605→   605→        <div className="space-y-4">
   606→   606→          {/* Employee */}
   607→   607→          <div className="space-y-2">
   608→   608→            <Label>Employee <span className="text-rose-500">*</span></Label>
   609→   609→            <EmployeeCombobox
   610→   610→              employees={employees}
   611→   611→              value={employeeId}
   612→   612→              onChange={setEmployeeId}
   613→   613→              placeholder="Search employees…"
   614→   614→              emptyText="No active employees found."
   615→   615→            />
   616→   616→          </div>
   617→   617→
   618→   618→          {/* Leave type */}
   619→   619→          <div className="space-y-2">
   620→   620→            <Label>Leave Type <span className="text-rose-500">*</span></Label>
   621→   621→            <Select
   622→   622→              value={leaveType ?? undefined}
   623→   623→              onValueChange={(v) => {
   624→   624→                setLeaveType(v as LeaveType);
   625→   625→                if (v !== 'other') setOtherTypeText('');
   626→   626→              }}
   627→   627→            >
   628→   628→              <SelectTrigger className="h-11 w-full" aria-label="Leave type">
   629→   629→                <SelectValue placeholder="Select leave type" />
   630→   630→              </SelectTrigger>
   631→   631→              <SelectContent>
   632→   632→                {LEAVE_TYPES.map((t) => (
   633→   633→                  <SelectItem key={t.value} value={t.value}>{t.label}</SelectItem>
   634→   634→                ))}
   635→   635→              </SelectContent>
   636→   636→            </Select>
   637→   637→          </div>
   638→   638→
   639→   639→          {/* Other → specify */}
   640→   640→          {leaveType === 'other' && (
   641→   641→            <motion.div
   642→   642→              initial={{ opacity: 0, y: -6 }}
   643→   643→              animate={{ opacity: 1, y: 0 }}
   644→   644→              className="space-y-2"
   645→   645→            >
   646→   646→              <Label htmlFor="other-type">Specify Leave Type <span className="text-rose-500">*</span></Label>
   647→   647→              <Input
   648→   648→                id="other-type"
   649→   649→                value={otherTypeText}
   650→   650→                onChange={(e) => setOtherTypeText(e.target.value)}
   651→   651→                placeholder="e.g. Study leave"
   652→   652→                className="h-11"
   653→   653→              />
   654→   654→            </motion.div>
   655→   655→          )}
   656→   656→
   657→   657→          {/* Dates */}
   658→   658→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   659→   659→            <div className="space-y-2">
   660→   660→              <Label htmlFor="start-date">Start Date <span className="text-rose-500">*</span></Label>
   661→   661→              <Input
   662→   662→                id="start-date"
   663→   663→                type="date"
   664→   664→                value={startDate}
   665→   665→                onChange={(e) => setStartDate(e.target.value)}
   666→   666→                className="h-11"
   667→   667→              />
   668→   668→            </div>
   669→   669→            <div className="space-y-2">
   670→   670→              <Label htmlFor="end-date">End Date <span className="text-rose-500">*</span></Label>
   671→   671→              <Input
   672→   672→                id="end-date"
   673→   673→                type="date"
   674→   674→                value={endDate}
   675→   675→                min={startDate || undefined}
   676→   676→                onChange={(e) => setEndDate(e.target.value)}
   677→   677→                className="h-11"
   678→   678→              />
   679→   679→            </div>
   680→   680→          </div>
   681→   681→
   682→   682→          {/* Total days (read-only, auto) */}
   683→   683→          <div className="space-y-2">

   706→   706→            />

   709→   709→
   710→   710→        <DialogFooter>
   711→   711→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={submitting}>
   712→   712→            Cancel

   715→   715→            onClick={handleSubmit}
   716→   716→            disabled={submitting}
   717→   717→            className="brand-gradient font-semibold text-white shadow-lg shadow-[var(--glow)] transition-transform active:scale-[0.98]"
   718→   718→          >
   719→   719→            {submitting ? <Loader2 size={16} className="animate-spin" /> : <CalendarPlus size={16} />}
   720→   720→            Submit Request
   721→   721→          </Button>
   722→   722→        </DialogFooter>
   723→   723→      </DialogContent>
   724→   724→    </Dialog>
   725→   725→  );
