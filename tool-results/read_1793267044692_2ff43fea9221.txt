   200→export function SitesPage() {
   201→  const user = useAuthStore((s) => s.user);
   202→  const isSuperAdmin = user?.role === 'super_admin';
   203→  const version = useRealtimeVersion(['site', 'employee']);
   204→
   205→  const [sites, setSites] = useState<SiteWithCount[] | null>(null);
   206→  const [employees, setEmployees] = useState<EmployeeVM[] | null>(null);
   207→
   208→  const [subView, setSubView] = useState<SubView>('list');
   209→  const [viewSiteId, setViewSiteId] = useState<string | null>(null);
   210→  const [tab, setTab] = useState<SiteTab>('active');
   211→  const [search, setSearch] = useState('');
   212→  const [debouncedSearch, setDebouncedSearch] = useState('');
   213→  const [togglingId, setTogglingId] = useState<string | null>(null);
   214→
   215→  // dialogs
   216→  const [formOpen, setFormOpen] = useState(false);
   217→  const [editTarget, setEditTarget] = useState<SiteWithCount | null>(null);
   218→  const [deleteTarget, setDeleteTarget] = useState<SiteWithCount | null>(null);
   219→  const [deactivateTarget, setDeactivateTarget] = useState<SiteWithCount | null>(null);
   220→  const [dialogBusy, setDialogBusy] = useState(false);
   221→
   222→  // attendance sheet overlay
   223→  const [attendanceSiteId, setAttendanceSiteId] = useState<string | null>(null);
   224→
   225→  // ---- data loading (re-fetches whenever sites or employees change) --------
   226→  useEffect(() => {
   227→    let cancelled = false;
   228→    (async () => {
   229→      try {
   230→        const [s, e] = await Promise.all([dataApi.listSites(), dataApi.listEmployees()]);
   231→        if (!cancelled) {
   232→          setSites(s);
   233→          setEmployees(e);
   234→        }
   235→      } catch {
   236→        if (!cancelled) {
   237→          setSites([]);
   238→          setEmployees([]);
   239→        }
   240→      }
   241→    })();
   242→    return () => {
   243→      cancelled = true;
   244→    };
   245→  }, [version]);
   246→
   247→  // ---- search debounce (300ms) ---------------------------------------------
   248→  useEffect(() => {
   249→    const t = setTimeout(() => setDebouncedSearch(search.trim().toLowerCase()), 300);
   250→    return () => clearTimeout(t);
   251→  }, [search]);
   252→
   253→  // ---- derived values --------------------------------------------------------
   254→  const activeSites = useMemo(() => (sites ?? []).filter((s) => s.is_active), [sites]);
   255→  const inactiveSites = useMemo(() => (sites ?? []).filter((s) => !s.is_active), [sites]);
   256→  const loading = sites === null;
   257→
   258→  const viewSite = useMemo(
   259→    () => (sites ?? []).find((s) => s.id === viewSiteId) ?? null,
   260→    [sites, viewSiteId],
   261→  );
   262→  const attendanceSite = useMemo(
   263→    () => (sites ?? []).find((s) => s.id === attendanceSiteId) ?? null,
   264→    [sites, attendanceSiteId],
   265→  );
   266→  const showEmployeesView = subView === 'employees' && !!viewSite;
   267→
   268→  const matchesSearch = (s: SiteWithCount): boolean => {
   269→    if (!debouncedSearch) return true;
   270→    const hay = `${s.name} ${s.client_name ?? ''} ${s.project_name ?? ''}`.toLowerCase();
   271→    return hay.includes(debouncedSearch);
   272→  };
   273→
   274→  // ---- actions ---------------------------------------------------------------
   275→
   276→  const openCreate = () => {
   277→    setEditTarget(null);
   278→    setFormOpen(true);
   279→  };
   280→  const openEdit = (site: SiteWithCount) => {
   281→    setEditTarget(site);
   282→    setFormOpen(true);
   283→  };
   284→
   285→  const openEmployees = (site: SiteWithCount) => {
   286→    setViewSiteId(site.id);
   287→    setSubView('employees');
   288→  };
   289→  const backToList = () => {
   290→    setSubView('list');
   291→    setViewSiteId(null);
   292→  };
   293→
   294→  /** Activate directly; deactivating first asks for confirmation. */
   295→  const handleToggleActive = async (site: SiteWithCount) => {
   296→    if (site.is_active) {
   297→      setDeactivateTarget(site);
   298→      return;
   299→    }
   300→    setTogglingId(site.id);
   301→    try {
   302→      await dataApi.updateSite(site.id, {
   303→        name: site.name,
   304→        client_name: site.client_name,
   305→        project_name: site.project_name,
   306→        is_active: true,
   307→      });
   308→      toast.success('Site activated', {
   309→        description: `"${site.name}" is active again — employees can be assigned to it.`,
   310→      });
   311→    } catch (err) {
   312→      toast.error(err instanceof Error ? err.message : 'Failed to activate the site');
   313→    } finally {
   314→      setTogglingId(null);
   315→    }
   316→  };
   317→
   318→  const handleConfirmDeactivate = async () => {
   319→    if (!deactivateTarget) return;
   320→    setDialogBusy(true);
   321→    try {
   322→      const { unassignedCount } = await dataApi.updateSite(deactivateTarget.id, {
   323→        name: deactivateTarget.name,
   324→        client_name: deactivateTarget.client_name,
   325→        project_name: deactivateTarget.project_name,
   326→        is_active: false,
   327→      });
   328→      toast.success('Site deactivated', {
   329→        description: `"${deactivateTarget.name}" was moved to the Inactive tab.${
   330→          unassignedCount > 0
   331→            ? ` ${unassignedCount} employee(s) were set to idle/unassigned and team leader status was cleared.`
   332→            : ''
   333→        }`,
   334→      });
   335→      setDeactivateTarget(null);
   336→    } catch (err) {
   337→      toast.error(err instanceof Error ? err.message : 'Failed to deactivate the site');
   338→    } finally {
   339→      setDialogBusy(false);
   340→    }
   341→  };
   342→
   343→  const handleConfirmDelete = async () => {
   344→    if (!deleteTarget) return;
   345→    setDialogBusy(true);
   346→    try {
   347→      await dataApi.deleteSite(deleteTarget.id);
   348→      toast.success('Site deleted', {
   349→        description: `"${deleteTarget.name}" was removed. Assigned employees were unassigned — not deleted.`,
   350→      });
   351→      setDeleteTarget(null);
   352→    } catch (err) {
   353→      toast.error(err instanceof Error ? err.message : 'Failed to delete the site');
   354→    } finally {
   355→      setDialogBusy(false);
   356→    }
   357→  };
   358→
   359→  // ---- grid renderer (shared by both tabs) ----------------------------------
   360→  const renderGrid = (list: SiteWithCount[], tabValue: SiteTab) => {
   361→    if (loading) {
   362→      return (
   363→        <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   364→          {Array.from({ length: 6 }).map((_, i) => (
   365→            <Skeleton key={i} className="h-52 rounded-xl" />
   366→          ))}
   367→        </div>
   368→      );
   369→    }
   370→    const filtered = list.filter(matchesSearch);
   371→    if (filtered.length === 0) {
   372→      const searching = debouncedSearch.length > 0;
   373→      return (
   374→        <EmptyState
   375→          icon={<Building2 size={22} />}
   376→          title={
   377→            searching
   378→              ? 'No matching sites'
   379→              : tabValue === 'active'
   380→                ? 'No active sites'
   381→                : 'No inactive sites'
   382→          }
   383→          description={
   384→            searching
   385→              ? 'Try a different search term — search covers site, client and project names.'
   386→              : tabValue === 'active'
   387→                ? 'Create your first work site to start assigning crews and tracking attendance.'
   388→                : 'Sites you deactivate will be moved here and can be reactivated at any time.'
   389→          }
   390→          action={
   391→            !searching && tabValue === 'active' ? (
   392→              <Button onClick={openCreate} className="brand-gradient font-semibold text-white">
   393→                <Plus size={16} /> New Site
   394→              </Button>
   395→            ) : undefined
   396→          }
   397→        />
   398→      );
   399→    }
   400→    return (
   401→      <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   402→        {filtered.map((site, i) => (
   403→          <SiteCard
   404→            key={site.id}
   405→            site={site}
   406→            index={i}
   407→            canDelete={isSuperAdmin}
   408→            busy={togglingId === site.id}
   409→            onToggle={handleToggleActive}
   410→            onEdit={openEdit}
   411→            onDelete={setDeleteTarget}
   412→            onViewEmployees={openEmployees}
   413→            onAttendance={(s) => setAttendanceSiteId(s.id)}
   414→          />
   415→        ))}
   416→      </div>
   417→    );
   418→  };
   419→
   420→  // ---- render ----------------------------------------------------------------
   421→  return (
   422→    <div className="mx-auto max-w-7xl">
   423→      <AnimatePresence mode="wait">
   424→        {showEmployeesView && viewSite ? (
   425→          <motion.div
   426→            key="site-employees"
   427→            initial={{ opacity: 0, x: 32 }}
   428→            animate={{ opacity: 1, x: 0 }}
   429→            exit={{ opacity: 0, x: 32 }}
   430→            transition={{ duration: 0.3, ease: EASE }}
   431→          >
   432→            <SiteEmployeesView site={viewSite} employees={employees ?? []} onBack={backToList} />
   433→          </motion.div>
   434→        ) : (
   435→          <motion.div
   436→            key="site-list"
   437→            initial={{ opacity: 0, y: 16 }}
   438→            animate={{ opacity: 1, y: 0 }}
   439→            exit={{ opacity: 0, y: -16 }}
   440→            transition={{ duration: 0.3, ease: EASE }}
   441→          >
   442→            <div className="space-y-6">
   443→              <PageHeader
   444→                title="Sites"
   445→                description="Manage work sites and the crews assigned to each location."
   446→                actions={
   447→                  <Button
   448→                    onClick={openCreate}
   449→                    className="h-10 brand-gradient font-semibold text-white shadow-lg shadow-[var(--glow)] transition-transform active:scale-[0.98]"
   450→                  >
   451→                    <Plus size={16} /> New Site
   452→                  </Button>
   453→                }
   454→              />
   455→
   456→              <Tabs value={tab} onValueChange={(v) => setTab(v as SiteTab)}>
   457→                <div className="card-glow rounded-xl border bg-card">
   458→                  {/* Tabs + shared search */}
   459→                  <div className="flex flex-col gap-3 border-b p-4 lg:flex-row lg:items-center lg:justify-between">
   460→                    <TabsList className="h-11 w-full gap-1 sm:w-auto">
   461→                      <TabsTrigger value="active" className="h-9 gap-2 px-4">
   462→                        <Power size={15} className="text-emerald-500" />
   463→                        Active
   464→                        <span className="rounded-full bg-emerald-500/15 px-2 py-0.5 text-[11px] font-semibold tabular-nums text-emerald-500">
   465→                          {activeSites.length}
   466→                        </span>
   467→                      </TabsTrigger>
   468→                      <TabsTrigger value="inactive" className="h-9 gap-2 px-4">
   469→                        <PowerOff size={15} className="text-zinc-400" />
   470→                        Inactive
   471→                        <span className="rounded-full bg-zinc-500/15 px-2 py-0.5 text-[11px] font-semibold tabular-nums text-zinc-400">
   472→                          {inactiveSites.length}
   473→                        </span>
   474→                      </TabsTrigger>
   475→                    </TabsList>
   476→                    <div className="relative w-full lg:max-w-xs">
   477→                      <Search size={16} className="pointer-events-none absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   478→                      <Input

   773→function SiteFormDialog({
   774→  open, onOpenChange, editing,
   775→}: {
   776→  open: boolean;
   777→  onOpenChange: (open: boolean) => void;
   778→  editing: SiteWithCount | null;
   779→}) {
   780→  const [name, setName] = useState('');
   781→  const [client, setClient] = useState('');
   782→  const [project, setProject] = useState('');
   783→  const [busy, setBusy] = useState(false);
   784→
   785→  // Reset the form every time the dialog opens
   786→  useEffect(() => {
   787→    if (open) {
   788→      setName(editing?.name ?? '');
   789→      setClient(editing?.client_name ?? '');
   790→      setProject(editing?.project_name ?? '');
   791→      setBusy(false);
   792→    }
   793→  }, [open, editing]);
   794→
   795→  const trimmedName = name.trim();
   796→  const unchanged =
   797→    !!editing &&
   798→    trimmedName === editing.name &&
   799→    client.trim() === (editing.client_name ?? '') &&
   800→    project.trim() === (editing.project_name ?? '');
   801→
   802→  const submit = async () => {
   803→    if (!trimmedName || busy || unchanged) return;
   804→    setBusy(true);
   805→    try {
   806→      if (editing) {
   807→        await dataApi.updateSite(editing.id, {
   808→          name: trimmedName,
   809→          client_name: client.trim() || null,
   810→          project_name: project.trim() || null,
   811→          is_active: editing.is_active,
   812→        });
   813→        toast.success('Site updated', { description: `Details for "${trimmedName}" were saved.` });
   814→      } else {
   815→        const created = await dataApi.createSite(
   816→          trimmedName,
   817→          client.trim() || null,
   818→          project.trim() || null,
   819→        );
   820→        toast.success('Site created', { description: `"${created.name}" has been added.` });
   821→      }
   822→      onOpenChange(false);
   823→    } catch (err) {
   824→      toast.error(err instanceof Error ? err.message : 'Failed to save the site');
   825→    } finally {
   826→      setBusy(false);
   827→    }
   828→  };
   829→
   830→  return (
   831→    <Dialog open={open} onOpenChange={onOpenChange}>
   832→      <DialogContent className="sm:max-w-md">
   833→        <DialogHeader>
   834→          <DialogTitle>{editing ? 'Edit Site' : 'Add New Site'}</DialogTitle>
   835→          <DialogDescription>
   836→            {editing
   837→              ? `Update the details of "${editing.name}". Assigned employees are unaffected.`
   838→              : 'Enter the details for the new work site.'}
   839→          </DialogDescription>
   840→        </DialogHeader>
   841→
   842→        <div className="space-y-4">
   843→          <div className="space-y-2">
   844→            <Label htmlFor="site-name">
   845→              Site Name <span className="text-rose-500">*</span>
   846→            </Label>
   847→            <Input
   848→              id="site-name"
   849→              value={name}
   850→              onChange={(e) => setName(e.target.value)}
   851→              placeholder="e.g. NPC — Umm Al Quwain"
   852→              className="h-11"
   853→              onKeyDown={(e) => {
   854→                if (e.key === 'Enter') submit();
   855→              }}
   856→            />
   857→          </div>
   858→          <div className="space-y-2">
   859→            <Label htmlFor="site-client">Client Name</Label>
   860→            <Input
   861→              id="site-client"
   862→              value={client}
   863→              onChange={(e) => setClient(e.target.value)}
   864→              placeholder="e.g. NPC"
   865→              className="h-11"
   866→              onKeyDown={(e) => {
   867→                if (e.key === 'Enter') submit();
   868→              }}
   869→            />
   870→          </div>
   871→          <div className="space-y-2">
   872→            <Label htmlFor="site-project">Project Name</Label>
   873→            <Input
   874→              id="site-project"
   875→              value={project}
   876→              onChange={(e) => setProject(e.target.value)}
   877→              placeholder="e.g. NPC — Umm Al Quwain"
   878→              className="h-11"
   879→              onKeyDown={(e) => {
   880→                if (e.key === 'Enter') submit();
   881→              }}
   882→            />
   883→          </div>
   884→        </div>
   885→
   886→        <DialogFooter>
   887→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={busy}>
   888→            Cancel
   889→          </Button>
   890→          <Button
   891→            onClick={submit}

   907→function SiteEmployeesView({
   908→  site, employees, onBack,
   909→}: {
   910→  site: SiteWithCount;
   911→  employees: EmployeeVM[];
   912→  onBack: () => void;
   913→}) {
   914→  const [search, setSearch] = useState('');
   915→  const [selected, setSelected] = useState<Set<string>>(new Set());
   916→  const [removeOpen, setRemoveOpen] = useState(false);
   917→  const [removing, setRemoving] = useState(false);
   918→  const [addingId, setAddingId] = useState<string | null>(null);
   919→  const [pickerValue, setPickerValue] = useState<string | null>(null);
   920→
   921→  const siteEmployees = useMemo(
   922→    () => employees.filter((e) => e.current_site_id === site.id && e.status !== 'deleted'),
   923→    [employees, site.id],
   924→  );
   925→  const leader = useMemo(
   926→    () => siteEmployees.find((e) => e.is_team_leader && e.team_leader_site_id === site.id) ?? null,
   927→    [siteEmployees, site.id],
   928→  );
   929→  // transferable: active employees that are not already on this site
   930→  const addable = useMemo(
   931→    () => employees.filter((e) => e.status === 'active' && e.current_site_id !== site.id),
   932→    [employees, site.id],
   933→  );
   934→
   935→  const filtered = useMemo(() => {
   936→    const q = search.trim().toLowerCase();
   937→    const list = q
   938→      ? siteEmployees.filter((e) =>
   939→          `${e.full_name} ${e.employee_code} ${e.position ?? ''} ${e.nationality ?? ''}`
   940→            .toLowerCase()
   941→            .includes(q),
   942→        )
   943→      : siteEmployees;
   944→    // team leader of this site first, everything else keeps list order
   945→    return [...list].sort((a, b) => {
   946→      const aLead = a.is_team_leader && a.team_leader_site_id === site.id;
   947→      const bLead = b.is_team_leader && b.team_leader_site_id === site.id;
   948→      if (aLead !== bLead) return aLead ? -1 : 1;
   949→      return 0;
   950→    });
   951→  }, [siteEmployees, search, site.id]);
   952→
   953→  const allSelected = filtered.length > 0 && filtered.every((e) => selected.has(e.id));
   954→  const someSelected = !allSelected && filtered.some((e) => selected.has(e.id));
   955→
   956→  const toggleOne = (id: string) => {
   957→    setSelected((prev) => {
   958→      const next = new Set(prev);
   959→      if (next.has(id)) next.delete(id);
   960→      else next.add(id);
   961→      return next;
   962→    });
   963→  };
   964→  const toggleAll = () => {
   965→    setSelected(allSelected ? new Set() : new Set(filtered.map((e) => e.id)));
   966→  };
   967→
   968→  // ---- add / transfer an employee onto this site -----------------------------
   969→  const handleAdd = async (id: string | null) => {
   970→    setPickerValue(null);
   971→    if (!id || addingId) return;
   972→    const emp = employees.find((e) => e.id === id);
   973→    if (!emp) return;
   974→    setAddingId(id);
   975→    try {
   976→      await dataApi.updateEmployee(id, toEmployeeInput(emp, { current_site_id: site.id }));
   977→      if (emp.site_name) {
   978→        toast.success('Employee transferred', {
   979→          description: `${emp.full_name} was transferred from "${emp.site_name}" to "${site.name}".`,
   980→        });
   981→      } else {
   982→        toast.success('Employee added', {
   983→          description: `${emp.full_name} was added to "${site.name}".`,
   984→        });
   985→      }
   986→    } catch (err) {
   987→      toast.error(err instanceof Error ? err.message : 'Failed to add the employee');
   988→    } finally {
   989→      setAddingId(null);
   990→    }
   991→  };
   992→
   993→  // ---- remove selected employees (set to idle, never delete) -----------------
   994→  const handleRemove = async () => {
   995→    if (selected.size === 0) return;
   996→    setRemoving(true);
   997→    try {
   998→      await Promise.all(
   999→        [...selected].map((id) => {
  1000→          const emp = employees.find((e) => e.id === id);
  1001→          if (!emp) return Promise.resolve();
  1002→          const leadsThisSite = emp.is_team_leader && emp.team_leader_site_id === site.id;
  1003→          return dataApi.updateEmployee(
  1004→            id,
  1005→            toEmployeeInput(emp, {
  1006→              current_site_id: null,
  1007→              ...(leadsThisSite ? { is_team_leader: false, team_leader_site_id: null } : {}),
  1008→            }),
  1009→          );
  1010→        }),
  1011→      );
  1012→      toast.success(`${selected.size} employee(s) removed`, {
  1013→        description: 'They were set to idle/unassigned — not deleted. Team leader status was cleared where applicable.',
  1014→      });
  1015→      setSelected(new Set());
  1016→      setRemoveOpen(false);
  1017→    } catch (err) {
  1018→      toast.error(err instanceof Error ? err.message : 'Failed to remove the employee(s)');
  1019→    } finally {
  1020→      setRemoving(false);
  1021→    }
  1022→  };
  1023→
  1024→  const meta = [site.client_name, site.project_name].filter(Boolean).join(' · ');
  1025→
  1026→  return (
  1027→    <div className="space-y-6">
  1028→      {/* ---- header: back, site identity, team leader callout ---- */}
  1029→      <div className="flex flex-col gap-4 lg:flex-row lg:items-start lg:justify-between">
  1030→        <div className="flex items-start gap-3">
  1031→          <Button
  1032→            variant="outline"
  1033→            size="icon"
