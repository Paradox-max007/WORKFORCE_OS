    85→    85→/** Parse a cell value → { kind, hours }. kind: 'absent' | 'hours' | 'clear' | 'invalid' */
    86→    86→function parseCell(raw: string): { kind: 'absent' | 'hours' | 'clear' | 'invalid'; hours: number | null } {
    87→    87→  const v = raw.trim();
    88→    88→  if (v === '') return { kind: 'clear', hours: null };
    89→    89→  if (v.toLowerCase() === 'a' || v.toLowerCase() === 'absent') return { kind: 'absent', hours: null };
    90→    90→  const n = Number(v);
    91→    91→  if (Number.isNaN(n) || !Number.isFinite(n) || n < 0 || n > 24) return { kind: 'invalid', hours: null };
    92→    92→  return { kind: 'hours', hours: Math.round(n * 100) / 100 };
    93→    93→}
