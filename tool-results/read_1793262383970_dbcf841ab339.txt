   106→/* ───────── Star Rating ───────── */
   107→function StarRating({ rating }: { rating: number }) {
   108→  const fullStars = Math.floor(rating);
   109→  const hasHalf = rating - fullStars >= 0.25 && rating - fullStars < 0.75;
   110→  const emptyStars = 5 - fullStars - (hasHalf ? 1 : 0);
   111→  const isFull5 = rating >= 4.75;
   112→
   113→  return (
   114→    <div className="flex items-center gap-0.5">
   115→      {isFull5 ? (
   116→        Array.from({ length: 5 }).map((_, i) => (
   117→          <Star key={i} className="h-3.5 w-3.5 fill-amber-400 text-amber-400" />
   118→        ))
   119→      ) : (
   120→        <>
   121→          {Array.from({ length: fullStars }).map((_, i) => (
   122→            <Star key={`full-${i}`} className="h-3.5 w-3.5 fill-amber-400 text-amber-400" />
   123→          ))}
   124→          {hasHalf && (
   125→            <div className="relative">
   126→              <Star className="h-3.5 w-3.5 text-slate-600" />
   127→              <div className="absolute inset-0 overflow-hidden w-1/2">
   128→                <Star className="h-3.5 w-3.5 fill-amber-400 text-amber-400" />
   129→              </div>
   130→            </div>
   131→          )}
   132→          {Array.from({ length: emptyStars }).map((_, i) => (
   133→            <Star key={`empty-${i}`} className="h-3.5 w-3.5 text-slate-600" />
   134→          ))}
   135→        </>
   136→      )}
   137→      <span className="text-xs text-slate-400 ml-1 font-medium">{rating.toFixed(1)}</span>
   138→    </div>
   139→  );
   140→}

   325→/* ───────── Add Employee Combobox ───────── */
   326→function AddEmployeeCombobox({
   327→  allEmployees,
   328→  currentSiteName,
   329→  currentSiteEmployeeIds,
   330→  onAdd,
   331→  loading,
   332→}: {
   333→  allEmployees: AllEmployee[];
   334→  currentSiteName: string;
   335→  currentSiteEmployeeIds: Set<string>;
   336→  onAdd: (employee: AllEmployee) => void;
   337→  loading: boolean;
   338→}) {
   339→  const [open, setOpen] = useState(false);
   340→  const [filter, setFilter] = useState('');
   341→
   342→  const filtered = useMemo(() => {
   343→    if (!filter) return allEmployees;
   344→    const q = filter.toLowerCase();
   345→    return allEmployees.filter(
   346→      (e) =>
   347→        e.fullName.toLowerCase().includes(q) ||
   348→        e.employeeId.toLowerCase().includes(q) ||
   349→        (e.position && e.position.toLowerCase().includes(q))
   350→    );
   351→  }, [allEmployees, filter]);
   352→
   353→  return (
   354→    <Popover open={open} onOpenChange={setOpen}>
   355→      <PopoverTrigger asChild>
   356→        <Button
   357→          variant="outline"
   358→          disabled={loading}
   359→          className="bg-slate-700/50 border-slate-600 text-slate-200 hover:bg-emerald-600 hover:text-white hover:border-emerald-600 gap-2 transition-all"
   360→        >
   361→          {loading ? (
   362→            <Loader2 className="h-4 w-4 animate-spin" />
   363→          ) : (
   364→            <UserPlus className="h-4 w-4" />
   365→          )}
   366→          Add Employee
   367→        </Button>
   368→      </PopoverTrigger>
   369→      <PopoverContent className="w-80 p-0 bg-slate-800 border-slate-600" align="start">
   370→        <Command className="bg-slate-800">
   371→          <CommandInput
   372→            placeholder="Search employees..."
   373→            value={filter}
   374→            onValueChange={setFilter}
   375→            className="text-white"
   376→          />
   377→          <CommandList className="max-h-64">
   378→            <CommandEmpty className="text-slate-400">No employees found.</CommandEmpty>
   379→            <CommandGroup>
   380→              {filtered.map((emp) => {
   381→                const isAlreadyInSite = currentSiteEmployeeIds.has(emp.id);
   382→                const isFromAnotherSite = emp.currentSite && emp.currentSite !== currentSiteName;
   383→                const isIdle = !emp.currentSite;
   384→
   385→                return (
   386→                  <CommandItem
   387→                    key={emp.id}
   388→                    value={`${emp.fullName} ${emp.employeeId}`}
   389→                    onSelect={() => {
   390→                      if (!isAlreadyInSite) {
   391→                        onAdd(emp);
   392→                        setOpen(false);
   393→                        setFilter('');
   394→                      }
   395→                    }}
   396→                    className={cn(
   397→                      'text-slate-200 data-[selected=true]:bg-slate-700 data-[selected=true]:text-white py-2.5',
   398→                      isAlreadyInSite && 'opacity-50 cursor-not-allowed'
   399→                    )}
   400→                  >
   401→                    <div className="flex flex-col flex-1 min-w-0">
   402→                      <div className="flex items-center gap-2">
   403→                        <span className="text-sm truncate">{emp.fullName}</span>
   404→                        {isAlreadyInSite && (
   405→                          <Badge className="bg-emerald-500/20 text-emerald-400 border-emerald-500/30 text-[10px] px-1.5 py-0 shrink-0">
   406→                            Current
   407→                          </Badge>
   408→                        )}
   409→                        {isFromAnotherSite && (
   410→                          <Badge className="bg-amber-500/20 text-amber-400 border-amber-500/30 text-[10px] px-1.5 py-0 shrink-0 flex items-center gap-0.5">
   411→                            <MapPin className="h-2.5 w-2.5" />
   412→                            {emp.currentSite}
   413→                          </Badge>
   414→                        )}
   415→                        {isIdle && (
   416→                          <Badge className="bg-slate-500/20 text-slate-400 border-slate-500/30 text-[10px] px-1.5 py-0 shrink-0">
   417→                            Idle
   418→                          </Badge>
   419→                        )}
   420→                      </div>
   421→                      <span className="text-[11px] text-slate-500">
   422→                        {emp.employeeId}
   423→                        {emp.position ? ` · ${emp.position}` : ''}
   424→                      </span>
   425→                    </div>
   426→                  </CommandItem>
   427→                );
   428→              })}
   429→            </CommandGroup>
   430→          </CommandList>
   431→        </Command>
   432→      </PopoverContent>
   433→    </Popover>
   434→  );
   435→}
