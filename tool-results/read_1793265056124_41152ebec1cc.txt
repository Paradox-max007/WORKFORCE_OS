   205→-- Am I a super admin?
   206→create or replace function public.is_super_admin()
   207→returns boolean language sql stable
   208→security definer set search_path = public
   209→as $$ select exists(select 1 from public.profiles where id = auth.uid() and role = 'super_admin'); $$;
   210→
   211→-- Am I staff (admin or super_admin)?
   212→create or replace function public.is_staff()
   213→returns boolean language sql stable
   214→security definer set search_path = public
   215→as $$ select exists(select 1 from public.profiles where id = auth.uid() and role in ('super_admin','admin')); $$;
   216→
   217→-- Does the current admin hold a given menu permission (super admins always true)?
   218→create or replace function public.has_menu_permission(p_menu_key text)
   219→returns boolean language sql stable
   220→security definer set search_path = public
   221→as $$
   222→  select
   223→    exists(select 1 from public.profiles where id = auth.uid() and role = 'super_admin')
   224→    or exists(
   225→      select 1 from public.menu_permission mp
   226→      join public.profiles p on p.id = mp.profile_id
   227→      where mp.profile_id = auth.uid() and mp.menu_key = p_menu_key and mp.allowed and p.role = 'admin'
   228→    );
   229→$$;

   231→-- Fan-out a notification to every super admin
   232→create or replace function public.notify_super_admins(p_title text, p_message text, p_type text)
   233→returns void language plpgsql
   234→security definer set search_path = public
   235→as $$
   236→begin
   237→  if not public.is_staff() then raise exception 'Access denied'; end if;
   238→  insert into public.notification (profile_id, title, message, type)
   239→  select id, p_title, p_message, p_type from public.profiles where role = 'super_admin';
   240→end; $$;

   264→-- New auth user → profile row (first user becomes super_admin, others 'pending'
   265→-- until a super admin promotes them from Admin Management)
   266→create or replace function public.handle_new_user()
   267→returns trigger language plpgsql
   268→security definer set search_path = public
   269→as $$
   270→begin
   271→  insert into public.profiles (id, email, name, role)
   272→  values (
   273→    new.id,
   274→    lower(new.email),
   275→    coalesce(new.raw_user_meta_data->>'name', split_part(new.email, '@', 1)),
   276→    case when (select count(*) from public.profiles) = 0 then 'super_admin' else 'pending' end
   277→  )
   278→  on conflict (id) do nothing;
   279→  return new;
   280→end; $$;
   281→
   282→drop trigger if exists on_auth_user_created on auth.users;
   283→create trigger on_auth_user_created
   284→  after insert on auth.users
   285→  for each row execute function public.handle_new_user();

   363→-- Update own display name + theme + accent
   364→create or replace function public.set_preferences(p_name text default null, p_theme text default null, p_accent text default null)
   365→returns public.profiles language plpgsql
   366→security definer set search_path = public
   367→as $$
   368→declare result public.profiles;
   369→begin
   370→  update public.profiles set
   371→    name   = coalesce(p_name, name),
   372→    theme  = coalesce(nullif(p_theme, ''), theme),
   373→    accent = coalesce(nullif(p_accent, ''), accent)
   374→  where id = auth.uid()
   375→  returning * into result;
   376→  return result;
   377→end; $$;

   379→-- Super admin edits another profile's name / role (email + password go via the
   380→-- service-role API route). Last-super-admin protection included.
   381→create or replace function public.update_profile_admin(p_id uuid, p_name text, p_role text)
   382→returns public.profiles language plpgsql
   383→security definer set search_path = public
   384→as $$
   385→declare result public.profiles;
   386→declare target_count int;
   387→begin
   388→  if not public.is_super_admin() then raise exception 'Only super admins can manage profiles'; end if;
   389→  if p_role not in ('admin','super_admin') then raise exception 'Invalid role'; end if;
   390→
   391→  -- last super admin protection
   392→  if (select role from public.profiles where id = p_id) = 'super_admin' and p_role <> 'super_admin' then
   393→    select count(*) into target_count from public.profiles where role = 'super_admin';
   394→    if target_count <= 1 then raise exception 'Cannot demote the last super admin'; end if;
   395→  end if;
   396→
   397→  update public.profiles set name = p_name, role = p_role, updated_at = now()
   398→  where id = p_id returning * into result;
   399→  return result;
   400→end; $$;
   401→
   402→-- Super admin deletes a profile (last-super-admin protected). The auth.users
   403→-- row is removed separately by the service-role API route.
   404→create or replace function public.delete_profile_admin(p_id uuid)
   405→returns void language plpgsql
   406→security definer set search_path = public
   407→as $$
   408→declare target_count int;
   409→begin
   410→  if not public.is_super_admin() then raise exception 'Only super admins can delete profiles'; end if;
   411→  if (select role from public.profiles where id = p_id) = 'super_admin' then
   412→    select count(*) into target_count from public.profiles where role = 'super_admin';
   413→    if target_count <= 1 then raise exception 'Cannot delete the last super admin'; end if;
   414→  end if;
   415→  delete from public.profiles where id = p_id;
   416→end; $$;

   418→-- Grant / revoke a menu permission for an admin
   419→create or replace function public.set_menu_permission(p_profile_id uuid, p_menu_key text, p_allowed boolean)
   420→returns void language plpgsql
   421→security definer set search_path = public
   422→as $$
   423→begin
   424→  if not public.is_super_admin() then raise exception 'Only super admins can manage permissions'; end if;
   425→  if (select role from public.profiles where id = p_profile_id) = 'super_admin' then
   426→    raise exception 'Super admins always have full access';
   427→  end if;
   428→  insert into public.menu_permission (profile_id, menu_key, allowed)
   429→  values (p_profile_id, p_menu_key, p_allowed)
   430→  on conflict (profile_id, menu_key) do update set allowed = p_allowed;
   431→end; $$;

   437→create or replace function public.create_site(p_name text, p_client_name text default null, p_project_name text default null)
   438→returns public.site language plpgsql
   439→security definer set search_path = public
   440→as $$
   441→declare result public.site;
   442→begin
   443→  if not public.has_menu_permission('sites') then raise exception 'Access denied'; end if;
   444→  if exists (select 1 from public.site where lower(name) = lower(p_name)) then
   445→    raise exception 'A site with this name already exists';
   446→  end if;
   447→  insert into public.site (name, client_name, project_name) values (p_name, p_client_name, p_project_name)
   448→  returning * into result;
   449→  return result;
   450→end; $$;

   452→-- Returns the site + how many employees got unassigned when deactivating.
   453→create or replace function public.update_site(
   454→  p_id uuid, p_name text, p_client_name text, p_project_name text, p_is_active boolean
   455→)
   456→returns jsonb language plpgsql
   457→security definer set search_path = public
   458→as $$
   459→declare result public.site;
   460→declare unassigned_count int := 0;
   461→declare was_active boolean;
   462→begin
   463→  if not public.has_menu_permission('sites') then raise exception 'Access denied'; end if;
   464→  if exists (select 1 from public.site where lower(name) = lower(p_name) and id <> p_id) then
   465→    raise exception 'A site with this name already exists';
   466→  end if;
   467→
   468→  select is_active into was_active from public.site where id = p_id;
   469→  update public.site set name = p_name, client_name = p_client_name, project_name = p_project_name, is_active = p_is_active
   470→  where id = p_id returning * into result;
   471→
   472→  if was_active = true and p_is_active = false then
   473→    select count(*) into unassigned_count from public.employee where current_site_id = p_id and status <> 'deleted';
   474→    update public.employee set current_site_id = null where current_site_id = p_id;
   475→    update public.employee set is_team_leader = false, team_leader_site_id = null
   476→      where team_leader_site_id = p_id or (current_site_id = p_id and is_team_leader);
   477→  end if;
   478→
   479→  return jsonb_build_object('site', to_jsonb(result), 'unassignedCount', unassigned_count);
   480→end; $$;
   481→
   482→create or replace function public.delete_site(p_id uuid)
   483→returns void language plpgsql
   484→security definer set search_path = public
   485→as $$
   486→begin
   487→  if not public.has_menu_permission('sites') then raise exception 'Access denied'; end if;
   488→  update public.employee set current_site_id = null where current_site_id = p_id;
   489→  update public.employee set is_team_leader = false, team_leader_site_id = null where team_leader_site_id = p_id;
   490→  delete from public.site where id = p_id;
   491→end; $$;

   698→-- Fetch a month (lazily auto-creating missing records, never for future days)
   699→create or replace function public.get_month_attendance(p_month text)
   700→returns setof public.attendance language plpgsql
   701→security definer set search_path = public
   702→as $$
   703→declare
   704→  m_start date := to_date(p_month || '-01', 'YYYY-MM-DD');
   705→  m_end   date := (m_start + interval '1 month - 1 day')::date;
   706→  today   date := current_date;
   707→  last_day date;
   708→begin
   709→  if not public.has_menu_permission('attendance') and not public.has_menu_permission('employees') then
   710→    raise exception 'Access denied';
   711→  end if;
   712→
   713→  if today < m_start then
   714→    last_day := m_start - 1;  -- future month: create nothing
   715→  elsif today < m_end then
   716→    last_day := today;
   717→  else
   718→    last_day := m_end;
   719→  end if;
   720→
   721→  -- auto-create missing records: present for site employees, no_site for idle
   722→  insert into public.attendance (employee_id, date, status)
   723→  select e.id, d::date, case when e.current_site_id is null then 'no_site' else 'present' end
   724→  from public.employee e
   725→  cross join generate_series(m_start, last_day, interval '1 day') d
   726→  where e.status <> 'deleted'
   727→    and not exists (
   728→      select 1 from public.attendance a
   729→      where a.employee_id = e.id and a.date = d::date
   730→    );
   731→
   732→  return query
   733→    select a.* from public.attendance a
   734→    join public.employee e on e.id = a.employee_id
   735→    where to_char(a.date, 'YYYY-MM') = p_month
   736→      and not a.is_hidden
   737→      and e.status <> 'deleted'
   738→    order by a.date;
   739→end; $$;

   741→-- Mark one day (upsert) with rating engine + auto-warning on 3 consecutive absences
   742→create or replace function public.mark_attendance(
   743→  p_employee_id uuid, p_date date, p_status text, p_overtime_hours numeric default null
   744→)
   745→returns public.attendance language plpgsql
   746→security definer set search_path = public
   747→as $$
   748→declare
   749→  result public.attendance;
   750→  prev public.attendance;
   751→  ot_delta numeric := 0;
   752→  walk date;
   753→  absent_dates date[] := array[p_date];
   754→begin
   755→  if p_status not in ('present','absent','no_site','overtime','not_marked') then
   756→    raise exception 'Invalid attendance status';
   757→  end if;
   758→  if not public.has_menu_permission('attendance') and not public.has_menu_permission('employees') then
   759→    raise exception 'Access denied';
   760→  end if;
   761→
   762→  select * into prev from public.attendance
   763→    where employee_id = p_employee_id and date = p_date;
   764→  if prev.id is not null then
   765→    ot_delta := coalesce(p_overtime_hours, 0) - coalesce(prev.overtime_hours, 0);
   766→  end if;
   767→
   768→  insert into public.attendance (employee_id, date, status, overtime_hours)
   769→  values (p_employee_id, p_date, p_status,
   770→          case when p_status = 'overtime' then p_overtime_hours else null end)
   771→  on conflict (employee_id, date) do update set
   772→    status = excluded.status,
   773→    overtime_hours = case when excluded.status = 'overtime' then excluded.overtime_hours else null end
   774→  returning * into result;
   775→
   776→  -- rating: +/- 0.2 stars per overtime hour delta
   777→  if ot_delta <> 0 then
   778→    update public.employee set rating = public.clamp_rating(rating + 0.2 * ot_delta) where id = p_employee_id;
   779→  end if;
   780→
   781→  -- auto-warning: 3 consecutive absences (walking back, skipping Fridays)
   782→  if p_status = 'absent' then
   783→    walk := p_date - 1;
   784→    while walk >= p_date - 7 loop
   785→      if extract(isodow from walk) = 5 then walk := walk - 1; continue; end if;
   786→      exit when not exists (
   787→        select 1 from public.attendance
   788→        where employee_id = p_employee_id and date = walk and status = 'absent' and not is_hidden
   789→      );
   790→      absent_dates := walk || absent_dates;
   791→      walk := walk - 1;
   792→    end loop;
   793→
   794→    if array_length(absent_dates, 1) >= 3 then
   795→      insert into public.warning (employee_id, reason, is_auto_generated, absent_dates, created_by)
   796→      values (p_employee_id,
   797→              'Auto-generated: ' || array_length(absent_dates,1) || ' consecutive absences detected (' ||
   798→              (select string_agg(to_char(d,'YYYY-MM-DD'), ', ') from unnest(absent_dates) d) || ')',
   799→              true, to_jsonb(absent_dates), auth.uid());
   800→      update public.employee set rating = public.clamp_rating(rating - 0.5) where id = p_employee_id;
   801→      perform public.notify_super_admins(
   802→        'Auto-Generated Warning',
   803→        (select full_name || ' (' || employee_code || ') was absent 3+ consecutive days' from public.employee where id = p_employee_id),
   804→        'warning'
   805→      );
   806→    end if;
   807→  end if;
   808→
   809→  return result;
   810→end; $$;

   812→-- Full rating recompute from a month's attendance:
   813→-- (present + overtime days / working days excluding Fridays) * 5
   814→create or replace function public.recalc_employee_rating(p_employee_id uuid, p_month text)
   815→returns numeric language plpgsql
   816→security definer set search_path = public
   817→as $$
   818→declare
   819→  worked int;
   820→  workdays int;
   821→  new_rating numeric;
   822→begin
   823→  select count(*) into worked from public.attendance
   824→    where employee_id = p_employee_id and to_char(date,'YYYY-MM') = p_month
   825→      and status in ('present','overtime') and not is_hidden;
   826→  select count(*) into workdays from generate_series(
   827→    to_date(p_month || '-01','YYYY-MM-DD'),
   828→    least(current_date, (to_date(p_month || '-01','YYYY-MM-DD') + interval '1 month - 1 day')::date),
   829→    interval '1 day') d
   830→  where extract(isodow from d) <> 5;
   831→
   832→  new_rating := public.clamp_rating(case when workdays > 0 then (worked::numeric / workdays) * 5 else 5 end);
   833→  update public.employee set rating = new_rating where id = p_employee_id;
   834→  return new_rating;
   835→end; $$;

   837→-- Dashboard chart data: per-status counts for the last N months (incl current)
   838→create or replace function public.get_monthly_attendance_summary(p_months int default 6)
   839→returns table (month text, status text, count bigint) language plpgsql
   840→security definer set search_path = public
   841→as $$
   842→begin
   843→  if not public.is_staff() then raise exception 'Access denied'; end if;
   844→  return query
   845→    with months as (
   846→      select to_char(date_trunc('month', current_date) - (n || ' month')::interval, 'YYYY-MM') as m
   847→      from generate_series(0, p_months - 1) n
   848→    )
   849→    select months.m,
   850→           a.status,
   851→           count(*)::bigint
   852→    from months
   853→    join public.attendance a on to_char(a.date, 'YYYY-MM') = months.m
   854→    join public.employee e on e.id = a.employee_id
   855→    where not a.is_hidden and e.status <> 'deleted' and a.status <> 'not_marked'
   856→    group by months.m, a.status
   857→    order by months.m;
   858→end; $$;

   949→create or replace function public.issue_fine(p_employee_id uuid, p_reason text, p_amount numeric)

   963→  insert into public.fine (employee_id, reason, amount, created_by) values (p_employee_id, p_reason, p_amount, auth.uid())
   964→  returning * into result;
   965→
   966→  update public.employee set rating = public.clamp_rating(rating - 1.0) where id = p_employee_id;
   967→
   968→  perform public.notify_super_admins('New Fine Issued',
   969→    emp.full_name || ' (' || emp.employee_code || ') was fined SAR ' || p_amount::text || ': ' || p_reason, 'fine');
   970→  return result;
   971→end; $$;

   989→-- ============================================================================
   990→
   991→create or replace function public.create_uniform_entry(
   992→  p_employee_id uuid, p_document_type text, p_document_number text, p_items jsonb,
   993→  p_site_name text default null, p_team_leader_name text default null,
   994→  p_created_at timestamptz default now(), p_token_number int default null,
   995→  p_is_renewal boolean default false, p_previous_token_id int default null
   996→)
   997→returns public.uniform_registry language plpgsql
   998→security definer set search_path = public
   999→as $$
  1000→declare
  1001→  result public.uniform_registry;
  1002→  emp public.employee;
  1003→  v_uniform_id int;
  1004→  v_token int;
  1005→begin
  1006→  if not public.is_staff() then raise exception 'Access denied'; end if;
  1007→  if p_document_type not in ('id','passport') then raise exception 'Invalid document type'; end if;
  1008→  select * into emp from public.employee where id = p_employee_id;
  1009→  if emp.id is null then raise exception 'Employee not found'; end if;
  1010→
  1011→  select coalesce(max(uniform_id),0) + 1 into v_uniform_id from public.uniform_registry;
  1012→  select coalesce(max(token_number),0) + 1 into v_token from public.uniform_registry;
  1013→  if p_token_number is not null and p_token_number > 0 then v_token := p_token_number; end if;
  1014→  if exists (select 1 from public.uniform_registry where uniform_id = v_uniform_id) then
  1015→    v_uniform_id := v_uniform_id + 1;
  1016→  end if;
  1017→
  1018→  insert into public.uniform_registry
  1019→    (uniform_id, token_number, employee_id, employee_name, document_type, document_number,
  1020→     items, site_name, team_leader_name, is_renewal, previous_token_id, created_at, renewal_date)
  1021→  values
  1022→    (v_uniform_id, v_token, p_employee_id, emp.full_name, p_document_type, p_document_number,
  1023→     p_items, nullif(p_site_name,''), nullif(p_team_leader_name,''), p_is_renewal, p_previous_token_id,
  1024→     p_created_at, p_created_at + interval '6 months')
  1025→  returning * into result;
  1026→  return result;
  1027→end; $$;
  1028→
  1029→create or replace function public.update_uniform_entry(
  1030→  p_id uuid, p_token_number int, p_document_type text, p_document_number text,
  1031→  p_items jsonb, p_site_name text default null, p_team_leader_name text default null,
  1032→  p_created_at timestamptz
  1033→)
  1034→returns public.uniform_registry language plpgsql
  1035→security definer set search_path = public
  1036→as $$
  1037→declare result public.uniform_registry;
  1038→begin
  1039→  if not public.is_staff() then raise exception 'Access denied'; end if;
  1040→  if p_document_type not in ('id','passport') then raise exception 'Invalid document type'; end if;
  1041→  update public.uniform_registry set
  1042→    token_number = p_token_number,
  1043→    document_type = p_document_type,
  1044→    document_number = p_document_number,
  1045→    items = p_items,
  1046→    site_name = nullif(p_site_name,''),
  1047→    team_leader_name = nullif(p_team_leader_name,''),
  1048→    created_at = p_created_at,
  1049→    renewal_date = p_created_at + interval '6 months'
  1050→  where id = p_id and not is_deleted and not is_hidden
  1051→  returning * into result;
  1052→  if result.id is null then raise exception 'Entry not found'; end if;
  1053→  return result;
  1054→end; $$;

  1057→returns void language plpgsql
