    99→    99→// Generate all dates of a given month (up to today)
   100→   100→function getMonthDates(yearNum: number, monthNum: number) {
   101→   101→  const now = new Date();
   102→   102→  const today = now.getFullYear() * 10000 + (now.getMonth() + 1) * 100 + now.getDate();
   103→   103→  const daysInMonth = new Date(yearNum, monthNum, 0).getDate();
   104→   104→  const dates: { value: string; label: string; dayNum: number }[] = [];
   105→   105→
   106→   106→  for (let d = 1; d <= daysInMonth; d++) {
   107→   107→    const dateKey = `${yearNum}-${String(monthNum).padStart(2, '0')}-${String(d).padStart(2, '0')}`;
   108→   108→    const dateNum = yearNum * 10000 + monthNum * 100 + d;
   109→   109→    const isFuture = dateNum > today;
   110→   110→    const dayOfWeek = new Date(yearNum, monthNum - 1, d).getDay();
   111→   111→    const isFriday = dayOfWeek === 5;
   112→   112→    const dayName = new Date(yearNum, monthNum - 1, d).toLocaleDateString('en-US', { weekday: 'short' });
   113→   113→
   114→   114→    dates.push({
   115→   115→      value: dateKey,
   116→   116→      label: `${dayName}, ${format(new Date(yearNum, monthNum - 1, d), 'MMM d')}${isFuture ? ' (upcoming)' : isFriday ? ' (Fri)' : ''}`,
   117→   117→      dayNum: d,
   118→   118→    });
   119→   119→  }
   120→   120→
   121→   121→  return dates;
   122→   122→}
