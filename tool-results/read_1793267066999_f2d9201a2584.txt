    50→    50→// client-side photo compression (max 300px, JPEG 0.8)
    51→    51→function compressPhoto(file: File): Promise<string> {
    52→    52→  return new Promise((resolve, reject) => {
    53→    53→    const reader = new FileReader();
    54→    54→    reader.onload = () => {
    55→    55→      const img = new window.Image();
    56→    56→      img.onload = () => {
    57→    57→        const scale = Math.min(1, 300 / Math.max(img.width, img.height));
    58→    58→        const canvas = document.createElement('canvas');
    59→    59→        canvas.width = Math.round(img.width * scale);
    60→    60→        canvas.height = Math.round(img.height * scale);
    61→    61→        const ctx = canvas.getContext('2d');
    62→    62→        if (!ctx) return reject(new Error('Canvas unavailable'));
    63→    63→        ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
    64→    64→        resolve(canvas.toDataURL('image/jpeg', 0.8));
    65→    65→      };
    66→    66→      img.onerror = () => reject(new Error('Invalid image'));
    67→    67→      img.src = reader.result as string;
    68→    68→    };
    69→    69→    reader.onerror = () => reject(new Error('Failed to read file'));
    70→    70→    reader.readAsDataURL(file);
    71→    71→  });
    72→    72→}
