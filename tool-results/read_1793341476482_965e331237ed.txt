    40→const MONTH_NAMES = ['January', 'February', 'March', 'April', 'May', 'June',
    41→  'July', 'August', 'September', 'October', 'November', 'December'];
    42→const WEEKDAYS = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
    43→const EASE: [number, number, number, number] = [0.22, 1, 0.36, 1];
    44→
    45→const pad = (n: number): string => String(n).padStart(2, '0');
    46→const ymd = (y: number, m: number, d: number): string => `${y}-${pad(m)}-${pad(d)}`;
    47→const todayYMD = (): string => {
    48→  const t = new Date();
    49→  return ymd(t.getFullYear(), t.getMonth() + 1, t.getDate());
    50→};
    51→const isSunday = (date: string): boolean => new Date(`${date}T00:00:00`).getDay() === 0;
    52→const daysInMonth = (y: number, m: number): number => new Date(y, m, 0).getDate();
    53→
    54→/** Parse a cell value → { kind, hours }. kind: 'absent' | 'hours' | 'clear' | 'invalid' */
    55→function parseCell(raw: string): { kind: 'absent' | 'hours' | 'clear' | 'invalid'; hours: number | null } {
    56→  const v = raw.trim();
    57→  if (v === '') return { kind: 'clear', hours: null };
    58→  if (v.toLowerCase() === 'a' || v.toLowerCase() === 'absent') return { kind: 'absent', hours: null };
    59→  const n = Number(v);
    60→  if (Number.isNaN(n) || !Number.isFinite(n) || n < 0 || n > 24) return { kind: 'invalid', hours: null };
    61→  return { kind: 'hours', hours: Math.round(n * 100) / 100 };
    62→}

    78→export function AttendancePage() {
    79→  const realtimeVersion = useRealtimeVersion(['attendance', 'employee', 'site', 'camp', 'admin_site']);
    80→  const user = useAuthStore((s) => s.user);
    81→  const isSuperAdmin = user?.role === 'super_admin';
    82→
    83→  const now = new Date();
    84→  const [monthKey, setMonthKey] = useState(`${now.getFullYear()}-${pad(now.getMonth() + 1)}`);
    85→  const [view, setView] = useState<'sheet' | 'calendar'>('sheet');
    86→
    87→  const [employees, setEmployees] = useState<EmployeeVM[] | null>(null);
    88→  const [sites, setSites] = useState<SiteWithCount[] | null>(null);
    89→  const [records, setRecords] = useState<Attendance[] | null>(null);
    90→  const [assignedSiteIds, setAssignedSiteIds] = useState<Set<string> | null>(isSuperAdmin ? null : new Set());
    91→
    92→  // filters
    93→  const [search, setSearch] = useState('');
    94→  const [debouncedSearch, setDebouncedSearch] = useState('');
    95→  const [siteFilter, setSiteFilter] = useState<string>('all');
    96→
    97→  // editing state: cellKey "empId|YYYY-MM-DD" → raw input text
    98→  const [edits, setEdits] = useState<Record<string, string>>({});
    99→  const [savingKeys, setSavingKeys] = useState<Set<string>>(new Set());
   100→
   101→  // calendar view state
   102→  const [calendarEmployeeId, setCalendarEmployeeId] = useState<string | null>(null);
   103→
   104→  const [year, month] = useMemo(() => {
   105→    const [y, m] = monthKey.split('-').map(Number);
   106→    return [y, m];
   107→  }, [monthKey]);
   108→
   109→  const today = todayYMD();
   110→  const days = daysInMonth(year, month);
   111→
   112→  // ---- data loading ----------------------------------------------------------
   113→  useEffect(() => {
   114→    let cancelled = false;
   115→    (async () => {
   116→      try {
   117→        const [emps, sts] = await Promise.all([dataApi.listEmployees(), dataApi.listSites()]);
   118→        if (!cancelled) { setEmployees(emps); setSites(sts); }
   119→      } catch (err) {
   120→        if (!cancelled) toast.error(err instanceof Error ? err.message : 'Failed to load');
   121→      }
   122→    })();
   123→    return () => { cancelled = true; };
   124→  }, [realtimeVersion]);
   125→
   126→  // admins: which sites they may mark (null = unrestricted super admin)
   127→  useEffect(() => {
   128→    if (isSuperAdmin) { setAssignedSiteIds(null); return; }
   129→    if (!user) return;
   130→    let cancelled = false;
   131→    dataApi.getSiteAssignments(user.id)
   132→      .then((ids) => { if (!cancelled) setAssignedSiteIds(new Set(ids)); })
   133→      .catch(() => { if (!cancelled) setAssignedSiteIds(new Set()); });
   134→    return () => { cancelled = true; };
   135→  }, [user, isSuperAdmin, realtimeVersion]);
   136→
   137→  useEffect(() => {
   138→    let cancelled = false;
   139→    dataApi.getMonthAttendance(monthKey)
   140→      .then((rows) => { if (!cancelled) setRecords(rows); })
   141→      .catch(() => { if (!cancelled) setRecords([]); });
   142→    return () => { cancelled = true; };
   143→  }, [monthKey, realtimeVersion]);
   144→
   145→  useEffect(() => {
   146→    const t = setTimeout(() => setDebouncedSearch(search), 300);
   147→    return () => clearTimeout(t);
   148→  }, [search]);
   149→
   150→  // ---- scoped crew (admins → their assigned sites only; super admins → all) ---
   151→  const scoped = useMemo(() => {
   152→    const list = (employees ?? []).filter((e) => e.status !== 'deleted');
   153→    if (assignedSiteIds === null) return list;
   154→    return list.filter((e) => e.current_site_id !== null && assignedSiteIds.has(e.current_site_id));
   155→  }, [employees, assignedSiteIds]);
   156→
   157→  // ---- sitewise sections (alphabetical inside each section) -------------------
   158→  const sections = useMemo<SiteSection[]>(() => {
   159→    const q = debouncedSearch.toLowerCase().trim();
   160→    const matches = (e: EmployeeVM) =>
   161→      !q || `${e.full_name} ${e.employee_code} ${e.position ?? ''} ${e.nationality ?? ''}`.toLowerCase().includes(q);
   162→    const byName = (a: EmployeeVM, b: EmployeeVM) => a.full_name.localeCompare(b.full_name);
   163→
   164→    const siteList = (sites ?? []).filter((s) => s.is_active);
   165→    const result: SiteSection[] = siteList
   166→      .filter((s) => siteFilter === 'all' || s.id === siteFilter)
   167→      .map((s) => ({
   168→        key: s.id,
   169→        title: s.name,
   170→        client: s.client_name,
   171→        emps: scoped
   172→          .filter((e) => e.current_site_id === s.id && matches(e))
   173→          .sort(byName),
   174→      }))
   175→      .filter((sec) => sec.emps.length > 0);
   176→
   177→    // idle employees — super admins only (admins can never mark them)
   178→    if (isSuperAdmin && (siteFilter === 'all' || siteFilter === 'idle')) {
   179→      const idle = scoped
   180→        .filter((e) => !e.current_site_id && matches(e))
   181→        .sort(byName);
   182→      if (idle.length > 0) {
   183→        result.push({
   184→          key: '__idle__',
   185→          title: 'Unassigned / Idle — no site',
   186→          client: null,
   187→          emps: idle,
   188→          isIdle: true,
   189→        });
   190→      }
   191→    }
   192→    return result;
   193→  }, [sites, scoped, siteFilter, debouncedSearch, isSuperAdmin]);
   194→
   195→  // flat crew for the calendar view combobox
   196→  const crew = useMemo(
   197→    () => sections.flatMap((s) => s.emps),
   198→    [sections],
   199→  );
   200→
   201→  // sites selectable in the filter (admins: only their assigned sites)
   202→  const selectableSites = useMemo(() => {
   203→    const active = (sites ?? []).filter((s) => s.is_active);
   204→    if (assignedSiteIds === null) return active;
   205→    return active.filter((s) => assignedSiteIds.has(s.id));
   206→  }, [sites, assignedSiteIds]);
   207→
   208→  // ---- record lookups ---------------------------------------------------------
   209→  const recordMap = useMemo(() => {
   210→    const map = new Map<string, Attendance>();
   211→    (records ?? []).forEach((r) => map.set(`${r.employee_id}|${r.date}`, r));
   212→    return map;
   213→  }, [records]);
   214→
   215→  const recordToText = (r: Attendance | undefined): string => {
   216→    if (!r) return '';
   217→    if (r.status === 'absent') return 'A';
   218→    if (r.status === 'present' && r.hours != null) return String(r.hours);
   219→    return '';
   220→  };
   221→
   222→  const cellText = (empId: string, date: string): string => {
   223→    const key = `${empId}|${date}`;
   224→    return edits[key] ?? recordToText(recordMap.get(key));
   225→  };
   226→
   227→  // ---- commit a cell ------------------------------------------------------------
   228→  const commitCell = async (empId: string, date: string) => {
   229→    const key = `${empId}|${date}`;
   230→    const raw = edits[key];
   231→    if (raw === undefined) return; // untouched
   232→
   233→    const current = recordToText(recordMap.get(key));
   234→    if (raw === current) { // no change → drop local edit
   235→      setEdits((prev) => { const n = { ...prev }; delete n[key]; return n; });
   236→      return;
   237→    }
   238→
   239→    const parsed = parseCell(raw);
   240→    if (parsed.kind === 'invalid') {
   241→      toast.error('Invalid value', { description: 'Enter hours (0–24, e.g. 8 or 10.5) or A for absent.' });
   242→      setEdits((prev) => { const n = { ...prev }; delete n[key]; return n; });
   243→      return;
   244→    }
   245→
   246→    setSavingKeys((prev) => new Set(prev).add(key));
   247→    try {
   248→      await dataApi.markAttendance(empId, date, parsed.kind === 'clear' ? '' : raw.trim());
   249→      // optimistic local sync (realtime refetch will reconcile)
   250→      setRecords((prev) => {
   251→        const list = prev ?? [];
   252→        const rest = list.filter((r) => !(r.employee_id === empId && r.date === date));
   253→        if (parsed.kind === 'clear') return rest;
   254→        const base = list.find((r) => r.employee_id === empId && r.date === date);
   255→        return [...rest, {
   256→          id: base?.id ?? `local-${key}`,
   257→          employee_id: empId,
   258→          date,
   259→          status: parsed.kind === 'absent' ? 'absent' : 'present',
   260→          hours: parsed.hours,
   261→          is_hidden: false,
   262→          created_at: base?.created_at ?? new Date().toISOString(),
   263→        }];
   264→      });
   265→      setEdits((prev) => { const n = { ...prev }; delete n[key]; return n; });
   266→    } catch (err) {
   267→      toast.error(err instanceof Error ? err.message : 'Failed to save');
   268→      setEdits((prev) => { const n = { ...prev }; delete n[key]; return n; });
   269→    } finally {
   270→      setSavingKeys((prev) => { const n = new Set(prev); n.delete(key); return n; });
   271→    }
   272→  };
   273→
   274→  const setCell = (empId: string, date: string, value: string) => {
   275→    const key = `${empId}|${date}`;
   276→    setEdits((prev) => ({ ...prev, [key]: value }));
   277→  };
   278→
   279→  // ---- row totals (TOTAL HOURS + ABSENCES) ---------------------------------------
   280→  const rowTotals = (empId: string): { hours: number; absents: number } => {
   281→    let hours = 0;
   282→    let absents = 0;
   283→    for (let d = 1; d <= days; d++) {
   284→      const date = ymd(year, month, d);
   285→      const parsed = parseCell(cellText(empId, date));
   286→      if (parsed.kind === 'hours') hours += parsed.hours ?? 0;
   287→      if (parsed.kind === 'absent') absents += 1;
   288→    }
   289→    return { hours: Math.round(hours * 100) / 100, absents };
   290→  };
   291→
   292→  // ---- month navigation ----------------------------------------------------------
   293→  const shiftMonth = (delta: number) => {
   294→    const d = new Date(year, month - 1 + delta, 1);
   295→    setMonthKey(`${d.getFullYear()}-${pad(d.getMonth() + 1)}`);
   296→  };
   297→
   298→  const loading = !employees || !records || !sites || (!isSuperAdmin && assignedSiteIds === null);
   299→  const calendarEmployee = crew.find((e) => e.id === calendarEmployeeId) ?? null;
   300→
   301→  const dayColumns = useMemo(() => {
   302→    // always show the FULL month (1 … last day): un-entered days stay blank,
   303→    // future days are rendered but locked against editing.
   304→    const cols: number[] = [];
   305→    for (let d = 1; d <= days; d++) cols.push(d);
   306→    return cols;
   307→  }, [days]);
   308→
   309→  // ---- one day cell (plain render fn — keeps input focus stable) ------------
   310→  const renderDayCell = (emp: EmployeeVM, date: string) => {
   311→    const key = `${emp.id}|${date}`;
   312→    const value = cellText(emp.id, date);
   313→    const busy = savingKeys.has(key);
   314→    const sun = isSunday(date);
   315→    const future = date > today;
   316→    const isA = value.trim().toLowerCase() === 'a';
   317→    return (
   318→      <td
   319→        key={key}
   320→        className={cn(
   321→          'border-x border-b border-foreground/15 p-0 text-center',
   322→          sun && 'bg-rose-500/[0.05]',
   323→          !sun && !future && date > new Date(Date.now() - 7 * 86400000).toISOString().slice(0, 10) && 'bg-emerald-500/[0.04]',
   324→          future && 'bg-muted/40',
   325→        )}
   326→      >
   327→        {future ? (
   328→          // future dates are visible but locked — nothing to mark yet
   329→          <div
   330→            className="flex h-9 w-10 cursor-not-allowed items-center justify-center"
   331→            title={`${date} — future date, not editable yet`}
   332→            aria-label={`${emp.full_name} — ${date} — future date, locked`}
   333→          />
   334→        ) : (
   335→          <div className="relative">
   336→            <input
   337→              value={value}
   338→              disabled={busy}
   339→              onChange={(e) => setCell(emp.id, date, e.target.value)}
   340→              onBlur={() => commitCell(emp.id, date)}
   341→              onKeyDown={(e) => { if (e.key === 'Enter') (e.target as HTMLInputElement).blur(); }}
   342→              aria-label={`${emp.full_name} — ${date}`}
   343→              className={cn(
   344→                'h-9 w-10 bg-transparent text-center text-[12px] font-medium tabular-nums outline-none transition-colors',
   345→                'hover:bg-brand/10 focus:bg-brand/15 focus:ring-1 focus:ring-brand/60 focus:ring-inset',
   346→                isA && 'bg-rose-500/10 font-bold text-rose-500',
   347→                value !== '' && !isA && 'text-foreground',
   348→                busy && 'opacity-50',
   349→              )}
   350→            />
   351→            {busy && (
   352→              <span className="pointer-events-none absolute inset-0 flex items-center justify-center">
   353→                <Loader2 size={10} className="animate-spin text-brand" />
   354→              </span>
   355→            )}
   356→          </div>
   357→        )}
   358→      </td>
   359→    );
   360→  };
   361→
   362→  // ---- a full site section (table with that site's crew) ---------------------------
   363→  const renderSheetSection = (section: SiteSection, index: number) => {
   364→    const totalHours = section.emps.reduce((sum, emp) => sum + rowTotals(emp.id).hours, 0);
   365→    return (
   366→      <motion.section
   367→        initial={{ opacity: 0, y: 14 }}
   368→        animate={{ opacity: 1, y: 0 }}
   369→        transition={{ duration: 0.35, delay: Math.min(index * 0.05, 0.3), ease: EASE }}
   370→        className="space-y-2"
   371→      >
   372→        {/* section band */}
   373→        <div className="flex flex-wrap items-center gap-x-3 gap-y-1.5 rounded-xl border bg-card px-4 py-2.5">
   374→          <span
   375→            className={cn(
   376→              'flex size-8 shrink-0 items-center justify-center rounded-lg',
   377→              section.isIdle ? 'bg-amber-500/10 text-amber-500' : 'bg-brand/10 text-brand',
   378→            )}
   379→          >
   380→            {section.isIdle ? <UserX size={16} /> : <Building2 size={16} />}
   381→          </span>
   382→          <div className="min-w-0">
   383→            <h3 className="truncate text-sm font-semibold">{section.title}</h3>
   384→            {section.client && (
   385→              <p className="truncate text-[11px] text-muted-foreground">Client: {section.client}</p>
   386→            )}
   387→          </div>
   388→          <div className="ml-auto flex items-center gap-2">
   389→            <span
   390→              className={cn(
   391→                'rounded-full px-2.5 py-0.5 text-[11px] font-semibold tabular-nums',
   392→                section.isIdle
   393→                  ? 'border border-amber-500/25 bg-amber-500/10 text-amber-600 dark:text-amber-400'
   394→                  : 'border border-brand/25 bg-brand/10 text-brand',
   395→              )}
   396→            >
   397→              {section.emps.length} employee{section.emps.length !== 1 ? 's' : ''}
   398→            </span>
   399→            {section.emps.length > 0 && (
   400→              <span className="hidden rounded-full border border-emerald-500/25 bg-emerald-500/10 px-2.5 py-0.5 text-[11px] font-semibold tabular-nums text-emerald-500 sm:inline-flex">
   401→                Site total: {Math.round(totalHours * 100) / 100}h
   402→              </span>
   403→            )}
   404→          </div>
   405→        </div>
   406→
   407→        {/* crew table */}
   408→        <div className="card-glow overflow-hidden rounded-xl border bg-card">
   409→          <div className="overflow-x-auto">
   410→            <table className="w-full min-w-max border-collapse text-sm">
   411→              <thead>
   412→                <tr className="border-b bg-muted/60 text-left text-[11px] uppercase tracking-wide text-muted-foreground">
   413→                  <th className="sticky left-0 z-20 w-[240px] min-w-[240px] border-x border-foreground/15 bg-muted/60 px-4 py-2.5 font-medium">Employee</th>
   414→                  <th className="sticky left-[240px] z-20 w-[120px] min-w-[120px] border-x border-b border-foreground/15 bg-muted/60 px-3 py-2.5 font-medium">Trade</th>
   415→                  {dayColumns.map((d) => {
   416→                    const date = ymd(year, month, d);
   417→                    const sun = isSunday(date);
   418→                    const future = date > today;
   419→                    return (
   420→                      <th
   421→                        key={d}
   422→                        className={cn(
   423→                          'w-10 min-w-10 border-x border-b border-foreground/15 px-1 py-2.5 text-center font-medium',
   424→                          sun && 'bg-rose-500/[0.07] text-rose-500',
   425→                          future && 'text-muted-foreground/50',
   426→                        )}
   427→                        title={`${MONTH_NAMES[month - 1]} ${d}${sun ? ' (Sunday)' : ''}${future ? ' — future (locked)' : ''}`}
   428→                      >
   429→                        <div>{d}</div>
   430→                        <div className={cn('text-[9px] font-normal normal-case opacity-70', sun && 'text-rose-500')}>
   431→                          {WEEKDAYS[new Date(`${date}T00:00:00`).getDay()]}
   432→                        </div>
   433→                      </th>
   434→                    );
   435→                  })}
   436→                  <th className="sticky right-[92px] z-20 w-[92px] border-x border-b border-foreground/15 bg-muted/60 px-2 py-2.5 text-center font-medium text-emerald-500">Total H</th>
   437→                  <th className="sticky right-0 z-20 w-[92px] border-l border-b border-foreground/15 bg-muted/60 px-2 py-2.5 text-center font-medium text-rose-500">Absent</th>
   438→                </tr>
   439→              </thead>
   440→              <tbody>
   441→                {section.emps.map((emp, i) => {
   442→                  const totals = rowTotals(emp.id);
   443→                  return (
   444→                    <motion.tr
   445→                      key={emp.id}
   446→                      initial={{ opacity: 0, y: 6 }}
   447→                      animate={{ opacity: 1, y: 0 }}
   448→                      transition={{ delay: Math.min(i * 0.02, 0.25), ease: EASE }}
   449→                      className="transition-colors"
   450→                    >
   451→                      {/* employee — stacked: name / employee ID / camp */}
   452→                      <td className="sticky left-0 z-10 border-x border-b border-foreground/15 bg-card px-4 py-2">
   453→                        <div className="flex items-center gap-2.5">
   454→                          <span className="flex size-8 shrink-0 items-center justify-center overflow-hidden rounded-full bg-brand/15 text-[10px] font-bold text-brand">
   455→                            {emp.photo ? <img src={emp.photo} alt={emp.full_name} className="size-full object-cover" /> : initials(emp.full_name)}
   456→                          </span>
   457→                          <div className="min-w-0">
