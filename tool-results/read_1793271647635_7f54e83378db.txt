   125→const nextEmployeeCode = (): string => {
   126→  const year = new Date().getFullYear();
   127→  const prefix = `PUL-${year}-`;
   128→  const max = db.employees.reduce((m, e) => {
   129→    if (e.employee_code.startsWith(prefix)) {
   130→      const n = parseInt(e.employee_code.slice(prefix.length), 10);
   131→      return Number.isNaN(n) ? m : Math.max(m, n);
   132→    }
   133→    return m;
   134→  }, 0);
   135→  return `${prefix}${String(max + 1).padStart(3, '0')}`;
   136→};

   201→    const p = db.profiles.find((x) => x.id === id);
   202→    if (!p) throw new Error('Profile not found');
   203→    if (input.name !== undefined) p.name = input.name;
   204→    if (input.email !== undefined) p.email = input.email.toLowerCase();
   205→    if (input.password) p.password = input.password;
   206→    if (input.role !== undefined) {
   207→      if (p.role === 'super_admin' && input.role !== 'super_admin') {
   208→        const superCount = db.profiles.filter((x) => x.role === 'super_admin').length;
   209→        if (superCount <= 1) throw new Error('Cannot demote the last super admin');
   210→      }
   211→      p.role = input.role;
   212→    }
   213→    saveAndEmit(['profiles']);
   214→    return stripPassword(p);
   215→  },

   459→  async deleteEmployeeCascade(id: string): Promise<void> {
   460→    const e = db.employees.find((x) => x.id === id);
   461→    if (!e) throw new Error('Employee not found');
   462→    if (e.status === 'pending_deletion') throw new Error('This employee already has a pending deletion request');
   463→
   464→    db.attendance.filter((a) => a.employee_id === id).forEach((a) => { a.is_hidden = true; });
   465→    db.warnings.filter((w) => w.employee_id === id).forEach((w) => { w.is_hidden = true; });
   466→    db.fines.filter((f) => f.employee_id === id).forEach((f) => { f.is_hidden = true; });
   467→    db.leaveRequests.filter((r) => r.employee_id === id).forEach((r) => { r.is_hidden = true; });
   468→    db.cancellationRequests.filter((r) => r.employee_id === id).forEach((r) => { r.is_hidden = true; });
   469→    db.uniforms.filter((u) => u.employee_id === id).forEach((u) => { u.is_deleted = true; u.is_hidden = true; });
   470→    const nameLc = e.full_name.toLowerCase();
   471→    const codeLc = e.employee_code.toLowerCase();
   472→    db.notifications.filter((n) =>
   473→      n.title.toLowerCase().includes(nameLc)
   474→      || n.message.toLowerCase().includes(nameLc)
   475→      || n.message.toLowerCase().includes(codeLc),
   476→    ).forEach((n) => { n.is_hidden = true; });
   477→
   478→    e.status = 'deleted';
   479→    e.current_site_id = null;
   480→    e.is_team_leader = false;
   481→    e.team_leader_site_id = null;
   482→    saveAndEmit(['employee', 'attendance', 'warning', 'fine', 'leave_request', 'cancellation_request', 'uniform_registry', 'notification']);
   483→  },

   506→  async getMonthAttendance(month: string): Promise<Attendance[]> {
   507→    const today = toYMD(new Date());
   508→    const [y, m] = month.split('-').map(Number);
   509→    const first = `${month}-01`;
   510→    const monthEnd = toYMD(new Date(y, m, 0));
   511→    const lastDay = today < first ? null : today < monthEnd ? today : monthEnd;
   512→
   513→    // auto-create missing records for non-deleted employees
   514→    if (lastDay) {
   515→      for (const e of db.employees) {
   516→        if (e.status === 'deleted') continue;
   517→        for (let d = fromYMD(first); toYMD(d) <= lastDay; d.setDate(d.getDate() + 1)) {
   518→          const dateStr = toYMD(d);
   519→          const exists = db.attendance.some((a) => a.employee_id === e.id && a.date === dateStr);
   520→          if (!exists) {
   521→            db.attendance.push({
   522→              id: uid(), employee_id: e.id, date: dateStr,
   523→              status: e.current_site_id ? 'present' : 'not_marked',
   524→              hours: e.current_site_id ? 8 : null,
   525→              is_hidden: false, created_at: nowISO(),
   526→            });
   527→          }
   528→        }
   529→      }
   530→      saveAndEmit(['attendance']);
   531→    }
   532→
   533→    return db.attendance.filter((a) => {
   534→      if (!a.date.startsWith(month) || a.is_hidden) return false;
   535→      const e = db.employees.find((x) => x.id === a.employee_id);
   536→      return e ? e.status !== 'deleted' : false;
   537→    });
   538→  },

   540→  /** Excel-style marking: value is 'A' (absent), a numeric hours string ('8',
   541→   *  '10.5') or '' / null to clear the cell. Mirrors the Postgres RPC exactly. */
   542→  async markAttendance(employeeId: string, date: string, value: string | null): Promise<void> {
   543→    const e = db.employees.find((x) => x.id === employeeId);
   544→    if (!e) throw new Error('Employee not found');
   545→
   546→    const trimmed = (value ?? '').trim();
   547→    const prev = db.attendance.find((a) => a.employee_id === employeeId && a.date === date);
   548→
   549→    // cleared cell → remove the record
   550→    if (trimmed === '') {
   551→      if (prev) {
   552→        db.attendance = db.attendance.filter((a) => a.id !== prev.id);
   553→        saveAndEmit(['attendance']);
   554→      }
   555→      return;
   556→    }
   557→
   558→    let status: Attendance['status'];
   559→    let hours: number | null = null;
   560→    if (trimmed.toLowerCase() === 'a' || trimmed.toLowerCase() === 'absent') {
   561→      status = 'absent';
   562→    } else {
   563→      const n = Number(trimmed);
   564→      if (Number.isNaN(n) || !Number.isFinite(n)) throw new Error(`Invalid value "${trimmed}". Enter hours (e.g. 8, 10.5) or A for absent.`);
   565→      if (n < 0 || n > 24) throw new Error('Hours must be between 0 and 24');
   566→      status = 'present';
   567→      hours = Math.round(n * 100) / 100;
   568→    }
   569→
   570→    const otDelta = prev
   571→      ? Math.max(0, (hours ?? 0) - 8) - Math.max(0, (prev.hours ?? 0) - 8)
   572→      : Math.max(0, (hours ?? 0) - 8);
   573→
   574→    if (prev) {
   575→      prev.status = status;
   576→      prev.hours = hours;
   577→    } else {
   578→      db.attendance.push({
   579→        id: uid(), employee_id: employeeId, date, status, hours,
   580→        is_hidden: false, created_at: nowISO(),
   581→      });
   582→    }
   583→
   584→    // rating: ±0.2 stars per overtime-hour delta (hours above 8/day)
   585→    if (otDelta !== 0) {
   586→      e.rating = clampRating(e.rating + 0.2 * otDelta);
   587→    }
   588→
   589→    // auto-warning on 3+ consecutive absences (skipping Fridays)
   590→    if (status === 'absent') {
   591→      const absentDates: string[] = [date];
   592→      let walk = addDaysStr(date, -1);
   593→      const floorDate = addDaysStr(date, -7);
   594→      while (walk >= floorDate) {
   595→        if (isFridayStr(walk)) { walk = addDaysStr(walk, -1); continue; }
   596→        const rec = db.attendance.find((a) => a.employee_id === employeeId && a.date === walk && !a.is_hidden);
   597→        if (rec && rec.status === 'absent') {
   598→          absentDates.unshift(walk);
   599→          walk = addDaysStr(walk, -1);
   600→        } else break;
   601→      }
   602→      if (absentDates.length >= 3) {
   603→        db.warnings.push({
   604→          id: uid(), employee_id: employeeId,
   605→          reason: `Auto-generated: ${absentDates.length} consecutive absences detected (${absentDates.join(', ')})`,
   606→          is_auto_generated: true, absent_dates: absentDates,
   607→          created_by: currentProfile()?.id ?? null, is_hidden: false, created_at: nowISO(),
   608→        });
   609→        e.rating = clampRating(e.rating - 0.5);
   610→        notifySuperAdmins(
   611→          'Auto-Generated Warning',
   612→          `${e.full_name} (${e.employee_code}) was absent 3+ consecutive days`,
   613→          'warning',
   614→        );
   615→        saveAndEmit(['warning', 'notification', 'employee', 'attendance']);
   616→        return;
   617→      }
   618→    }
   619→    saveAndEmit(['attendance', 'employee']);
   620→  },

   622→  async getMonthlyAttendanceSummary(months = 6): Promise<MonthlySummaryRow[]> {
   623→    const rows: MonthlySummaryRow[] = [];
   624→    const now = new Date();
   625→    for (let back = months - 1; back >= 0; back--) {
   626→      const d = new Date(now.getFullYear(), now.getMonth() - back, 1);
   627→      const m = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
   628→      const counts: Record<string, number> = {};
   629→      for (const a of db.attendance) {
   630→        if (!a.date.startsWith(m) || a.is_hidden || a.status === 'not_marked') continue;
   631→        const e = db.employees.find((x) => x.id === a.employee_id);
   632→        if (!e || e.status === 'deleted') continue;
   633→        counts[a.status] = (counts[a.status] ?? 0) + 1;
   634→      }
   635→      for (const [status, count] of Object.entries(counts)) {
   636→        rows.push({ month: m, status: status as Attendance['status'], count });
   637→      }
   638→    }
   639→    return rows;
   640→  },
