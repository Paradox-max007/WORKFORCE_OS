    39→export const todayYMD = (): string => {
    40→  const d = new Date();
    41→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
    42→};

    47→/** Build a full UNIFORM_ITEMS-shaped record from a sparse checkbox state. */
    48→export const buildItemsRecord = (items: Record<string, boolean>): Record<string, boolean> =>
    49→  Object.fromEntries(UNIFORM_ITEMS.map((i) => [i.key, items[i.key] === true]));

    51→/** "Expiry" live-preview helper: created date + 6 months, formatted 'd MMM yyyy'. */
    52→export const formatExpiryPreview = (createdYMD: string): string | null => {
    53→  if (!createdYMD) return null;
    54→  const d = new Date(createdYMD);
    55→  if (Number.isNaN(d.getTime())) return null;
    56→  const expiry = new Date(d);
    57→  expiry.setMonth(expiry.getMonth() + 6);
    58→  if (Number.isNaN(expiry.getTime())) return null;
    59→  return format(expiry, 'd MMM yyyy');
    60→};
    61→
    62→const isoToYMD = (iso: string): string => iso.slice(0, 10);

   107→/** Renewal status vs renewal_date: expired → emerald, ≤30 days → amber, else slate. */
   108→export function RenewalBadge({ renewalDate }: { renewalDate: string }) {
   109→  const days = daysUntil(renewalDate);
   110→  if (days < 0) {
   111→    return (
   112→      <span className="inline-flex items-center whitespace-nowrap rounded-full border border-emerald-500/25 bg-emerald-500/10 px-2 py-0.5 text-[11px] font-medium text-emerald-500">
   113→        Renewal Available
   114→      </span>
   115→    );
   116→  }
   117→  if (days <= 30) {
   118→    return (
   119→      <span className="inline-flex items-center whitespace-nowrap rounded-full border border-amber-500/25 bg-amber-500/10 px-2 py-0.5 text-[11px] font-medium text-amber-500">
   120→        {days} days remaining
   121→      </span>
   122→    );
   123→  }
   124→  return (
   125→    <span className="inline-flex items-center whitespace-nowrap rounded-full border border-slate-500/25 bg-slate-500/10 px-2 py-0.5 text-[11px] font-medium text-slate-400">
   126→      Active
   127→    </span>
   128→  );

   189→// UniformEntryDetails

   191→export function UniformEntryDetails({
   192→  employee,
   193→  onBack,
   194→  autoOpenAdd = false,
   195→  renewEntryId = null,
   196→  onRenewEntry,
   197→}: UniformEntryDetailsProps) {
   198→  const version = useRealtimeVersion(['uniform_registry', 'employee', 'site']);
   199→
   200→  const [entries, setEntries] = useState<UniformEntry[] | null>(null);
   201→  const [formOpen, setFormOpen] = useState<boolean>(autoOpenAdd);
   202→  const [submitting, setSubmitting] = useState(false);
   203→  const [editingId, setEditingId] = useState<string | null>(null);
   204→  const [draft, setDraft] = useState<EditDraft | null>(null);
   205→  const [tab, setTab] = useState('');
   206→
   207→  const initAddForm = (): AddFormState => ({
   208→    document_type: employee.passport_number ? 'passport' : employee.id_number ? 'id' : 'passport',
   209→    document_number: employee.passport_number ?? employee.id_number ?? '',
   210→    created_at: todayYMD(),
   211→    site_name: '',
   212→    team_leader_name: '',
   213→    items: {},
   214→  });
   215→
   216→  const [addForm, setAddForm] = useState<AddFormState>(initAddForm);
   217→
   218→  const toggleForm = (next: boolean) => {
   219→    setFormOpen(next);
   220→    if (next) setAddForm(initAddForm());
   221→  };
   222→
   223→  // ---- data loading ---------------------------------------------------------
   224→  useEffect(() => {
   225→    let cancelled = false;
   226→    dataApi.listEmployeeUniforms(employee.id)
   227→      .then((rows) => { if (!cancelled) setEntries(rows); })
   228→      .catch(() => { if (!cancelled) setEntries([]); });
   229→    return () => { cancelled = true; };
   230→  }, [employee.id, version]);
   231→
   232→  // ---- month groups -----------------------------------------------------------
   233→  const groups = useMemo<MonthGroup[]>(() => {
   234→    if (!entries) return [];
   235→    const map = new Map<string, UniformEntry[]>();
   236→    for (const e of entries) {
   237→      const d = new Date(e.created_at);
   238→      if (Number.isNaN(d.getTime())) continue;
   239→      const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
   240→      const arr = map.get(key);
   241→      if (arr) arr.push(e);
   242→      else map.set(key, [e]);
   243→    }
   244→    return Array.from(map.entries())
   245→      .sort((a, b) => b[0].localeCompare(a[0]))
   246→      .map(([key, list]) => {
   247→        const dayList = list.map((e) => daysUntil(e.renewal_date));
   248→        return {
   249→          key,
   250→          label: `${MONTHS_SHORT[parseInt(key.slice(5), 10) - 1]} ${key.slice(0, 4)}`,
   251→          entries: list,
   252→          soon: dayList.some((dd) => dd >= 0 && dd <= 30),
   253→          expired: dayList.some((dd) => dd < 0),
   254→        };
   255→      });
   256→  }, [entries]);
   257→
   258→  const activeKey = tab && groups.some((g) => g.key === tab) ? tab : groups[0]?.key ?? '';
   259→  const activeGroup = groups.find((g) => g.key === activeKey) ?? null;
   260→  const addExpiryPreview = formatExpiryPreview(addForm.created_at);
   261→  const editExpiryPreview = draft ? formatExpiryPreview(draft.created_at) : null;
   262→
   263→  // ---- mutations ---------------------------------------------------------------
   264→  const patchAdd = (patch: Partial<AddFormState>) => setAddForm((prev) => ({ ...prev, ...patch }));
   265→  const patchAddItem = (key: string, value: boolean) =>
   266→    setAddForm((prev) => ({ ...prev, items: { ...prev.items, [key]: value } }));
   267→  const patchDraft = (patch: Partial<EditDraft>) =>
   268→    setDraft((prev) => (prev ? { ...prev, ...patch } : prev));
   269→  const patchDraftItem = (key: string, value: boolean) =>
   270→    setDraft((prev) => (prev ? { ...prev, items: { ...prev.items, [key]: value } } : prev));
   271→
   272→  const submitAdd = async () => {
   273→    if (!addForm.document_number.trim()) {
   274→      toast.error('Document number is required');
   275→      return;
   276→    }
   277→    if (!UNIFORM_ITEMS.some((i) => addForm.items[i.key] === true)) {
   278→      toast.error('Select at least one item given');
   279→      return;
   280→    }
   281→    if (!addForm.created_at) {
   282→      toast.error('Created date is required');
   283→      return;
   284→    }
   285→    setSubmitting(true);
   286→    try {
   287→      await dataApi.createUniformEntry({
   288→        employee_id: employee.id,
   289→        document_type: addForm.document_type,
   290→        document_number: addForm.document_number.trim(),
   291→        items: buildItemsRecord(addForm.items),
   292→        site_name: addForm.site_name.trim() || null,
   293→        team_leader_name: addForm.team_leader_name.trim() || null,
   294→        created_at: new Date(addForm.created_at).toISOString(),
   295→        token_number: null, // assigned automatically
   296→        is_renewal: false,
   297→        previous_token_id: null,
   298→      });
   299→      toast.success('Uniform entry created');
   300→      toggleForm(false);
   301→    } catch (err) {
   302→      toast.error(errMessage(err));
   303→    } finally {
   304→      setSubmitting(false);
   305→    }
   306→  };
   307→
   308→  const startEdit = (e: UniformEntry) => {
   309→    setEditingId(e.id);
   310→    setDraft({
   311→      token_number: String(e.token_number),
   312→      document_type: e.document_type,
   313→      document_number: e.document_number,
   314→      site_name: e.site_name ?? '',
   315→      team_leader_name: e.team_leader_name ?? '',
   316→      created_at: isoToYMD(e.created_at),
   317→      items: { ...e.items },
   318→    });
   319→  };
   320→
   321→  const cancelEdit = () => {
   322→    setEditingId(null);
   323→    setDraft(null);
   324→  };
   325→
   326→  const saveEdit = async (entry: UniformEntry) => {
   327→    if (!draft) return;
   328→    const token = parseInt(draft.token_number, 10);
   329→    if (!Number.isInteger(token) || token <= 0) {
   330→      toast.error('Token number must be a positive whole number');
   331→      return;
   332→    }
   333→    if (!draft.document_number.trim()) {
   334→      toast.error('Document number is required');
   335→      return;
   336→    }
   337→    if (!UNIFORM_ITEMS.some((i) => draft.items[i.key] === true)) {
   338→      toast.error('Select at least one item given');
   339→      return;
   340→    }
   341→    if (!draft.created_at) {
   342→      toast.error('Created date is required');
   343→      return;
   344→    }
   345→    setSubmitting(true);
   346→    try {
   347→      await dataApi.updateUniformEntry(entry.id, {
   348→        token_number: token,
   349→        document_type: draft.document_type,
   350→        document_number: draft.document_number.trim(),
   351→        items: buildItemsRecord(draft.items),
   352→        site_name: draft.site_name.trim() || null,
   353→        team_leader_name: draft.team_leader_name.trim() || null,
   354→        created_at: new Date(draft.created_at).toISOString(),
   355→      });
   356→      toast.success('Entry updated');
   357→      cancelEdit();
   358→    } catch (err) {
   359→      toast.error(errMessage(err));
   360→    } finally {
   361→      setSubmitting(false);
   362→    }
   363→  };
   364→
   365→  const handleDelete = async (entry: UniformEntry) => {
   366→    try {
   367→      await dataApi.deleteUniformEntry(entry.id);
   368→      toast.success('Uniform entry deleted');
   369→      if (editingId === entry.id) cancelEdit();
   370→    } catch (err) {
   371→      toast.error(errMessage(err));
   372→    }
   373→  };
   374→
   375→  const triggerRenew = (entry: UniformEntry) => {
   376→    onRenewEntry?.(entry);
   377→    onBack();
   378→  };
   379→
   380→  // ---- render ------------------------------------------------------------------
   381→  return (
   382→    <div className="mx-auto max-w-7xl">
   383→      {/* Header */}
   384→      <div className="mb-6 flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   385→        <div className="flex items-start gap-3">
   386→          <Button
   387→            variant="outline"
   388→            size="icon"
   389→            onClick={onBack}
   390→            aria-label="Back to uniform registry"
   391→            title="Back to uniform registry"
   392→            className="mt-0.5 rounded-lg"
   393→          >
   394→            <ArrowLeft size={16} />
   395→          </Button>
   396→          <motion.div
   397→            initial={{ opacity: 0, y: 10 }}
   398→            animate={{ opacity: 1, y: 0 }}
   399→            transition={{ duration: 0.35, ease: [0.22, 1, 0.36, 1] }}
   400→            className="flex items-start gap-3"
   401→          >
   402→            <span className="flex size-11 shrink-0 items-center justify-center rounded-xl bg-brand/15 text-sm font-bold text-brand">
   403→              {initials(employee.full_name)}
   404→            </span>
   405→            <div>
   406→              <h1 className="text-xl font-bold uppercase leading-tight tracking-tight md:text-2xl">
   407→                {employee.full_name}
   408→              </h1>
   409→              <p className="mt-1 text-[11px] font-medium uppercase tracking-wide text-muted-foreground">
   410→                Employee ID: {employee.employee_code} · {entries?.length ?? 0} entries total
   411→              </p>
   412→            </div>
   413→          </motion.div>
   414→        </div>
   415→        <Button
   416→          className="brand-gradient rounded-lg text-brand-foreground shadow-lg shadow-brand/20 hover:opacity-90"
   417→          onClick={() => toggleForm(!formOpen)}
   418→        >
   419→          <Plus size={16} />
   420→          Add Entry
   421→        </Button>
   422→      </div>
   423→
   424→      {/* Inline Add form (expandable card) */}
   425→      <AnimatePresence initial={false}>
   426→        {formOpen && (
