   123→const nextEmployeeCode = (): string => {
   124→  const year = new Date().getFullYear();
   125→  const prefix = `PUL-${year}-`;
   126→  const max = db.employees.reduce((m, e) => {
   127→    if (e.employee_code.startsWith(prefix)) {
   128→      const n = parseInt(e.employee_code.slice(prefix.length), 10);
   129→      return Number.isNaN(n) ? m : Math.max(m, n);
   130→    }
   131→    return m;
   132→  }, 0);
   133→  return `${prefix}${String(max + 1).padStart(3, '0')}`;
   134→};

   392→  async deleteEmployeeCascade(id: string): Promise<void> {
   393→    const e = db.employees.find((x) => x.id === id);
   394→    if (!e) throw new Error('Employee not found');
   395→    if (e.status === 'pending_deletion') throw new Error('This employee already has a pending deletion request');

   397→    db.attendance.filter((a) => a.employee_id === id).forEach((a) => { a.is_hidden = true; });
   398→    db.warnings.filter((w) => w.employee_id === id).forEach((w) => { w.is_hidden = true; });
   399→    db.fines.filter((f) => f.employee_id === id).forEach((f) => { f.is_hidden = true; });
   400→    db.leaveRequests.filter((r) => r.employee_id === id).forEach((r) => { r.is_hidden = true; });
   401→    db.cancellationRequests.filter((r) => r.employee_id === id).forEach((r) => { r.is_hidden = true; });
   402→    db.uniforms.filter((u) => u.employee_id === id).forEach((u) => { u.is_deleted = true; u.is_hidden = true; });
   403→    const nameLc = e.full_name.toLowerCase();
   404→    const codeLc = e.employee_code.toLowerCase();
   405→    db.notifications.filter((n) =>
   406→      n.title.toLowerCase().includes(nameLc)
   407→      || n.message.toLowerCase().includes(nameLc)
   408→      || n.message.toLowerCase().includes(codeLc),
   409→    ).forEach((n) => { n.is_hidden = true; });
   410→
   411→    e.status = 'deleted';
   412→    e.current_site_id = null;
   413→    e.is_team_leader = false;
   414→    e.team_leader_site_id = null;
   415→    saveAndEmit(['employee', 'attendance', 'warning', 'fine', 'leave_request', 'cancellation_request', 'uniform_registry', 'notification']);

   439→  async getMonthAttendance(month: string): Promise<Attendance[]> {
   440→    const today = toYMD(new Date());
   441→    const [y, m] = month.split('-').map(Number);
   442→    const first = `${month}-01`;
   443→    const monthEnd = toYMD(new Date(y, m, 0));
   444→    const lastDay = today < first ? null : today < monthEnd ? today : monthEnd;
   445→
   446→    // auto-create missing records for non-deleted employees
   447→    if (lastDay) {
   448→      for (const e of db.employees) {
   449→        if (e.status === 'deleted') continue;
   450→        for (let d = fromYMD(first); toYMD(d) <= lastDay; d.setDate(d.getDate() + 1)) {
   451→          const dateStr = toYMD(d);
   452→          const exists = db.attendance.some((a) => a.employee_id === e.id && a.date === dateStr);
   453→          if (!exists) {
   454→            db.attendance.push({
   455→              id: uid(), employee_id: e.id, date: dateStr,
   456→              status: e.current_site_id ? 'present' : 'no_site',
   457→              overtime_hours: null, is_hidden: false, created_at: nowISO(),
   458→            });
   459→          }
   460→        }
   461→      }
   462→      saveAndEmit(['attendance']);
   463→    }
   464→
   465→    return db.attendance.filter((a) => {
   466→      if (!a.date.startsWith(month) || a.is_hidden) return false;
   467→      const e = db.employees.find((x) => x.id === a.employee_id);
   468→      return e ? e.status !== 'deleted' : false;
   469→    });
   470→  },

   472→  async markAttendance(employeeId: string, date: string, status: AttendanceStatus, overtimeHours: number | null): Promise<void> {
   473→    const e = db.employees.find((x) => x.id === employeeId);
   474→    if (!e) throw new Error('Employee not found');
   475→
   476→    const prev = db.attendance.find((a) => a.employee_id === employeeId && a.date === date);
   477→    const otDelta = prev ? (overtimeHours ?? 0) - (prev.overtime_hours ?? 0) : (status === 'overtime' ? overtimeHours ?? 0 : 0);
   478→
   479→    if (prev) {
   480→      prev.status = status;
   481→      prev.overtime_hours = status === 'overtime' ? overtimeHours : null;
   482→    } else {
   483→      db.attendance.push({
   484→        id: uid(), employee_id: employeeId, date, status,
   485→        overtime_hours: status === 'overtime' ? overtimeHours : null,
   486→        is_hidden: false, created_at: nowISO(),
   487→      });
   488→    }
   489→
   490→    // rating: ±0.2 stars per overtime hour delta
   491→    if (otDelta !== 0) {
   492→      e.rating = clampRating(e.rating + 0.2 * otDelta);
   493→    }
   494→
   495→    // auto-warning on 3+ consecutive absences (skipping Fridays)
   496→    if (status === 'absent') {
   497→      const absentDates: string[] = [date];
   498→      let walk = addDaysStr(date, -1);
   499→      const floorDate = addDaysStr(date, -7);
   500→      while (walk >= floorDate) {
   501→        if (isFridayStr(walk)) { walk = addDaysStr(walk, -1); continue; }
   502→        const rec = db.attendance.find((a) => a.employee_id === employeeId && a.date === walk && !a.is_hidden);
   503→        if (rec && rec.status === 'absent') {
   504→          absentDates.unshift(walk);
   505→          walk = addDaysStr(walk, -1);
   506→        } else break;
   507→      }
   508→      if (absentDates.length >= 3) {
   509→        db.warnings.push({
   510→          id: uid(), employee_id: employeeId,
   511→          reason: `Auto-generated: ${absentDates.length} consecutive absences detected (${absentDates.join(', ')})`,
   512→          is_auto_generated: true, absent_dates: absentDates,
   513→          created_by: currentProfile()?.id ?? null, is_hidden: false, created_at: nowISO(),
   514→        });
   515→        e.rating = clampRating(e.rating - 0.5);
   516→        notifySuperAdmins(
   517→          'Auto-Generated Warning',
   518→          `${e.full_name} (${e.employee_code}) was absent 3+ consecutive days`,
   519→          'warning',
   520→        );
   521→        saveAndEmit(['warning', 'notification', 'employee', 'attendance']);
   522→        return;
   523→      }
   524→    }
   525→    saveAndEmit(['attendance', 'employee']);
   526→  },
   527→
   528→  async getMonthlyAttendanceSummary(months = 6): Promise<MonthlySummaryRow[]> {
   529→    const rows: MonthlySummaryRow[] = [];
   530→    const now = new Date();
   531→    for (let back = months - 1; back >= 0; back--) {
   532→      const d = new Date(now.getFullYear(), now.getMonth() - back, 1);
   533→      const m = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
   534→      const counts: Record<string, number> = {};
   535→      for (const a of db.attendance) {
   536→        if (!a.date.startsWith(m) || a.is_hidden || a.status === 'not_marked') continue;
   537→        const e = db.employees.find((x) => x.id === a.employee_id);
   538→        if (!e || e.status === 'deleted') continue;
   539→        counts[a.status] = (counts[a.status] ?? 0) + 1;
   540→      }
   541→      for (const [status, count] of Object.entries(counts)) {
   542→        rows.push({ month: m, status: status as AttendanceStatus, count });
   543→      }
   544→    }
   545→    return rows;
   546→  },

   716→  async createUniformEntry(input: UniformEntryInput): Promise<UniformEntry> {
   717→    const e = db.employees.find((x) => x.id === input.employee_id);
   718→    if (!e) throw new Error('Employee not found');
   719→
   720→    const uniformId = db.counters.uniform_id + 1;
   721→    let token = db.counters.token_number + 1;
   722→    if (input.token_number && input.token_number > 0) token = input.token_number;
   723→    db.counters.uniform_id = uniformId;
   724→    db.counters.token_number = token;
   725→
   726→    const created = new Date(input.created_at);
   727→    if (Number.isNaN(created.getTime())) throw new Error('Invalid created date');
   728→    const entry: UniformEntry = {
   729→      id: uid(), uniform_id: uniformId, token_number: token,
   730→      employee_id: e.id, employee_name: e.full_name,
   731→      document_type: input.document_type, document_number: input.document_number,
   732→      items: input.items, site_name: input.site_name, team_leader_name: input.team_leader_name,
   733→      is_renewal: input.is_renewal, previous_token_id: input.previous_token_id,
   734→      is_deleted: false, is_hidden: false,
   735→      created_at: created.toISOString(),
   736→      renewal_date: new Date(created.getTime() + 182.5 * 86400000).toISOString(),
   737→    };
   738→    db.uniforms.push(entry);
   739→    saveAndEmit(['uniform_registry']);
   740→    return entry;
   741→  },
