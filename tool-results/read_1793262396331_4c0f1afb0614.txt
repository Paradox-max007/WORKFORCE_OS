    39→function formatDateDisplay(date: Date): string {
    40→  const day = String(date.getDate()).padStart(2, '0');
    41→  const month = String(date.getMonth() + 1).padStart(2, '0');
    42→  const year = date.getFullYear();
    43→  return `${day}/${month}/${year}`;
    44→}
    45→
    46→function parseDateInput(value: string): Date {
    47→  const parts = value.split('/');
    48→  if (parts.length === 3) {
    49→    const [day, month, year] = parts.map(Number);
    50→    if (day && month && year) return new Date(year, month - 1, day);
    51→  }
    52→  return new Date();
    53→}

    90→/* ───────── Page Chunk Helper ───────── */
    91→function chunkRows<T>(items: T[], perPage: number): T[][] {
    92→  const chunks: T[][] = [];
    93→  for (let i = 0; i < items.length; i += perPage) {
    94→    chunks.push(items.slice(i, i + perPage));
    95→  }
    96→  return chunks;
    97→}

   308→/* ───────── Main Component ───────── */
   309→export function AttendanceSheet({ site, employees, onClose }: AttendanceSheetProps) {
   310→  const pageRefs = useRef<(HTMLDivElement | null)[]>([]);
   311→  const [date, setDate] = useState<Date>(new Date());
   312→  const [dateInput, setDateInput] = useState(formatDateDisplay(new Date()));
   313→  const [isGenerating, setIsGenerating] = useState(false);
   314→
   315→  // Editable info fields
   316→  const [clientName, setClientName] = useState(site.clientName || '');
   317→  const [projectName, setProjectName] = useState(site.projectName || site.name);
   318→  const [strengthInput, setStrengthInput] = useState(String(employees.length));
   319→
   320→  // Editable employee data
   321→  const [employeeData, setEmployeeData] = useState(() =>
   322→    employees.map((emp) => ({
   323→      id: emp.id,
   324→      fullName: emp.fullName,
   325→      code: '',
   326→      position: emp.position || '',
   327→      isTeamLeader: emp.isTeamLeader,
   328→      isSupervisor: emp.position?.toLowerCase().includes('supervisor') ?? false,
   329→    }))
   330→  );
   331→
   332→  // Sort
   333→  const sortedEmployees = useMemo(() => {
   334→    return [...employeeData].sort((a, b) => {
   335→      if (a.isTeamLeader && !b.isTeamLeader) return -1;
   336→      if (!a.isTeamLeader && b.isTeamLeader) return 1;
   337→      if (a.isSupervisor && !b.isSupervisor) return -1;
   338→      if (!a.isSupervisor && b.isSupervisor) return 1;
   339→      return a.fullName.localeCompare(b.fullName);
   340→    });
   341→  }, [employeeData]);
   342→
   343→  const updateEmployee = useCallback(
   344→    (id: string, field: 'fullName' | 'code' | 'position' | 'serialNo', value: string) => {
   345→      setEmployeeData((prev) =>
   346→        prev.map((emp) =>
   347→          emp.id === id
   348→            ? { ...emp, [field]: value, isSupervisor: field === 'position' ? value.toLowerCase().includes('supervisor') : emp.isSupervisor }
   349→            : emp
   350→        )
   351→      );
   352→    },
   353→    []
   354→  );
   355→
   356→  const handleDateChange = useCallback((value: string) => {
   357→    setDateInput(value);
   358→    const parsed = parseDateInput(value);
   359→    if (!isNaN(parsed.getTime())) setDate(parsed);
   360→  }, []);
   361→
   362→  const getDisplayTrade = useCallback((emp: { isTeamLeader: boolean; isSupervisor?: boolean; position?: string }) => {
   363→    const pos = emp.position || '';
   364→    if (emp.isTeamLeader) return pos ? `${pos} / TL` : 'TL';
   365→    if (emp.isSupervisor) return pos ? `${pos} / SUPERVISOR` : 'SUPERVISOR';
   366→    return pos;
   367→  }, []);
   368→
   369→  // Build rows: only employees (extras are separate table now)
   370→  const employeeRows = useMemo(() => {
   371→    return sortedEmployees.map((emp) => ({
   372→      type: 'employee' as const,
   373→      ...emp,
   374→    }));
   375→  }, [sortedEmployees]);
   376→
   377→  const extraRowItems = useMemo(() => {
   378→    return Array.from({ length: EXTRA_ROWS }, () => ({ type: 'extra' as const }));
   379→  }, []);
   380→
   381→  // Chunk employee rows into pages
   382→  const FIRST_PAGE_ROWS = ROWS_PER_PAGE - 6;
   383→  const pages = useMemo(() => {
   384→    if (employeeRows.length <= FIRST_PAGE_ROWS) return [employeeRows];
   385→    const result: typeof employeeRows[] = [employeeRows.slice(0, FIRST_PAGE_ROWS)];
   386→    const remaining = employeeRows.slice(FIRST_PAGE_ROWS);
   387→    result.push(...chunkRows(remaining, ROWS_PER_PAGE));
   388→    return result;
   389→  }, [employeeRows]);
   390→
   391→  // Generate HTML for all pages (shared by PDF and Print)
   392→  const generateAllPagesHtml = useCallback(() => {
   393→    let allHtml = '';
   394→    pages.forEach((pageEmployeeRows, pageIdx) => {
   395→      const isFirstPage = pageIdx === 0;
   396→      const isLastPage = pageIdx === pages.length - 1;
   397→      const serialOffset = pageIdx === 0 ? 0 : pages.slice(0, pageIdx).flat().length;
   398→
   399→      allHtml += buildPageHtml({
   400→        employeeRows: pageEmployeeRows,
   401→        extraRows: isLastPage ? extraRowItems : [],
   402→        pageIdx,
   403→        totalPages: pages.length,
   404→        clientName,
   405→        projectName,
   406→        dateInput,
   407→        strengthInput,
   408→        sortedEmployees,
   409→        getDisplayTrade,
   410→        contentWidth: '794px',
   411→        contentPadding: '38px',
   412→        isFirstPage,
   413→        isLastPage,
   414→        serialOffset,
   415→      });
   416→    });
   417→    return allHtml;
   418→  }, [pages, extraRowItems, clientName, projectName, dateInput, strengthInput, sortedEmployees, getDisplayTrade]);
   419→
   420→  /* ── Download PDF directly (jsPDF + html2canvas) ── */
   421→  const handleDownloadPDF = useCallback(async () => {
   422→    if (isGenerating) return;
   423→    setIsGenerating(true);
   424→
   425→    try {
   426→      const iframe = document.createElement('iframe');
   427→      iframe.style.position = 'fixed';
   428→      iframe.style.left = '-9999px';
   429→      iframe.style.top = '-9999px';
   430→      iframe.style.width = '794px';
   431→      iframe.style.height = '1123px';
   432→      document.body.appendChild(iframe);
   433→
   434→      const iframeDoc = iframe.contentDocument || iframe.contentWindow?.document;
   435→      if (!iframeDoc) {
   436→        document.body.removeChild(iframe);
   437→        setIsGenerating(false);
   438→        return;
   439→      }
   440→
   441→      iframeDoc.open();
   442→      iframeDoc.write(`<!DOCTYPE html><html><head><style>${getPrintCSS()}</style></head><body>`);
   443→      iframeDoc.write(generateAllPagesHtml());
   444→      iframeDoc.write(`</body></html>`);
   445→      iframeDoc.close();
   446→
   447→      await new Promise((resolve) => setTimeout(resolve, 600));
   448→
   449→      const pdf = new jsPDF({ orientation: 'portrait', unit: 'mm', format: 'a4' });
   450→      const pageWidth = pdf.internal.pageSize.getWidth();
   451→      const pageHeight = pdf.internal.pageSize.getHeight();
   452→
   453→      const pageDivs = iframeDoc.querySelectorAll('.page');
   454→
   455→      for (let i = 0; i < pageDivs.length; i++) {
   456→        if (i > 0) pdf.addPage();
   457→
   458→        const canvas = await html2canvas(pageDivs[i] as HTMLElement, {
   459→          scale: 2,
   460→          useCORS: true,
   461→          allowTaint: true,
   462→          backgroundColor: '#ffffff',
   463→          logging: false,
   464→        });
   465→
   466→        const imgData = canvas.toDataURL('image/png');
   467→        pdf.addImage(imgData, 'PNG', 0, 0, pageWidth, pageHeight);
   468→      }
   469→
   470→      const fileName = `attendance-${site.name.replace(/\s+/g, '-')}-${date.toISOString().split('T')[0]}.pdf`;
   471→      pdf.save(fileName);
   472→
   473→      document.body.removeChild(iframe);
   474→    } catch (error) {
   475→      console.error('Error generating PDF:', error);
   476→    } finally {
   477→      setIsGenerating(false);
   478→    }
   479→  }, [isGenerating, site.name, date, generateAllPagesHtml]);
   480→
   481→  /* ── Print with @page margin:0 to suppress browser headers/footers ── */
   482→  const handlePrint = useCallback(() => {
   483→    const iframe = document.createElement('iframe');
   484→    iframe.style.position = 'fixed';
   485→    iframe.style.left = '-9999px';
   486→    iframe.style.top = '-9999px';
   487→    iframe.style.width = '794px';
   488→    iframe.style.height = '1123px';
   489→    document.body.appendChild(iframe);
   490→
   491→    const iframeDoc = iframe.contentDocument || iframe.contentWindow?.document;
   492→    if (!iframeDoc) {
   493→      document.body.removeChild(iframe);
   494→      return;
   495→    }
   496→
   497→    iframeDoc.open();
   498→    iframeDoc.write(`<!DOCTYPE html><html><head><style>${getPrintCSS()}</style></head><body>`);
   499→    iframeDoc.write(generateAllPagesHtml());
   500→    iframeDoc.write(`</body></html>`);
   501→    iframeDoc.close();
   502→
   503→    setTimeout(() => {
   504→      iframe.contentWindow?.focus();
   505→      iframe.contentWindow?.print();
   506→      setTimeout(() => {
   507→        document.body.removeChild(iframe);
   508→      }, 2000);
   509→    }, 600);
   510→  }, [generateAllPagesHtml]);
   511→
   512→  return (
   513→    <>
