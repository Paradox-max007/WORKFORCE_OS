   258→-- Am I a super admin?
   259→create or replace function public.is_super_admin()
   260→returns boolean language sql stable
   261→security definer set search_path = public
   262→as $$ select exists(select 1 from public.profiles where id = auth.uid() and role = 'super_admin'); $$;

   264→-- Am I staff (admin or super_admin)?
   265→create or replace function public.is_staff()
   266→returns boolean language sql stable
   267→security definer set search_path = public
   268→as $$ select exists(select 1 from public.profiles where id = auth.uid() and role in ('super_admin','admin')); $$;
   269→
   270→-- Does the current admin hold a given menu permission (super admins always true)?
   271→create or replace function public.has_menu_permission(p_menu_key text)
   272→returns boolean language sql stable
   273→security definer set search_path = public
   274→as $$
   275→  select
   276→    exists(select 1 from public.profiles where id = auth.uid() and role = 'super_admin')
   277→    or exists(
   278→      select 1 from public.menu_permission mp
   279→      join public.profiles p on p.id = mp.profile_id
   280→      where mp.profile_id = auth.uid() and mp.menu_key = p_menu_key and mp.allowed and p.role = 'admin'
   281→    );
   282→$$;

   317→-- New auth user → profile row (first user becomes super_admin, others 'pending'
   318→-- until a super admin promotes them from Admin Management)
   319→create or replace function public.handle_new_user()
   320→returns trigger language plpgsql
   321→security definer set search_path = public
   322→as $$
   323→begin
   324→  insert into public.profiles (id, email, name, role)
   325→  values (
   326→    new.id,
   327→    lower(new.email),
   328→    coalesce(new.raw_user_meta_data->>'name', split_part(new.email, '@', 1)),
   329→    case when (select count(*) from public.profiles) = 0 then 'super_admin' else 'pending' end
   330→  )
   331→  on conflict (id) do nothing;
   332→  return new;
   333→end; $$;
   334→
   335→drop trigger if exists on_auth_user_created on auth.users;
   336→create trigger on_auth_user_created
   337→  after insert on auth.users
   338→  for each row execute function public.handle_new_user();

   429→-- Update own display name + theme + accent
   430→create or replace function public.set_preferences(p_name text default null, p_theme text default null, p_accent text default null)
   431→returns public.profiles language plpgsql
   432→security definer set search_path = public
   433→as $$
   434→declare result public.profiles;
   435→begin
   436→  update public.profiles set
   437→    name   = coalesce(p_name, name),
   438→    theme  = coalesce(nullif(p_theme, ''), theme),
   439→    accent = coalesce(nullif(p_accent, ''), accent)
   440→  where id = auth.uid()
   441→  returning * into result;
   442→  return result;
   443→end; $$;

   445→-- Super admin edits another profile's name / role (email + password go via the
   446→-- service-role API route). Last-super-admin protection included.
   447→create or replace function public.update_profile_admin(p_id uuid, p_name text, p_role text)
   448→returns public.profiles language plpgsql
   449→security definer set search_path = public
   450→as $$
   451→declare result public.profiles;
   452→declare target_count int;
   453→begin
   454→  if not public.is_super_admin() then raise exception 'Only super admins can manage profiles'; end if;
   455→  if p_role not in ('admin','super_admin') then raise exception 'Invalid role'; end if;
   456→
   457→  -- last super admin protection
   458→  if (select role from public.profiles where id = p_id) = 'super_admin' and p_role <> 'super_admin' then
   459→    select count(*) into target_count from public.profiles where role = 'super_admin';
   460→    if target_count <= 1 then raise exception 'Cannot demote the last super admin'; end if;
   461→  end if;
   462→
   463→  update public.profiles set name = p_name, role = p_role, updated_at = now()
   464→  where id = p_id returning * into result;
   465→  return result;
   466→end; $$;

   468→-- Super admin deletes a profile (last-super-admin protected). The auth.users
   469→-- row is removed separately by the service-role API route.
   470→create or replace function public.delete_profile_admin(p_id uuid)
   471→returns void language plpgsql
   472→security definer set search_path = public
   473→as $$
   474→declare target_count int;
   475→begin
   476→  if not public.is_super_admin() then raise exception 'Only super admins can delete profiles'; end if;
   477→  if (select role from public.profiles where id = p_id) = 'super_admin' then
   478→    select count(*) into target_count from public.profiles where role = 'super_admin';
   479→    if target_count <= 1 then raise exception 'Cannot delete the last super admin'; end if;
   480→  end if;
   481→  delete from public.profiles where id = p_id;
   482→end; $$;

   484→-- Grant / revoke a menu permission for an admin
   485→create or replace function public.set_menu_permission(p_profile_id uuid, p_menu_key text, p_allowed boolean)
   486→returns void language plpgsql

   488→as $$
   489→begin

   682→create or replace function public.create_employee(p_employee jsonb)
   683→returns public.employee language plpgsql
   684→security definer set search_path = public
   685→as $$
   686→declare result public.employee;
   687→declare v_site uuid := nullif(p_employee->>'current_site_id','');
   688→declare v_camp uuid := nullif(p_employee->>'camp_id','');
   689→declare v_block text := nullif(btrim(coalesce(p_employee->>'camp_block','')), '');
   690→declare v_room  text := nullif(btrim(coalesce(p_employee->>'camp_room','')), '');
   691→declare v_leader_site uuid := nullif(p_employee->>'team_leader_site_id','');
   692→declare v_is_leader boolean := coalesce((p_employee->>'is_team_leader')::boolean, false);
   693→-- manual employee ID: use it as-is; blank → auto-generate (PUL-YYYY-NNN)
   694→declare v_code text := nullif(btrim(coalesce(p_employee->>'employee_code','')), '');
   695→declare conflicting public.employee;
   696→begin
   697→  if not public.has_menu_permission('employees') then raise exception 'Access denied'; end if;
   698→
   699→  -- manual IDs must be unique (case-insensitive)
   700→  if v_code is not null then
   701→    select * into conflicting from public.employee
   702→      where lower(employee_code) = lower(v_code) limit 1;
   703→    if conflicting.id is not null then
   704→      raise exception 'Employee ID "%" is already taken by % — use another ID or leave it blank to auto-generate', v_code, conflicting.full_name;
   705→    end if;
   706→  end if;
   707→
   708→  -- site must exist and be active
   709→  if v_site is not null and not exists (select 1 from public.site where id = v_site and is_active) then
   710→    raise exception 'The selected site does not exist or is inactive';
   711→  end if;
   712→  -- camp must exist and be active
   713→  if v_camp is not null and not exists (select 1 from public.camp where id = v_camp and is_active) then
   714→    raise exception 'The selected camp does not exist or is inactive';
   715→  end if;
   716→  -- block/room only make sense while the employee stays in a camp
   717→  if v_camp is null then
   718→    v_block := null;
   719→    v_room := null;
   720→  end if;
   721→
   722→  -- one team leader per site
   723→  if v_is_leader and v_leader_site is not null then
   724→    select * into conflicting from public.employee
   725→      where is_team_leader and team_leader_site_id = v_leader_site and status <> 'deleted' limit 1;
   726→    if conflicting.id is not null then
   727→      raise exception 'Another employee (%) is already the team leader of this site', conflicting.full_name;
   728→    end if;
   729→  end if;
   730→
   731→  insert into public.employee (
   732→    full_name, employee_code, nationality, date_of_birth, phone, email, address,
   733→    emergency_contact, position, join_date, company_name, passport_number, passport_status,
   734→    id_number, id_status, current_site_id, rating, status, photo, is_team_leader, team_leader_site_id,
   735→    camp_id, camp_block, camp_room
   736→  ) values (
   737→    p_employee->>'full_name', coalesce(v_code, public.next_employee_code()),
   738→    nullif(p_employee->>'nationality',''), (p_employee->>'date_of_birth')::date,
   739→    nullif(p_employee->>'phone',''), nullif(p_employee->>'email',''), nullif(p_employee->>'address',''),
   740→    nullif(p_employee->>'emergency_contact',''), nullif(p_employee->>'position',''),
   741→    (p_employee->>'join_date')::date, nullif(p_employee->>'company_name',''),
   742→    nullif(p_employee->>'passport_number',''), nullif(p_employee->>'passport_status',''),
   743→    nullif(p_employee->>'id_number',''), nullif(p_employee->>'id_status',''),
   744→    v_site, 5.0, 'active', nullif(p_employee->>'photo',''), v_is_leader, v_leader_site, v_camp,
   745→    v_block, v_room
   746→  ) returning * into result;
   747→  return result;
   748→end; $$;

   764→  if not public.has_menu_permission('employees') then raise exception 'Access denied'; end if;
   765→
   766→  if v_site is not null and not exists (select 1 from public.site where id = v_site and is_active) then
   767→    raise exception 'The selected site does not exist or is inactive';
   768→  end if;
   769→  if v_camp is not null and not exists (select 1 from public.camp where id = v_camp and is_active) then
   770→    raise exception 'The selected camp does not exist or is inactive';

   774→    v_block := null;

   782→    if conflicting.id is not null then
   783→      raise exception 'Another employee (%) is already the team leader of this site', conflicting.full_name;

   785→  end if;

   788→  select current_site_id into v_prev_site from public.employee where id = p_id;

   791→    full_name         = p_employee->>'full_name',

   793→    date_of_birth     = (p_employee->>'date_of_birth')::date,
   794→    phone             = nullif(p_employee->>'phone',''),
   795→    email             = nullif(p_employee->>'email',''),

   797→    emergency_contact = nullif(p_employee->>'emergency_contact',''),
   798→    position          = nullif(p_employee->>'position',''),

   800→    company_name      = nullif(p_employee->>'company_name',''),
   801→    passport_number   = nullif(p_employee->>'passport_number',''),
   802→    passport_status   = nullif(p_employee->>'passport_status',''),
   803→    id_number         = nullif(p_employee->>'id_number',''),
   804→    id_status         = nullif(p_employee->>'id_status',''),
   805→    current_site_id   = v_site,
   806→    camp_id           = v_camp,
   807→    camp_block        = v_block,
   808→    camp_room         = v_room,
   809→    photo             = nullif(p_employee->>'photo',''),
   810→    is_team_leader    = v_is_leader,
   811→    team_leader_site_id = case when v_is_leader then v_leader_site else null end,
   812→    updated_at        = now()
   813→  where id = p_id returning * into result;
   814→
   815→  -- log the site move (site → site, site → idle, idle → site) so the attendance

   817→  -- across the merged moved-days cells
   818→  if v_prev_site is distinct from v_site then

   821→            coalesce(p_effective_date, (now() at time zone 'Asia/Dubai')::date));

   823→  return result;
   824→end; $$;

   858→-- Admin-initiated removal request → CancellationRequest + pending_deletion + notify

   886→-- Review a cancellation request (super admin only)

   921→-- 9. ATTENDANCE RPCs
   922→-- ============================================================================
   923→
   924→-- Fetch one month's REAL attendance records (un-entered days stay blank in the
   925→-- marking sheet). Normal admins only receive rows for employees working at THEIR assigned sites.
   926→-- ---------------------------------------------------------------------------
   927→-- Date-effective employee site assignment (mirrors src/lib/site-timeline.ts).
   928→--
   929→-- Answers "which site did this employee belong to on THIS date" from the
   930→-- site_transfer log — NOT from employee.current_site_id (that only reflects
   931→-- the present and breaks historical attendance after a move).
   932→--
   933→-- The "25th at 8 PM" rule: a day already marked at the OLD site before the
   934→-- move was recorded keeps its old-site assignment — the move is enforced from
   935→-- the first day that was NOT already marked (marked on the 25th + moved on
   936→-- the 25th → effective from the 26th). Existing attendance records are never
   937→-- rewritten. Multiple transfers fold chronologically (effective_date, then
   938→-- created_at), so A → B → C in one month resolves correctly per date.
   939→-- ---------------------------------------------------------------------------
   940→create or replace function public.emp_site_on_date(p_emp uuid, p_date date)
   941→returns uuid
   942→language plpgsql
   943→stable
   944→security definer set search_path = public
   945→as $$
   946→declare
   947→  t record;
   948→  v_site uuid;
   949→  v_first uuid;
   950→  v_start date;
   951→begin
   952→  -- no logged move → the employee simply sits at their current site
   953→  if not exists (select 1 from public.site_transfer where employee_id = p_emp) then
   954→    return (select current_site_id from public.employee where id = p_emp);
   955→  end if;
   956→
   957→  -- site before any logged move = where the earliest move started from
   958→  select st.from_site_id into v_first
   959→    from public.site_transfer st
   960→    where st.employee_id = p_emp
   961→    order by st.effective_date, st.created_at
   962→    limit 1;
   963→  v_site := v_first;
   964→
   965→  -- fold every move whose display date is on/before the requested date
   966→  for t in
   967→    select st.from_site_id, st.to_site_id, st.effective_date, st.created_at
   968→    from public.site_transfer st
   969→    where st.employee_id = p_emp
   970→    order by st.effective_date, st.created_at
   971→  loop
   972→    -- display date: skip days already marked at the old site BEFORE this
   973→    -- move was recorded (bounded by the requested date)
   974→    v_start := t.effective_date;
   975→    while v_start <= p_date and exists (
   976→      select 1 from public.attendance a
   977→      where a.employee_id = p_emp
   978→        and a.date = v_start
   979→        and a.created_at <= t.created_at
   980→    ) loop
   981→      v_start := v_start + 1;
   982→    end loop;
   983→    if v_start <= p_date then
   984→      v_site := t.to_site_id;
   985→    end if;
   986→  end loop;
   987→
   988→  return v_site;
   989→end; $$;

   993→create or replace function public.get_month_attendance(p_month text)

  1025→-- Mark one day Excel-style (upsert) with rating engine + auto-warning on 3 consecutive absences.
  1026→--   p_value: 'A' (absent) | numeric hours text e.g. '8', '10.5' (present) | '' / null (clear cell)
  1027→create or replace function public.mark_attendance(
  1028→  p_employee_id uuid, p_date date, p_value text default null
  1029→)
  1030→returns public.attendance language plpgsql
  1031→security definer set search_path = public
  1032→as $$
  1033→declare
  1034→  result public.attendance;
  1035→  prev public.attendance;
  1036→  v_hours numeric := null;
  1037→  v_status text;
  1038→  ot_delta numeric := 0;
  1039→  walk date;
  1040→  absent_dates date[] := array[p_date];
  1041→  v_emp_site uuid;
  1042→begin
  1043→  if not public.has_menu_permission('attendance') and not public.has_menu_permission('employees') then
  1044→    raise exception 'Access denied';
  1045→  end if;
  1046→
  1047→  -- site-scoped marking is DATE-EFFECTIVE: normal admins may only mark days
  1048→  -- the employee belonged to one of their assigned sites on (an employee
  1049→  -- moved mid-month stays markable at the old site for the days before the
  1050→  -- move and at the new site afterwards). Idle days remain super-admin only.
  1051→  select public.emp_site_on_date(p_employee_id, p_date) into v_emp_site;
  1052→  if not public.is_super_admin() then
  1053→    if v_emp_site is null then
  1054→      raise exception 'This employee has no site on this date — only super admins can mark idle employees';
  1055→    end if;
  1056→    if not exists (
  1057→      select 1 from public.admin_site where profile_id = auth.uid() and site_id = v_emp_site
  1058→    ) then
  1059→      raise exception 'Access denied — you can only mark attendance for your assigned sites';
  1060→    end if;
  1061→  end if;
  1062→
  1063→  -- parse the cell value
  1064→  if p_value is null or btrim(p_value) = '' then
  1065→    -- cleared cell: remove the record entirely
  1066→    delete from public.attendance where employee_id = p_employee_id and date = p_date;
  1067→    return prev;
  1068→  elsif lower(btrim(p_value)) in ('a', 'absent') then
  1069→    v_status := 'absent';
  1070→  else
  1071→    begin
  1072→      v_hours := btrim(p_value)::numeric;
  1073→    exception when others then
  1074→      raise exception 'Invalid value %. Enter hours (e.g. 8, 10.5) or A for absent', p_value;
  1075→    end;
  1076→    if v_hours < 0 or v_hours > 24 then
  1077→      raise exception 'Hours must be between 0 and 24';
  1078→    end if;
  1079→    v_status := 'present';
  1080→  end if;
  1081→
  1082→  select * into prev from public.attendance
  1083→    where employee_id = p_employee_id and date = p_date;
  1084→  if prev.id is not null then
  1085→    ot_delta := greatest(0, coalesce(v_hours,0) - 8) - greatest(0, coalesce(prev.hours,0) - 8);
  1086→  else
  1087→    ot_delta := greatest(0, coalesce(v_hours,0) - 8);
  1088→  end if;
  1089→
  1090→  insert into public.attendance (employee_id, date, status, hours)
  1091→  values (p_employee_id, p_date, v_status, case when v_status = 'present' then v_hours else null end)
  1092→  on conflict (employee_id, date) do update set
  1093→    status = excluded.status,
  1094→    hours = excluded.hours
  1095→  returning * into result;
  1096→
  1097→  -- rating: +/- 0.2 stars per overtime-hour delta (hours above 8/day)
  1098→  if ot_delta <> 0 then
  1099→    update public.employee set rating = public.clamp_rating(rating + 0.2 * ot_delta) where id = p_employee_id;
  1100→  end if;
  1101→
  1102→  -- auto-warning: 3 consecutive absences (walking back, skipping Fridays)
  1103→  if v_status = 'absent' then
  1104→    walk := p_date - 1;
  1105→    while walk >= p_date - 7 loop
  1106→      if extract(isodow from walk) = 5 then walk := walk - 1; continue; end if;
  1107→      exit when not exists (
  1108→        select 1 from public.attendance
  1109→        where employee_id = p_employee_id and date = walk and status = 'absent' and not is_hidden
  1110→      );
  1111→      absent_dates := walk || absent_dates;
  1112→      walk := walk - 1;
  1113→    end loop;
  1114→
  1115→    if array_length(absent_dates, 1) >= 3 then
  1116→      insert into public.warning (employee_id, reason, is_auto_generated, absent_dates, created_by)
  1117→      values (p_employee_id,
  1118→              'Auto-generated: ' || array_length(absent_dates,1) || ' consecutive absences detected (' ||
  1119→              (select string_agg(to_char(d,'YYYY-MM-DD'), ', ') from unnest(absent_dates) d) || ')',
  1120→              true, to_jsonb(absent_dates), auth.uid());
  1121→      update public.employee set rating = public.clamp_rating(rating - 0.5) where id = p_employee_id;
  1122→      perform public.notify_super_admins(
  1123→        'Auto-Generated Warning',
  1124→        (select full_name || ' (' || employee_code || ') was absent 3+ consecutive days' from public.employee where id = p_employee_id),
  1125→        'warning'
  1126→      );
  1127→    end if;
  1128→  end if;
  1129→
  1130→  return result;
  1131→end; $$;

  1133→-- Bulk day entry: apply the same Excel-style marking to many employees in ONE
  1134→-- round-trip (used by the "Bulk Day Entry" dialog). Each entry reuses the full
  1135→-- mark_attendance engine — including the DATE-EFFECTIVE site authorization
  1136→-- (emp_site_on_date), rating, auto-warnings — so bulk and single-cell marking
  1137→-- can never disagree.
  1138→--   p_entries: [{"employee_id": uuid, "date": "YYYY-MM-DD", "value": "10" | "A" | ""}]
  1139→-- Returns the number of applied entries.
  1140→create or replace function public.bulk_mark_attendance(p_entries jsonb)
  1141→returns integer language plpgsql
  1142→security definer set search_path = public
  1143→as $$
  1144→declare
  1145→  entry jsonb;
  1146→  v_emp uuid;
  1147→  v_date date;
  1148→  v_value text;
  1149→  applied integer := 0;
  1150→begin
  1151→  if not public.has_menu_permission('attendance') and not public.has_menu_permission('employees') then
  1152→    raise exception 'Access denied';
  1153→  end if;
  1154→
  1155→  if p_entries is null or jsonb_typeof(p_entries) <> 'array' then
  1156→    raise exception 'p_entries must be a JSON array';
  1157→  end if;
  1158→
  1159→  for entry in select * from jsonb_array_elements(p_entries) loop
  1160→    v_emp := null; v_date := null; v_value := null;
  1161→    begin
  1162→      v_emp := (entry->>'employee_id')::uuid;
  1163→      v_date := (entry->>'date')::date;
  1164→      v_value := entry->>'value';
  1165→    exception when others then
  1166→      raise exception 'Invalid bulk entry: %', entry::text;
  1167→    end;
  1168→
  1169→    if v_emp is null or v_date is null then
  1170→      raise exception 'Invalid bulk entry (missing employee_id or date): %', entry::text;
  1171→    end if;
  1172→
  1173→    perform public.mark_attendance(v_emp, v_date, v_value);
  1174→    applied := applied + 1;
  1175→  end loop;
  1176→
  1177→  return applied;
  1178→end; $$;

  1180→-- Full rating recompute from a month's attendance:
  1181→-- (worked days / working days excluding Fridays) * 5
  1182→create or replace function public.recalc_employee_rating(p_employee_id uuid, p_month text)
  1183→returns numeric language plpgsql
  1184→security definer set search_path = public
  1185→as $$
  1186→declare
  1187→  worked int;
  1188→  workdays int;
  1189→  new_rating numeric;
  1190→begin
  1191→  select count(*) into worked from public.attendance
  1192→    where employee_id = p_employee_id and to_char(date,'YYYY-MM') = p_month
  1193→      and status = 'present' and not is_hidden;
  1194→  select count(*) into workdays from generate_series(
  1195→    to_date(p_month || '-01','YYYY-MM-DD'),
  1196→    least(current_date, (to_date(p_month || '-01','YYYY-MM-DD') + interval '1 month - 1 day')::date),
  1197→    interval '1 day') d
  1198→  where extract(isodow from d) <> 5;
  1199→
  1200→  new_rating := public.clamp_rating(case when workdays > 0 then (worked::numeric / workdays) * 5 else 5 end);
  1201→  update public.employee set rating = new_rating where id = p_employee_id;
  1202→  return new_rating;
  1203→end; $$;

  1205→-- Dashboard chart data: per-status counts for the last N months (incl current)
  1206→create or replace function public.get_monthly_attendance_summary(p_months int default 6)
  1207→returns table (month text, status text, count bigint) language plpgsql
  1208→security definer set search_path = public
  1209→as $$
  1210→begin
  1211→  if not public.is_staff() then raise exception 'Access denied'; end if;

  1213→    with months as (
  1214→      select to_char(date_trunc('month', current_date) - (n || ' month')::interval, 'YYYY-MM') as m
  1215→      from generate_series(0, p_months - 1) n
  1216→    )
  1217→    select months.m,
  1218→           a.status,
  1219→           count(*)::bigint
  1220→    from months
  1221→    join public.attendance a on to_char(a.date, 'YYYY-MM') = months.m
  1222→    join public.employee e on e.id = a.employee_id
  1223→    where not a.is_hidden and e.status <> 'deleted' and a.status <> 'not_marked'
  1224→    group by months.m, a.status
  1225→    order by months.m;
