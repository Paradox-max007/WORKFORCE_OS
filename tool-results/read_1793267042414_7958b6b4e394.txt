    50→// client-side photo compression (max 300px, JPEG 0.8)
    51→function compressPhoto(file: File): Promise<string> {
    52→  return new Promise((resolve, reject) => {
    53→    const reader = new FileReader();
    54→    reader.onload = () => {
    55→      const img = new window.Image();
    56→      img.onload = () => {
    57→        const scale = Math.min(1, 300 / Math.max(img.width, img.height));
    58→        const canvas = document.createElement('canvas');
    59→        canvas.width = Math.round(img.width * scale);
    60→        canvas.height = Math.round(img.height * scale);
    61→        const ctx = canvas.getContext('2d');
    62→        if (!ctx) return reject(new Error('Canvas unavailable'));
    63→        ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
    64→        resolve(canvas.toDataURL('image/jpeg', 0.8));
    65→      };
    66→      img.onerror = () => reject(new Error('Invalid image'));
    67→      img.src = reader.result as string;
    68→    };
    69→    reader.onerror = () => reject(new Error('Failed to read file'));
    70→    reader.readAsDataURL(file);
    71→  });
    72→}
