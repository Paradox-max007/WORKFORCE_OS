    44→  'July', 'August', 'September', 'October', 'November', 'December'];

    49→const pad = (n: number): string => String(n).padStart(2, '0');
    50→const ymd = (y: number, m: number, d: number): string => `${y}-${pad(m)}-${pad(d)}`;

    58→interface MonthPoint { y: number; m: number }

    61→const monthFromIndex = (i: number): MonthPoint => ({ y: Math.floor(i / 12), m: (i % 12) + 1 });
    62→const monthOf = (iso: string): MonthPoint => {
    63→  const d = iso.length === 10 ? new Date(`${iso}T00:00:00`) : new Date(iso);
    64→  return { y: d.getFullYear(), m: d.getMonth() + 1 };

    83→/** "1, 2, 3, 5" → [1, 2, 3, 5] (validated, deduped, sorted) */
    84→const parseMonthsInput = (raw: string): number[] => {
    85→  const tokens = raw.split(/[,\s]+/).map((t) => t.trim()).filter(Boolean);
    86→  const nums = tokens.map(Number).filter((n) => Number.isInteger(n) && n >= 1 && n <= 12);
    87→  return [...new Set(nums)].sort((a, b) => a - b);
    88→};
    89→
    90→const invalidMonthTokens = (raw: string): string[] => {
    91→  const tokens = raw.split(/[,\s]+/).map((t) => t.trim()).filter(Boolean);
    92→  return tokens.filter((t) => {
    93→    const n = Number(t);
    94→    return !Number.isInteger(n) || n < 1 || n > 12;
    95→  });
    96→};

    98→const totalsFor = (records: Attendance[]): { hours: number; absents: number } => {
    99→  let hours = 0;
   100→  let absents = 0;
   101→  records.forEach((r) => {
   102→    if (r.status === 'present' && r.hours != null) hours += r.hours;
   103→    else if (r.status === 'absent') absents += 1;
   104→  });
   105→  return { hours: Math.round(hours * 100) / 100, absents };
   106→};
   107→
   108→const describeMonths = (points: MonthPoint[]): string => {
   109→  if (points.length === 0) return '';
   110→  if (points.length === 1) return monthLabel(points[0]);
   111→  const sorted = [...points].sort((a, b) => monthIndexOf(a) - monthIndexOf(b));
   112→  if (sorted.every((p) => p.y === sorted[0].y)) {
   113→    return `${sorted.map((p) => MONTH_SHORT[p.m - 1]).join(', ')} ${sorted[0].y}`;
   114→  }
   115→  return `${monthLabel(sorted[0])} – ${monthLabel(sorted[sorted.length - 1])}`;
   116→};

   118→/** Lifetime range: first working month = join month, or the earliest month that
   119→ *  has marked attendance (whichever is earlier — probed with a bounded look-back
   120→ *  that never precedes the employee record). Last working month = the current
   121→ *  month, or the updated_at month for deleted / pending-deletion employees. */
   122→async function computeLifetimeRange(emp: EmployeeVM): Promise<{ first: MonthPoint; last: MonthPoint }> {
   123→  const current = currentMonthPoint();
   124→  let last = current;
   125→  if ((emp.status === 'deleted' || emp.status === 'pending_deletion') && emp.updated_at) {
   126→    const lu = monthOf(emp.updated_at);
   127→    if (monthIndexOf(lu) < monthIndexOf(current)) last = lu;
   128→  }
   129→  const created = monthOf(emp.created_at);
   130→  let first = emp.join_date ? monthOf(emp.join_date) : created;
   131→  const floor = Math.max(monthIndexOf(created), monthIndexOf(first) - 6);
   132→  for (let i = monthIndexOf(first) - 1; i >= floor; i--) {
   133→    try {
   134→      const rows = await dataApi.getMonthAttendance(monthKeyOf(monthFromIndex(i)));
   135→      const hasMarks = rows.some((r) => r.employee_id === emp.id && r.status !== 'not_marked');
   136→      if (!hasMarks) break;
   137→      first = monthFromIndex(i);
   138→    } catch {
   139→      break;
   140→    }
   141→  }
   142→  if (monthIndexOf(first) > monthIndexOf(last)) first = last;
   143→  return { first, last };
   144→}
   145→
   146→const enumerateMonths = (first: MonthPoint, last: MonthPoint): MonthPoint[] => {
   147→  const out: MonthPoint[] = [];
   148→  for (let i = monthIndexOf(first); i <= monthIndexOf(last); i++) out.push(monthFromIndex(i));
   149→  return out;
   150→};

   197→/**
   198→ * Calendar-style month timesheet. Two variants:
   199→ *  - themed (default): app tokens, used on the Track Record page section.
   200→ *  - print: explicit white/black classes so the captured PDF looks the same in
   201→ *    light & dark themes (html2canvas-pro captures the real computed colors).
   202→ */
   203→function MonthGrid({ year, month, records, print = false }: {
   204→  year: number; month: number; records: Attendance[]; print?: boolean;
   205→}) {
   206→  const today = todayYMD();
   207→  const lead = new Date(year, month - 1, 1).getDay();
   208→  const days = daysInMonth(year, month);
   209→  const byDate = useMemo(() => {
   210→    const m = new Map<string, Attendance>();
   211→    records.forEach((r) => m.set(r.date, r));
   212→    return m;
   213→  }, [records]);
   214→
   215→  return (
   216→    <div className={cn('grid grid-cols-7', print ? 'gap-1' : 'gap-1.5')}>
   217→      {WEEKDAYS.map((w) => (
   218→        <div
   219→          key={w}
   220→          className={cn(
   221→            'pb-1 text-center text-[11px] font-semibold uppercase tracking-wide',
   222→            print
   223→              ? (w === 'Fri' ? 'text-rose-600' : 'text-zinc-500')
   224→              : (w === 'Fri' ? 'text-rose-500' : 'text-muted-foreground'),
   225→          )}
   226→        >
   227→          {w}
   228→        </div>
   229→      ))}
   230→      {Array.from({ length: lead }).map((_, i) => <div key={`lead-${i}`} />)}
   231→      {Array.from({ length: days }).map((_, idx) => {
   232→        const d = idx + 1;
   233→        const date = ymd(year, month, d);
   234→        const rec = byDate.get(date);
   235→        const future = date > today;
   236→        const fri = isFriday(date);
   237→        const absent = rec?.status === 'absent';
   238→        const hasHours = rec?.status === 'present' && rec.hours != null;
   239→        return (
   240→          <div
   241→            key={d}
   242→            role="img"
   243→            aria-label={`${date}${fri ? ', Friday' : ''}${future ? ', upcoming' : absent ? ', absent' : hasHours ? `, ${rec?.hours} hours worked` : ', not marked'}`}
   244→            className={cn(
   245→              'relative flex min-h-16 flex-col items-center justify-center rounded-lg border text-sm',
   246→              print ? 'border-zinc-300 bg-white text-zinc-900' : 'border-border/60 bg-background/60',
   247→              fri && (print ? 'border-rose-200 bg-rose-50' : 'border-rose-500/20 bg-rose-500/[0.04]'),
   248→              future && 'opacity-40',
   249→              date === today && (print ? 'ring-2 ring-zinc-500' : 'ring-2 ring-brand/50'),
   250→            )}
   251→          >
   252→            <span className={cn('absolute left-1.5 top-1 text-[10px]', print ? 'text-zinc-500' : 'text-muted-foreground')}>
   253→              {d}
   254→            </span>
   255→            {absent ? (
   256→              <span className={cn('text-lg font-bold', print ? 'text-rose-600' : 'text-rose-500')}>A</span>
   257→            ) : hasHours ? (
   258→              <span className={cn('text-base font-semibold tabular-nums', print ? 'text-emerald-700' : 'text-emerald-500')}>
   259→                {rec?.hours}h
   260→              </span>
   261→            ) : (
   262→              <span className={cn('text-xs', print ? 'text-zinc-400' : 'text-muted-foreground/50')}>—</span>
   263→            )}
   264→          </div>
   265→        );
   266→      })}
   267→    </div>
   268→  );
   269→}

   341→function ShareDialogContent({ emp, onPreviewChange }: {
   342→  emp: EmployeeVM;
   343→  onPreviewChange: (hasPreview: boolean) => void;
   344→}) {
   345→  const [includeTrackRecord, setIncludeTrackRecord] = useState(true);
   346→  const [shareMode, setShareMode] = useState<'lifetime' | 'specific'>('lifetime');
   347→  const [specificYear, setSpecificYear] = useState(currentMonthPoint().y);
   348→  const [monthsInput, setMonthsInput] = useState('');
   349→  const [lifetimeRange, setLifetimeRange] = useState<{ first: MonthPoint; last: MonthPoint } | null>(null);
   350→  const [previewMonths, setPreviewMonths] = useState<MonthPreview[] | null>(null);
   351→  const [previewBusy, setPreviewBusy] = useState(false);
   352→  const [simpleBusy, setSimpleBusy] = useState(false);
   353→  const [pdfBusy, setPdfBusy] = useState(false);
   354→  const [waBusy, setWaBusy] = useState(false);
   355→  const previewRef = useRef<HTMLDivElement>(null);
   356→  const employeeId = emp.id;
   357→
   358→  const parsedMonths = useMemo(() => parseMonthsInput(monthsInput), [monthsInput]);
   359→  const invalidTokens = useMemo(() => invalidMonthTokens(monthsInput), [monthsInput]);
   360→  const lifetimeCount = useMemo(() => (
   361→    lifetimeRange ? monthIndexOf(lifetimeRange.last) - monthIndexOf(lifetimeRange.first) + 1 : 0
   362→  ), [lifetimeRange]);
   363→
   364→  // Live "first working month → last working month" helper text.
   365→  useEffect(() => {
   366→    if (!includeTrackRecord || shareMode !== 'lifetime') return;
   367→    let cancelled = false;
   368→    (async () => {
   369→      const range = await computeLifetimeRange(emp);
   370→      if (!cancelled) setLifetimeRange(range);
   371→    })();
   372→    return () => { cancelled = true; };
   373→    // Deps are stable on purpose: the emp object identity changes on every realtime
   374→    // refetch, but only the employee id matters here (join_date/updated_at are fixed
   375→    // within a dialog session).
   376→  }, [employeeId, includeTrackRecord, shareMode]);
   377→
   378→  const buildPreview = async () => {
   379→    let points: MonthPoint[];
   380→    if (shareMode === 'specific') {
   381→      if (parsedMonths.length === 0) {
   382→        toast.error('No months selected', { description: 'Enter at least one month number (1–12), e.g. 1, 2, 3.' });
   383→        return;
   384→      }
   385→      points = parsedMonths.map((m) => ({ y: specificYear, m }));
   386→    } else {
   387→      const range = lifetimeRange ?? await computeLifetimeRange(emp);
   388→      points = enumerateMonths(range.first, range.last);
   389→    }
   390→    setPreviewBusy(true);
   391→    try {
   392→      const results = await Promise.all(
   393→        points.map((p) => dataApi.getMonthAttendance(monthKeyOf(p)).catch(() => [] as Attendance[])),
   394→      );
   395→      const previews: MonthPreview[] = points.map((p, i) => ({
   396→        point: p,
   397→        records: results[i].filter((r) => r.employee_id === employeeId),
   398→      }));
   399→      setPreviewMonths(previews);
   400→      onPreviewChange(true);
   401→      toast.success(`Preview ready — ${previews.length} month${previews.length !== 1 ? 's' : ''}`);
   402→    } catch (err) {
   403→      toast.error(err instanceof Error ? err.message : 'Failed to build the preview');
   404→    } finally {
   405→      setPreviewBusy(false);
   406→    }
   407→  };
   408→
   409→  /** Old simple flow: CV PDF + wa.me, no track record. */
   410→  const simpleShare = () => {
   411→    setSimpleBusy(true);
   412→    try {
   413→      downloadEmployeeCv(emp);
   414→      window.open(`https://wa.me/?text=${encodeURIComponent('Employee details attached')}`, '_blank');
   415→      toast.success('CV downloaded — WhatsApp opened');
   416→    } finally {
   417→      setSimpleBusy(false);
   418→    }
   419→  };
   420→
   421→  const exportPreview = async (): Promise<boolean> => {
   422→    const el = previewRef.current;
   423→    if (!el) {
   424→      toast.error('The preview is not ready yet');
   425→      return false;
   426→    }
   427→    try {
   428→      await exportTrackRecordPdf(el, trackRecordFilename(emp));
   429→      return true;
   430→    } catch (err) {
   431→      toast.error(err instanceof Error ? err.message : 'PDF export failed');
   432→      return false;
   433→    }
   434→  };
   435→
   436→  const downloadPreview = async () => {
   437→    setPdfBusy(true);
   438→    const ok = await exportPreview();
   439→    if (ok) toast.success('Track record PDF downloaded');
   440→    setPdfBusy(false);
   441→  };
   442→
   443→  const sharePreview = async () => {
   444→    setWaBusy(true);
   445→    const summary = previewMonths ? describeMonths(previewMonths.map((p) => p.point)) : '';
   446→    const waUrl = `https://wa.me/?text=${encodeURIComponent(`Track record for ${emp.full_name} (${emp.employee_code}) — ${summary}. PDF attached.`)}`;
   447→    // opened synchronously inside the click so popup blockers allow it
   448→    const win = window.open('', '_blank');
   449→    const ok = await exportPreview();
