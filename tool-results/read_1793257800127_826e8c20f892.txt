   392→   392→  async deleteEmployeeCascade(id: string): Promise<void> {
   393→   393→    const e = db.employees.find((x) => x.id === id);
   394→   394→    if (!e) throw new Error('Employee not found');
   395→   395→    if (e.status === 'pending_deletion') throw new Error('This employee already has a pending deletion request');
   396→   396→
   397→   397→    db.attendance.filter((a) => a.employee_id === id).forEach((a) => { a.is_hidden = true; });
   398→   398→    db.warnings.filter((w) => w.employee_id === id).forEach((w) => { w.is_hidden = true; });
   399→   399→    db.fines.filter((f) => f.employee_id === id).forEach((f) => { f.is_hidden = true; });
   400→   400→    db.leaveRequests.filter((r) => r.employee_id === id).forEach((r) => { r.is_hidden = true; });
   401→   401→    db.cancellationRequests.filter((r) => r.employee_id === id).forEach((r) => { r.is_hidden = true; });
   402→   402→    db.uniforms.filter((u) => u.employee_id === id).forEach((u) => { u.is_deleted = true; u.is_hidden = true; });
   403→   403→    const nameLc = e.full_name.toLowerCase();
   404→   404→    const codeLc = e.employee_code.toLowerCase();
   405→   405→    db.notifications.filter((n) =>
   406→   406→      n.title.toLowerCase().includes(nameLc)
   407→   407→      || n.message.toLowerCase().includes(nameLc)
   408→   408→      || n.message.toLowerCase().includes(codeLc),
   409→   409→    ).forEach((n) => { n.is_hidden = true; });
   410→   410→
   411→   411→    e.status = 'deleted';
   412→   412→    e.current_site_id = null;
   413→   413→    e.is_team_leader = false;
   414→   414→    e.team_leader_site_id = null;
   415→   415→    saveAndEmit(['employee', 'attendance', 'warning', 'fine', 'leave_request', 'cancellation_request', 'uniform_registry', 'notification']);

   439→   439→  async getMonthAttendance(month: string): Promise<Attendance[]> {
   440→   440→    const today = toYMD(new Date());
   441→   441→    const [y, m] = month.split('-').map(Number);
   442→   442→    const first = `${month}-01`;
   443→   443→    const monthEnd = toYMD(new Date(y, m, 0));
   444→   444→    const lastDay = today < first ? null : today < monthEnd ? today : monthEnd;
   445→   445→
   446→   446→    // auto-create missing records for non-deleted employees
   447→   447→    if (lastDay) {
   448→   448→      for (const e of db.employees) {
   449→   449→        if (e.status === 'deleted') continue;
   450→   450→        for (let d = fromYMD(first); toYMD(d) <= lastDay; d.setDate(d.getDate() + 1)) {
   451→   451→          const dateStr = toYMD(d);
   452→   452→          const exists = db.attendance.some((a) => a.employee_id === e.id && a.date === dateStr);
   453→   453→          if (!exists) {
   454→   454→            db.attendance.push({
   455→   455→              id: uid(), employee_id: e.id, date: dateStr,
   456→   456→              status: e.current_site_id ? 'present' : 'no_site',
   457→   457→              overtime_hours: null, is_hidden: false, created_at: nowISO(),
   458→   458→            });
   459→   459→          }
   460→   460→        }
   461→   461→      }
   462→   462→      saveAndEmit(['attendance']);
   463→   463→    }
   464→   464→
   465→   465→    return db.attendance.filter((a) => {
   466→   466→      if (!a.date.startsWith(month) || a.is_hidden) return false;
   467→   467→      const e = db.employees.find((x) => x.id === a.employee_id);
   468→   468→      return e ? e.status !== 'deleted' : false;
   469→   469→    });

   472→   472→  async markAttendance(employeeId: string, date: string, status: AttendanceStatus, overtimeHours: number | null): Promise<void> {
   473→   473→    const e = db.employees.find((x) => x.id === employeeId);
   474→   474→    if (!e) throw new Error('Employee not found');
   475→   475→
   476→   476→    const prev = db.attendance.find((a) => a.employee_id === employeeId && a.date === date);
   477→   477→    const otDelta = prev ? (overtimeHours ?? 0) - (prev.overtime_hours ?? 0) : (status === 'overtime' ? overtimeHours ?? 0 : 0);
   478→   478→
   479→   479→    if (prev) {
   480→   480→      prev.status = status;
   481→   481→      prev.overtime_hours = status === 'overtime' ? overtimeHours : null;
   482→   482→    } else {
   483→   483→      db.attendance.push({
   484→   484→        id: uid(), employee_id: employeeId, date, status,
   485→   485→        overtime_hours: status === 'overtime' ? overtimeHours : null,
   486→   486→        is_hidden: false, created_at: nowISO(),
   487→   487→      });
   488→   488→    }
   489→   489→
   490→   490→    // rating: ±0.2 stars per overtime hour delta
   491→   491→    if (otDelta !== 0) {
   492→   492→      e.rating = clampRating(e.rating + 0.2 * otDelta);
   493→   493→    }
   494→   494→
   495→   495→    // auto-warning on 3+ consecutive absences (skipping Fridays)
   496→   496→    if (status === 'absent') {
   497→   497→      const absentDates: string[] = [date];
   498→   498→      let walk = addDaysStr(date, -1);
   499→   499→      const floorDate = addDaysStr(date, -7);
   500→   500→      while (walk >= floorDate) {
   501→   501→        if (isFridayStr(walk)) { walk = addDaysStr(walk, -1); continue; }
   502→   502→        const rec = db.attendance.find((a) => a.employee_id === employeeId && a.date === walk && !a.is_hidden);
   503→   503→        if (rec && rec.status === 'absent') {
   504→   504→          absentDates.unshift(walk);
   505→   505→          walk = addDaysStr(walk, -1);
   506→   506→        } else break;
   507→   507→      }
   508→   508→      if (absentDates.length >= 3) {
   509→   509→        db.warnings.push({
   510→   510→          id: uid(), employee_id: employeeId,
   511→   511→          reason: `Auto-generated: ${absentDates.length} consecutive absences detected (${absentDates.join(', ')})`,
   512→   512→          is_auto_generated: true, absent_dates: absentDates,
   513→   513→          created_by: currentProfile()?.id ?? null, is_hidden: false, created_at: nowISO(),
   514→   514→        });
   515→   515→        e.rating = clampRating(e.rating - 0.5);
   516→   516→        notifySuperAdmins(
   517→   517→          'Auto-Generated Warning',
   518→   518→          `${e.full_name} (${e.employee_code}) was absent 3+ consecutive days`,
   519→   519→          'warning',
   520→   520→        );
   521→   521→        saveAndEmit(['warning', 'notification', 'employee', 'attendance']);
   522→   522→        return;
   523→   523→      }
   524→   524→    }
   525→   525→    saveAndEmit(['attendance', 'employee']);
   526→   526→  },

   528→   528→  async getMonthlyAttendanceSummary(months = 6): Promise<MonthlySummaryRow[]> {
   529→   529→    const rows: MonthlySummaryRow[] = [];
   530→   530→    const now = new Date();
   531→   531→    for (let back = months - 1; back >= 0; back--) {
   532→   532→      const d = new Date(now.getFullYear(), now.getMonth() - back, 1);
   533→   533→      const m = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
   534→   534→      const counts: Record<string, number> = {};
   535→   535→      for (const a of db.attendance) {
   536→   536→        if (!a.date.startsWith(m) || a.is_hidden || a.status === 'not_marked') continue;
   537→   537→        const e = db.employees.find((x) => x.id === a.employee_id);
   538→   538→        if (!e || e.status === 'deleted') continue;
   539→   539→        counts[a.status] = (counts[a.status] ?? 0) + 1;
   540→   540→      }
   541→   541→      for (const [status, count] of Object.entries(counts)) {
   542→   542→        rows.push({ month: m, status: status as AttendanceStatus, count });
   543→   543→      }
   544→   544→    }
   545→   545→    return rows;
   546→   546→  },
