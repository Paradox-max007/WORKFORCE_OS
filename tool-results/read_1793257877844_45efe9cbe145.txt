   108→function formatFormalDate(dateStr: string): string {
   109→  try {
   110→    const d = new Date(dateStr);
   111→    return d.toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' });
   112→  } catch {
   113→    return dateStr;
   114→  }
   115→}

   117→function generateLeavePdfDoc(req: LeaveRequest): jsPDF {
   118→  const doc = new jsPDF();
   119→  const typeLabels: Record<string, string> = {
   120→    casual: 'Casual Leave',
   121→    sick: 'Sick Leave',
   122→    annual: 'Annual Leave',
   123→    emergency: 'Emergency Leave',
   124→    marriage: 'Marriage Leave',
   125→    other: `Other (${req.otherTypeText || ''})`,
   126→  };
   127→  const leaveType = typeLabels[req.type] || req.type;
   128→  const submissionDate = formatFormalDate(new Date().toISOString());
   129→  const statusLabel = req.status.charAt(0).toUpperCase() + req.status.slice(1);
   130→
   131→  // Company header
   132→  doc.setFontSize(18);
   133→  doc.setFont('helvetica', 'bold');
   134→  doc.text('ARABIAN SHIELD MANPOWER LLC.', 105, 25, { align: 'center' });
   135→  doc.setDrawColor(139, 0, 0);
   136→  doc.setLineWidth(1.5);
   137→  doc.line(20, 32, 190, 32);
   138→
   139→  // Title
   140→  let y = 42;
   141→  doc.setFontSize(16);
   142→  doc.setFont('helvetica', 'bold');
   143→  doc.setTextColor(139, 0, 0);
   144→  doc.text('LEAVE APPLICATION FORM', 105, y, { align: 'center' });
   145→  y += 12;
   146→
   147→  doc.setTextColor(30, 30, 30);
   148→  doc.setFontSize(10);
   149→  doc.setFont('helvetica', 'normal');
   150→  doc.text(`Reference: LEAVE-${req.id.substring(0, 8)}`, 20, y);
   151→  doc.text(`Application Date: ${formatFormalDate(req.createdAt)}`, 190, y, { align: 'right' });
   152→  doc.text(`Status: ${statusLabel}`, 190, y + 7, { align: 'right' });
   153→  y += 14;
   154→
   155→  // Separator
   156→  doc.setDrawColor(180, 180, 180);
   157→  doc.setLineWidth(0.3);
   158→  doc.line(20, y, 190, y);
   159→  y += 10;
   160→
   161→  // Section: Employee Information
   162→  doc.setFont('helvetica', 'bold');
   163→  doc.setFontSize(11);
   164→  doc.setTextColor(139, 0, 0);
   165→  doc.text('EMPLOYEE INFORMATION', 20, y);
   166→  y += 8;
   167→  doc.setDrawColor(180, 180, 180);
   168→  doc.line(20, y, 190, y);
   169→  y += 7;
   170→
   171→  doc.setTextColor(30, 30, 30);
   172→  doc.setFontSize(10);
   173→  const empFields = [
   174→    ['Employee Name', req.employee.fullName],
   175→    ['Employee ID', req.employee.employeeId],
   176→    ['Designation', req.employee.position || 'N/A'],
   177→    ['Company', req.employee.companyName || 'N/A'],
   178→    ['Contact', req.employee.phone || 'N/A'],
   179→    ['Nationality', req.employee.nationality || 'N/A'],
   180→  ];
   181→  for (const [label, value] of empFields) {
   182→    doc.setFont('helvetica', 'bold');
   183→    doc.text(`${label}:`, 20, y);
   184→    doc.setFont('helvetica', 'normal');
   185→    doc.text(value, 70, y);
   186→    y += 7;
   187→  }
   188→  y += 5;
   189→
   190→  // Section: Leave Details
   191→  doc.setFont('helvetica', 'bold');
   192→  doc.setFontSize(11);
   193→  doc.setTextColor(139, 0, 0);
   194→  doc.text('LEAVE DETAILS', 20, y);
   195→  y += 8;
   196→  doc.setDrawColor(180, 180, 180);
   197→  doc.line(20, y, 190, y);
   198→  y += 7;
   199→
   200→  doc.setTextColor(30, 30, 30);
   201→  doc.setFontSize(10);
   202→  const leaveFields = [
   203→    ['Leave Type', leaveType],
   204→    ['Start Date', formatFormalDate(req.startDate)],
   205→    ['End Date', formatFormalDate(req.endDate)],
   206→    ['Total Days', `${req.totalDays} day${req.totalDays > 1 ? 's' : ''}`],
   207→  ];
   208→  for (const [label, value] of leaveFields) {
   209→    doc.setFont('helvetica', 'bold');
   210→    doc.text(`${label}:`, 20, y);
   211→    doc.setFont('helvetica', 'normal');
   212→    doc.text(value, 70, y);
   213→    y += 7;
   214→  }
   215→  y += 5;
   216→
   217→  // Section: Reason
   218→  doc.setFont('helvetica', 'bold');
   219→  doc.setFontSize(11);
   220→  doc.setTextColor(139, 0, 0);
   221→  doc.text('REASON FOR LEAVE', 20, y);
   222→  y += 8;
   223→  doc.setDrawColor(180, 180, 180);
   224→  doc.line(20, y, 190, y);
   225→  y += 7;
   226→
   227→  doc.setTextColor(50, 50, 50);
   228→  doc.setFont('helvetica', 'italic');
   229→  doc.setFontSize(10);
   230→  const reasonLines = doc.splitTextToSize(req.reason, 170);
   231→  doc.text(reasonLines, 20, y);
   232→  y += reasonLines.length * 6 + 10;
   233→
   234→  // Review status if applicable
   235→  if (req.status !== 'pending' && req.reviewedBy) {
   236→    doc.setTextColor(30, 30, 30);
   237→    doc.setFont('helvetica', 'bold');
   238→    doc.setFontSize(10);
   239→    doc.text(`${statusLabel} by: ${req.reviewedBy}`, 20, y);
   240→    y += 10;
   241→  }
   242→
   243→  // Signature area
   244→  y += 10;
   245→  doc.setDrawColor(160, 160, 160);
   246→  doc.setLineWidth(0.3);
   247→  doc.line(20, y, 190, y);
   248→  y += 15;
   249→
   250→  doc.setTextColor(30, 30, 30);
   251→  doc.setFontSize(10);
   252→  doc.setFont('helvetica', 'normal');
   253→  doc.text('Employee Signature: _______________', 25, y);
   254→  doc.text('HR / Office Admin: _______________', 115, y);
   255→  y += 20;
   256→
   257→  // Footer
   258→  doc.setDrawColor(200, 200, 200);
   259→  doc.line(20, y, 190, y);
   260→  y += 7;
   261→  doc.setFontSize(8);
   262→  doc.setTextColor(120, 120, 120);
   263→  doc.text('This is a computer-generated document from Arabian Shield Manpower LLC.', 105, y, { align: 'center' });
   264→
   265→  return doc;
   266→}

   340→async function sendLeaveRequestWhatsApp(req: LeaveRequest) {
   341→  try {
   342→    const doc = generateLeavePdfDoc(req);
   343→    const pdfBlob = doc.output('blob');
   344→    const file = new File([pdfBlob], `leave-application-${req.employee.fullName.replace(/\s+/g, '-')}.pdf`, { type: 'application/pdf' });
   345→
   346→    if (navigator.share && navigator.canShare && navigator.canShare({ files: [file] })) {
   347→      await navigator.share({
   348→        files: [file],
   349→        title: 'Leave Application',
   350→        text: `Leave Application - ${req.employee.fullName} (${req.employee.employeeId})`,
   351→      });
   352→    } else {
   353→      // Fallback: download PDF first, then open WhatsApp with text
   354→      const filename = `leave-application-${req.employee.fullName.replace(/\s+/g, '-').toLowerCase()}-${req.id.substring(0, 8)}.pdf`;
   355→      doc.save(filename);
   356→      const message = encodeURIComponent(
   357→        `*LEAVE APPLICATION - ARABIAN SHIELD MANPOWER LLC.*\n\nEmployee: ${req.employee.fullName}\nID: ${req.employee.employeeId}\nType: ${req.type}\nFrom: ${formatFormalDate(req.startDate)} To: ${formatFormalDate(req.endDate)}\nDays: ${req.totalDays}\nReason: ${req.reason}\n\nPlease find the attached PDF for complete details.`
   358→      );
   359→      window.open(`https://wa.me/?text=${message}`, '_blank');
   360→    }
   361→  } catch (err) {
   362→    // User cancelled share or error
   363→    if ((err as Error).name !== 'AbortError') {
   364→      console.error('Share failed:', err);
   365→    }
   366→  }
   367→}

   369→/* ───────── Searchable Employee Dropdown ───────── */
   370→interface SearchableEmployeeSelectProps {
   371→  employees: Employee[];
   372→  value: string;
   373→  onChange: (id: string) => void;

   376→function SearchableEmployeeSelect({ employees, value, onChange }: SearchableEmployeeSelectProps) {
   377→  const [open, setOpen] = useState(false);
   378→  const [search, setSearch] = useState('');
   379→  const containerRef = useRef<HTMLDivElement>(null);
   380→  const inputRef = useRef<HTMLInputElement>(null);
   381→
   382→  const selected = employees.find((e) => e.id === value);
   383→
   384→  const filtered = search
   385→    ? employees.filter(
   386→        (e) =>
   387→          e.fullName.toLowerCase().includes(search.toLowerCase()) ||
   388→          e.employeeId.toLowerCase().includes(search.toLowerCase()) ||
   389→          (e.idNumber && e.idNumber.toLowerCase().includes(search.toLowerCase()))
   390→      )
   391→    : employees;
   392→
   393→  useEffect(() => {
   394→    function handleClickOutside(e: MouseEvent) {
   395→      if (containerRef.current && !containerRef.current.contains(e.target as Node)) {
   396→        setOpen(false);
   397→        setSearch('');
   398→      }
   399→    }
   400→    document.addEventListener('mousedown', handleClickOutside);
   401→    return () => document.removeEventListener('mousedown', handleClickOutside);
   402→  }, []);
   403→
   404→  useEffect(() => {
   405→    if (open && inputRef.current) {
   406→      setTimeout(() => inputRef.current?.focus(), 50);
   407→    }
   408→  }, [open]);
   409→
   410→  return (
   411→    <div ref={containerRef} className="relative w-full">
   412→      <button
   413→        type="button"
   414→        onClick={() => setOpen(!open)}
   415→        className="flex items-center gap-2 w-full h-10 rounded-lg border border-slate-600 bg-slate-900 px-3 text-sm text-white hover:bg-slate-800 transition-colors text-left"
   416→      >
   417→        <User className="h-4 w-4 text-slate-500 shrink-0" />
   418→        <span className="truncate flex-1">
   419→          {selected ? `${selected.fullName} - ${selected.idNumber || selected.employeeId}` : 'Select employee...'}
   420→        </span>
   421→      </button>
   422→
   423→      {open && (
   424→        <div className="absolute top-full left-0 right-0 mt-1 z-50 bg-slate-800 border border-slate-600 rounded-lg shadow-xl shadow-black/40 overflow-hidden">
   425→          <div className="p-2 border-b border-slate-700">
   426→            <div className="relative">
   427→              <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-slate-400" />
   428→              <input
   429→                ref={inputRef}
   430→                type="text"
   431→                value={search}
   432→                onChange={(e) => setSearch(e.target.value)}
   433→                placeholder="Search by name or ID..."
   434→                className="w-full h-8 pl-8 pr-3 bg-slate-900 border border-slate-600 rounded-md text-sm text-white placeholder:text-slate-500 focus:outline-none focus:border-blue-500/50"
   435→              />
   436→            </div>
   437→          </div>
   438→          <div className="max-h-56 overflow-y-auto">
   439→            {filtered.length === 0 ? (
   440→              <div className="px-3 py-4 text-center text-sm text-slate-500">No employees found</div>
   441→            ) : (
   442→              filtered.map((emp) => (
   443→                <button
   444→                  key={emp.id}
   445→                  type="button"
   446→                  onClick={() => {
   447→                    onChange(emp.id);
   448→                    setOpen(false);
   449→                    setSearch('');
   450→                  }}
   451→                  className={cn(
   452→                    'flex items-center gap-2 w-full px-3 py-2 text-sm text-left transition-colors hover:bg-slate-700/50',
   453→                    emp.id === value ? 'bg-slate-700/70 text-white' : 'text-slate-300'
   454→                  )}
   455→                >
   456→                  <span className="truncate flex-1">{emp.fullName} - {emp.idNumber || emp.employeeId}</span>
   457→                </button>
   458→              ))
   459→            )}
   460→          </div>
   461→        </div>
   462→      )}
   463→    </div>
   464→  );
   465→}
