    86→function isFutureDate(day: number, month: number, year: number): boolean {
    87→  const today = new Date();
    88→  today.setHours(0, 0, 0, 0);
    89→  const check = new Date(year, month - 1, day);
    90→  return check > today;
    91→}

    93→function isFriday(year: number, month: number, day: number): boolean {
    94→  return new Date(year, month - 1, day).getDay() === 5;

    97→function getRelativeDateLabel(day: number, month: number, year: number): string | null {
    98→  const today = new Date();
    99→  today.setHours(0, 0, 0, 0);
   100→  const target = new Date(year, month - 1, day);
   101→  target.setHours(0, 0, 0, 0);
   102→  const diffMs = today.getTime() - target.getTime();
   103→  const diffDays = Math.round(diffMs / (1000 * 60 * 60 * 24));
   104→
   105→  if (diffDays === 0) return 'Today';
   106→  if (diffDays === 1) return 'Yesterday';
   107→  if (diffDays === 2) return '2 days ago';
   108→  if (diffDays === 3) return '3 days ago';
   109→  if (diffDays === 4) return '4 days ago';
   110→  if (diffDays === 5) return '5 days ago';
   111→  if (diffDays === 6) return '6 days ago';
   112→  return null; // older than 6 days — just show day number
   113→}

   125→interface StatusDropdownProps {
   126→  employeeId: string;
   127→  date: string;
   128→  currentStatus: StatusOption;
   129→  currentOvertimeHours: number | null;
   130→  onClose: () => void;

   134→
   135→function StatusDropdown({
   136→  employeeId,
   137→  date,
   138→  currentStatus,
   139→  currentOvertimeHours,
   140→  onClose,
   141→  onStatusChange,
   142→  position,
   143→}: StatusDropdownProps) {
   144→  const [selectedStatus, setSelectedStatus] = useState<StatusOption>(currentStatus);
   145→  const [overtimeHours, setOvertimeHours] = useState<string>(
   146→    currentOvertimeHours ? String(currentOvertimeHours) : '2'
   147→  );
   148→  const ref = useRef<HTMLDivElement>(null);
   149→
   150→  useEffect(() => {
   151→    function handleClickOutside(e: MouseEvent) {
   152→      if (ref.current && !ref.current.contains(e.target as Node)) {
   153→        onClose();
   154→      }
   155→    }
   156→    document.addEventListener('mousedown', handleClickOutside);
   157→    return () => document.removeEventListener('mousedown', handleClickOutside);
   158→  }, [onClose]);
   159→
   160→  const handleConfirm = () => {
   161→    const hours = selectedStatus === 'overtime' ? parseFloat(overtimeHours) || 0 : null;
   162→    onStatusChange(employeeId, date, selectedStatus, hours);
   163→    onClose();
   164→  };
   165→
   166→  return (
   167→    <div
   168→      ref={ref}
   169→      className="fixed z-[100] w-52 rounded-xl border border-slate-600 bg-slate-800 p-2 shadow-xl shadow-black/40"
   170→      style={{ top: Math.max(8, position.top - 240), left: Math.min(position.left, window.innerWidth - 220) }}
   171→    >
   172→      <div className="mb-2 px-2 py-1.5 text-xs font-medium text-slate-400 border-b border-slate-700">
   173→        {date}
   174→      </div>
   175→      <div className="flex flex-col gap-0.5">
   176→        {STATUS_OPTIONS.map((status) => {
   177→          const cfg = STATUS_CONFIG[status];
   178→          return (
   179→            <button
   180→              key={status}
   181→              onClick={() => setSelectedStatus(status)}
   182→              className={cn(
   183→                'flex items-center gap-2 rounded-lg px-2 py-1.5 text-sm transition-colors text-left w-full',
   184→                selectedStatus === status ? 'bg-slate-700/80 text-white' : 'text-slate-300 hover:bg-slate-700/50'
   185→              )}
   186→            >
   187→              <span className={cn('h-2.5 w-2.5 rounded-full shrink-0', cfg.dotColor)} />
   188→              <span>{cfg.label}</span>
   189→              {selectedStatus === status && (
   190→                <Check className="ml-auto h-3.5 w-3.5 text-white" />
   191→              )}
   192→            </button>
   193→          );
   194→        })}
   195→      </div>
   196→      {selectedStatus === 'overtime' && (
   197→        <div className="mt-2 px-2">
   198→          <label className="text-xs text-slate-400 mb-1 block">Overtime Hours</label>
   199→          <Input
   200→            type="number"
   201→            min="0"
   202→            max="24"
   203→            step="0.5"
   204→            value={overtimeHours}
   205→            onChange={(e) => setOvertimeHours(e.target.value)}
   206→            className="h-8 bg-slate-900 border-slate-600 text-white text-sm"
   207→          />
   208→        </div>
   209→      )}
   210→      <div className="mt-2 pt-2 border-t border-slate-700">
   211→        <Button
   212→          onClick={handleConfirm}
   213→          size="sm"
   214→          className="w-full h-8 bg-white hover:bg-gray-200 text-black text-xs font-medium"
   215→        >
   216→          Save
   217→        </Button>
   218→      </div>
   219→    </div>
   220→  );
   221→}

   371→
   372→function ListView({
   373→  employees,
   374→  attendanceMap,
   375→  daysInMonth,
   376→  monthStr,
   377→  yearStr,
   378→  month,
   379→  year,
   380→  loading,
   381→  isCurrentMonthView,
   382→  onStatusChange,
   383→}: ListViewProps) {
   384→  const [dropdown, setDropdown] = useState<{
   385→    employeeId: string;
   386→    date: string;
   387→    status: StatusOption;
   388→    overtimeHours: number | null;
   389→    position: { top: number; left: number };
   390→  } | null>(null);
   391→
   392→  // For current month: today on the left, all previous dates to the right (no future dates)
   393→  // For past months: normal order (1, 2, 3... last day)
   394→  const displayDays = useMemo(() => {
   395→    if (isCurrentMonthView) {
   396→      const today = new Date();
   397→      const currentDay = today.getDate();
   398→      // Only show days from 1 up to today, reversed: today on left, 1st on right
   399→      const days = Array.from({ length: currentDay }, (_, i) => currentDay - i);
   400→      return days;
   401→    }
   402→    return Array.from({ length: daysInMonth }, (_, i) => i + 1);
   403→  }, [daysInMonth, isCurrentMonthView]);
   404→
   405→  // Get label for each day column header
   406→  const getDayLabel = useCallback(
   407→    (day: number): string => {
   408→      if (!isCurrentMonthView) return String(day);
   409→      const relLabel = getRelativeDateLabel(day, month, year);
   410→      return relLabel || String(day);
   411→    },
   412→    [isCurrentMonthView, month, year]
   413→  );
   414→
   415→  // Check if a day is within the "relative label" zone (last 7 days for current month)
   416→  const isRecentDay = useCallback(
   417→    (day: number): boolean => {
   418→      if (!isCurrentMonthView) return false;
   419→      return getRelativeDateLabel(day, month, year) !== null;
   420→    },
   421→    [isCurrentMonthView, month, year]
   422→  );
   423→
   424→  if (loading) {
   425→    return (
   426→      <Card className="bg-slate-800/50 border-slate-700/50">
   427→        <CardContent className="p-4">
   428→          <div className="space-y-3">
   429→            {Array.from({ length: 5 }).map((_, i) => (
   430→              <Skeleton key={i} className="h-10 w-full bg-slate-700/50" />
   431→            ))}
   432→          </div>
   433→        </CardContent>
   434→      </Card>
   435→    );
   436→  }
   437→
   438→  if (employees.length === 0) {
   439→    return (
   440→      <Card className="bg-slate-800/50 border-slate-700/50">
   441→        <CardContent className="py-16 text-center">
   442→          <p className="text-sm text-slate-400">No active employees found</p>
   443→        </CardContent>
   444→      </Card>
   445→    );
   446→  }
   447→
   448→  return (
   449→    <Card className="bg-slate-800/50 border-slate-700/50 overflow-hidden">
   450→      <ScrollArea className="w-full">
   451→        <div className="min-w-[1000px]">
   452→          {/* Header row */}
   453→          <div className="flex items-center bg-slate-900/80 border-b border-slate-700 text-xs font-medium text-slate-400 sticky top-0 z-10">
   454→            <div className="w-52 shrink-0 px-4 py-3">Employee</div>
   455→            <div className="w-28 shrink-0 px-3 py-3">ID</div>
   456→            <div className="w-36 shrink-0 px-3 py-3">Site</div>
   457→            <div className="flex-1 flex">
   458→              {displayDays.map((day) => {
   459→                const isFri = isFriday(year, month, day);
   460→                const label = getDayLabel(day);
   461→                const recent = isRecentDay(day);
   462→                return (
   463→                  <div
   464→                    key={day}
   465→                    className={cn(
   466→                      'w-16 shrink-0 text-center py-3 leading-tight',
   467→                      isFri && 'text-red-400/50',
   468→                      recent && 'text-emerald-400 font-semibold'
   469→                    )}
   470→                  >
   471→                    <span className={cn(recent && 'text-[10px] block')}>{label}</span>

   493→                      {emp.fullName}

   497→                    <span className="text-xs text-slate-400 font-mono">{emp.employeeId}</span>
   498→                  </div>
   499→                  <div className="w-36 shrink-0 px-3 py-2.5">
   500→                    {emp.currentSite ? (
   501→                      <Badge variant="secondary" className="text-[10px] px-1.5 py-0 bg-emerald-500/15 text-emerald-400 border-emerald-500/30 truncate max-w-full block">

   512→                      const dateStr = formatDate(day, monthStr, yearStr);
   513→                      const record = attendanceMap.get(`${emp.id}-${dateStr}`);
   514→                      const status = record?.status || (emp.currentSite ? 'present' : 'no_site');
   515→                      const cfg = STATUS_CONFIG[status];
   516→                      const isFri = isFriday(year, month, day);
   517→                      const recent = isRecentDay(day);
   518→
   519→                      return (
   520→                        <div
   521→                          key={day}
   522→                          className={cn(
   523→                            'w-16 shrink-0 flex items-center justify-center py-1.5',
   524→                            isFri && 'bg-red-500/5',
   525→                            recent && 'bg-emerald-500/5'
   526→                          )}
   527→                        >
   528→                          <button
   529→                            onClick={(e) => {
   530→                              const rect = e.currentTarget.getBoundingClientRect();
   531→                              setDropdown({
   532→                                employeeId: emp.id,
   533→                                date: dateStr,
   534→                                status: record?.status || (emp.currentSite ? 'present' : 'no_site'),
   535→                                overtimeHours: record?.overtimeHours || null,
   536→                                position: { top: rect.top, left: rect.left },
   537→                              });
