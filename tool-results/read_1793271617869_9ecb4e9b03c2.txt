    66→type SubView = 'list' | 'residents';
    67→type CampTab = 'active' | 'inactive';
    68→
    69→const EASE: [number, number, number, number] = [0.22, 1, 0.36, 1];
    70→
    71→/** "" → null, otherwise the trimmed value (what the data layer expects). */
    72→const nullify = (v: string): string | null => (v.trim() === '' ? null : v.trim());
    73→
    74→/** Capacity string → whole non-negative number, or null when blank. */
    75→const parseCapacity = (raw: string): number | null => {
    76→  const t = raw.trim();
    77→  if (t === '') return null;
    78→  const n = Number(t);
    79→  if (Number.isNaN(n) || n < 0) return null; // invalid text is guarded by the caller
    80→  return Math.max(0, Math.floor(n));
    81→};
    82→
    83→const isCapacityInvalid = (raw: string): boolean => {
    84→  const t = raw.trim();
    85→  if (t === '') return false;
    86→  const n = Number(t);
    87→  return Number.isNaN(n) || n < 0;
    88→};

    91→// Main page
    92→// ---------------------------------------------------------------------------
    93→export function CampsPage() {
    94→  const version = useRealtimeVersion(['camp', 'employee']);
    95→
    96→  const [camps, setCamps] = useState<CampWithCount[] | null>(null);
    97→  const [employees, setEmployees] = useState<EmployeeVM[] | null>(null);
    98→
    99→  const [subView, setSubView] = useState<SubView>('list');
   100→  const [viewCampId, setViewCampId] = useState<string | null>(null);
   101→  const [tab, setTab] = useState<CampTab>('active');
   102→  const [search, setSearch] = useState('');
   103→  const [debouncedSearch, setDebouncedSearch] = useState('');
   104→  const [togglingId, setTogglingId] = useState<string | null>(null);
   105→
   106→  // dialogs
   107→  const [formOpen, setFormOpen] = useState(false);
   108→  const [editTarget, setEditTarget] = useState<CampWithCount | null>(null);
   109→  const [deleteTarget, setDeleteTarget] = useState<CampWithCount | null>(null);
   110→  const [dialogBusy, setDialogBusy] = useState(false);
   111→
   112→  // ---- data loading (re-fetches whenever camps or employees change) --------
   113→  useEffect(() => {
   114→    let cancelled = false;
   115→    (async () => {
   116→      try {
   117→        const [c, e] = await Promise.all([dataApi.listCamps(), dataApi.listEmployees()]);
   118→        if (!cancelled) {
   119→          setCamps(c);
   120→          setEmployees(e);
   121→        }
   122→      } catch {
   123→        if (!cancelled) {
   124→          setCamps([]);
   125→          setEmployees([]);
   126→        }
   127→      }
   128→    })();
   129→    return () => {
   130→      cancelled = true;
   131→    };
   132→  }, [version]);
   133→
   134→  // ---- search debounce (300ms) ---------------------------------------------
   135→  useEffect(() => {
   136→    const t = setTimeout(() => setDebouncedSearch(search.trim().toLowerCase()), 300);
   137→    return () => clearTimeout(t);
   138→  }, [search]);
   139→
   140→  // ---- derived values --------------------------------------------------------
   141→  const activeCamps = useMemo(() => (camps ?? []).filter((c) => c.is_active), [camps]);
   142→  const inactiveCamps = useMemo(() => (camps ?? []).filter((c) => !c.is_active), [camps]);
   143→  const housedCount = useMemo(
   144→    () => (employees ?? []).filter((e) => e.camp_id !== null && e.status === 'active').length,
   145→    [employees],
   146→  );
   147→  const loading = camps === null;
   148→
   149→  const viewCamp = useMemo(
   150→    () => (camps ?? []).find((c) => c.id === viewCampId) ?? null,
   151→    [camps, viewCampId],
   152→  );
   153→  const showResidentsView = subView === 'residents' && !!viewCamp;
   154→
   155→  const matchesSearch = (c: CampWithCount): boolean => {
   156→    if (!debouncedSearch) return true;
   157→    const hay = `${c.name} ${c.location ?? ''} ${c.notes ?? ''}`.toLowerCase();
   158→    return hay.includes(debouncedSearch);
   159→  };
   160→
   161→  // ---- actions ---------------------------------------------------------------
   162→
   163→  const openCreate = () => {
   164→    setEditTarget(null);
   165→    setFormOpen(true);
   166→  };
   167→  const openEdit = (camp: CampWithCount) => {
   168→    setEditTarget(camp);
   169→    setFormOpen(true);
   170→  };
   171→
   172→  const openResidents = (camp: CampWithCount) => {
   173→    setViewCampId(camp.id);
   174→    setSubView('residents');
   175→  };
   176→  const backToList = () => {
   177→    setSubView('list');
   178→    setViewCampId(null);
   179→  };
   180→
   181→  /** Toggle active/inactive. Deactivating KEEPS residents — it only stops new
   182→   *  assignments — so no confirmation is needed and it is instantly reversible. */
   183→  const handleToggleActive = async (camp: CampWithCount) => {
   184→    setTogglingId(camp.id);
   185→    try {
   186→      await dataApi.updateCamp(camp.id, {
   187→        name: camp.name,
   188→        location: camp.location,
   189→        capacity: camp.capacity,
   190→        notes: camp.notes,
   191→        is_active: !camp.is_active,
   192→      });
   193→      if (camp.is_active) {
   194→        toast.success('Camp deactivated', {
   195→          description: `"${camp.name}" no longer accepts new residents — current residents keep their rooms.`,
   196→        });
   197→      } else {
   198→        toast.success('Camp activated', {
   199→          description: `"${camp.name}" can accept new residents again.`,
   200→        });
   201→      }
   202→    } catch (err) {
   203→      toast.error(err instanceof Error ? err.message : 'Failed to update the camp');
   204→    } finally {
   205→      setTogglingId(null);
   206→    }
   207→  };
   208→
   209→  const handleConfirmDelete = async () => {
   210→    if (!deleteTarget) return;
   211→    setDialogBusy(true);
   212→    try {
   213→      await dataApi.deleteCamp(deleteTarget.id);
   214→      toast.success('Camp deleted', {
   215→        description: `"${deleteTarget.name}" was removed.${
   216→          deleteTarget.employee_count > 0
   217→            ? ` ${deleteTarget.employee_count} employee(s) were left without a camp — not deleted.`
   218→            : ''
   219→        }`,
   220→      });
   221→      setDeleteTarget(null);
   222→    } catch (err) {
   223→      toast.error(err instanceof Error ? err.message : 'Failed to delete the camp');
   224→    } finally {
   225→      setDialogBusy(false);
   226→    }
   227→  };
   228→
   229→  // ---- grid renderer (shared by both tabs) ----------------------------------
   230→  const renderGrid = (list: CampWithCount[], tabValue: CampTab) => {
   231→    if (loading) {
   232→      return <SkeletonRows rows={6} />;
   233→    }
   234→    const filtered = list.filter(matchesSearch);
   235→    if (filtered.length === 0) {
   236→      const searching = debouncedSearch.length > 0;
   237→      const noCampsAtAll = (camps ?? []).length === 0;
   238→      return (
   239→        <EmptyState
   240→          icon={<Tent size={22} />}
   241→          title={
   242→            searching
   243→              ? 'No matching camps'
   244→              : tabValue === 'active'
   245→                ? noCampsAtAll
   246→                  ? 'No camps yet'
   247→                  : 'No active camps'
   248→                : 'No inactive camps'
   249→          }
   250→          description={
   251→            searching
   252→              ? 'Try a different search term — search covers camp name, location and notes.'
   253→              : tabValue === 'active'
   254→                ? noCampsAtAll
   255→                  ? 'Create your first camp — the accommodation where your crews stay.'
   256→                  : 'All camps are inactive. Reactivate one from the Inactive tab to accept residents again.'
   257→                : 'Camps you deactivate will be moved here. Residents keep their rooms — a deactivated camp just stops accepting new ones.'
   258→          }
   259→          action={
   260→            !searching && tabValue === 'active' ? (
   261→              <Button onClick={openCreate} className="brand-gradient font-semibold text-white">
   262→                <Plus size={16} /> New Camp
   263→              </Button>
   264→            ) : undefined
   265→          }
   266→        />
   267→      );
   268→    }
   269→    return (
   270→      <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   271→        <AnimatePresence>
   272→          {filtered.map((camp, i) => (
   273→            <CampCard
   274→              key={camp.id}
   275→              camp={camp}
   276→              index={i}
   277→              busy={togglingId === camp.id}
   278→              onToggle={handleToggleActive}
   279→              onEdit={openEdit}
   280→              onDelete={setDeleteTarget}
   281→              onViewResidents={openResidents}
   282→            />
   283→          ))}
   284→        </AnimatePresence>
   285→      </div>
   286→    );
   287→  };
   288→
   289→  // ---- render ----------------------------------------------------------------
   290→  return (
   291→    <div className="mx-auto max-w-7xl">
   292→      <AnimatePresence mode="wait">
   293→        {showResidentsView && viewCamp ? (
   294→          <motion.div
   295→            key="camp-residents"
   296→            initial={{ opacity: 0, x: 32 }}
   297→            animate={{ opacity: 1, x: 0 }}
   298→            exit={{ opacity: 0, x: 32 }}
   299→            transition={{ duration: 0.3, ease: EASE }}
   300→          >
   301→            <CampResidentsView
   302→              camp={viewCamp}
   303→              camps={camps ?? []}
   304→              employees={employees}
   305→              onBack={backToList}
   306→            />
   307→          </motion.div>
   308→        ) : (
   309→          <motion.div
   310→            key="camp-list"
   311→            initial={{ opacity: 0, y: 16 }}
   312→            animate={{ opacity: 1, y: 0 }}
   313→            exit={{ opacity: 0, y: -16 }}
   314→            transition={{ duration: 0.3, ease: EASE }}
   315→          >
   316→            <div className="space-y-6">
   317→              <PageHeader
   318→                title="Camp Management"
   319→                description="Accommodation where your crews stay"
   320→                actions={
   321→                  <Button
   322→                    onClick={openCreate}
   323→                    className="h-10 brand-gradient font-semibold text-white shadow-lg shadow-[var(--glow)] transition-transform active:scale-[0.98]"
   324→                  >
   325→                    <Plus size={16} /> New Camp
   326→                  </Button>
   327→                }
   328→              />
   329→
   330→              {/* ---- summary chips ---- */}
   331→              <div className="grid grid-cols-1 gap-3 sm:grid-cols-3 sm:gap-4">
   332→                <StatChip
   333→                  icon={<Tent size={17} />}
   334→                  label="Total Camps"
   335→                  value={(camps ?? []).length}
   336→                  tone="brand"
   337→                  index={0}
   338→                  loading={loading}
   339→                />
   340→                <StatChip
   341→                  icon={<BedDouble size={17} />}
   342→                  label="Active Camps"
   343→                  value={activeCamps.length}
   344→                  tone="emerald"
   345→                  index={1}
   346→                  loading={loading}
   347→                />
   348→                <StatChip
   349→                  icon={<Users size={17} />}
   350→                  label="Employees Housed"
   351→                  value={housedCount}
   352→                  tone="amber"
   353→                  index={2}
   354→                  loading={loading}
   355→                />
   356→              </div>
   357→
   358→              {/* ---- tabs + search ---- */}
   359→              <Tabs value={tab} onValueChange={(v) => setTab(v as CampTab)}>
   360→                <div className="card-glow rounded-xl border bg-card">
   361→                  <div className="flex flex-col gap-3 border-b p-4 lg:flex-row lg:items-center lg:justify-between">
   362→                    <TabsList className="h-11 w-full gap-1 sm:w-auto">
   363→                      <TabsTrigger value="active" className="h-9 gap-2 px-4">
   364→                        <Power size={15} className="text-emerald-500" />
   365→                        Active
   366→                        <span className="rounded-full bg-emerald-500/15 px-2 py-0.5 text-[11px] font-semibold tabular-nums text-emerald-500">
   367→                          {activeCamps.length}
   368→                        </span>
   369→                      </TabsTrigger>
   370→                      <TabsTrigger value="inactive" className="h-9 gap-2 px-4">
   371→                        <PowerOff size={15} className="text-zinc-400" />
   372→                        Inactive
   373→                        <span className="rounded-full bg-zinc-500/15 px-2 py-0.5 text-[11px] font-semibold tabular-nums text-zinc-400">
   374→                          {inactiveCamps.length}
   375→                        </span>
   376→                      </TabsTrigger>
   377→                    </TabsList>
   378→                    <div className="relative w-full lg:max-w-xs">
   379→                      <Search size={16} className="pointer-events-none absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   380→                      <Input
   381→                        value={search}
   382→                        onChange={(e) => setSearch(e.target.value)}
   383→                        placeholder="Search camps by name, location or notes…"
   384→                        aria-label="Search camps"
   385→                        className="h-10 pl-9 pr-9"
   386→                      />
   387→                      {search && (
   388→                        <button
   389→                          type="button"
   390→                          onClick={() => setSearch('')}
   391→                          aria-label="Clear search"
   392→                          className="absolute right-2.5 top-1/2 -translate-y-1/2 rounded-sm text-muted-foreground transition-colors hover:text-foreground"
   393→                        >
   394→                          <X size={14} />
   395→                        </button>
   396→                      )}
   397→                    </div>
   398→                  </div>
   399→
   400→                  <TabsContent value="active" className="mt-0 p-4">
   401→                    {renderGrid(activeCamps, 'active')}
   402→                  </TabsContent>
   403→                  <TabsContent value="inactive" className="mt-0 p-4">
   404→                    {renderGrid(inactiveCamps, 'inactive')}
   405→                  </TabsContent>
   406→                </div>
   407→              </Tabs>
   408→            </div>
   409→          </motion.div>
   410→        )}
   411→      </AnimatePresence>
   412→
   413→      {/* ---- Create / Edit camp dialog ---- */}
   414→      <CampFormDialog open={formOpen} onOpenChange={setFormOpen} editing={editTarget} />
   415→
   416→      {/* ---- Delete confirmation ---- */}
   417→      <AlertDialog
   418→        open={!!deleteTarget}
   419→        onOpenChange={(o) => {
   420→          if (!o) setDeleteTarget(null);
   421→        }}
   422→      >
   423→        <AlertDialogContent>
   424→          <AlertDialogHeader>
   425→            <AlertDialogTitle className="flex items-center gap-2">
   426→              <Trash2 className="size-5 text-rose-500" /> Delete Camp
   427→            </AlertDialogTitle>
   428→            <AlertDialogDescription>
   429→              Are you sure you want to permanently delete{' '}
   430→              <span className="font-medium text-foreground">{deleteTarget?.name}</span>?
   431→            </AlertDialogDescription>
   432→          </AlertDialogHeader>
   433→          <div className="rounded-lg border border-rose-500/25 bg-rose-500/10 p-3">
   434→            <p className="text-sm font-medium text-rose-600 dark:text-rose-400">
   435→              {deleteTarget && deleteTarget.employee_count > 0
   436→                ? `${deleteTarget.employee_count} employee${deleteTarget.employee_count === 1 ? '' : 's'} will be left without a camp — they are NOT deleted.`
   437→                : 'No employees are housed here.'}
   438→            </p>
   439→            <p className="mt-1 text-xs text-muted-foreground">
   440→              Their camp assignment will simply be cleared. This action cannot be undone.
   441→            </p>
   442→          </div>
   443→          <AlertDialogFooter>
   444→            <AlertDialogCancel disabled={dialogBusy}>Cancel</AlertDialogCancel>
   445→            <Button
   446→              onClick={handleConfirmDelete}
   447→              disabled={dialogBusy}
   448→              className="bg-rose-600 font-semibold text-white hover:bg-rose-600/90"
   449→            >
   450→              {dialogBusy ? <Loader2 size={16} className="animate-spin" /> : <Trash2 size={16} />}
   451→              Delete Camp
   452→            </Button>
   453→          </AlertDialogFooter>
   454→        </AlertDialogContent>
   455→      </AlertDialog>
   456→    </div>
   457→  );
   458→}

   460→// ---------------------------------------------------------------------------

   462→// ---------------------------------------------------------------------------
   463→function StatChip({
   464→  icon, label, value, tone, index, loading,
   465→}: {
   466→  icon: ReactNode;
   467→  label: string;
   468→  value: number;
   469→  tone: 'brand' | 'emerald' | 'amber';
   470→  index: number;
   471→  loading: boolean;
   472→}) {
   473→  const tones: Record<string, string> = {
   474→    brand: 'bg-brand/10 text-brand',
   475→    emerald: 'bg-emerald-500/10 text-emerald-500',
   476→    amber: 'bg-amber-500/10 text-amber-500',
   477→  };
   478→  return (
   479→    <motion.div
   480→      initial={{ opacity: 0, y: 8 }}
   481→      animate={{ opacity: 1, y: 0 }}
   482→      transition={{ delay: Math.min(index * 0.05, 0.3), duration: 0.35, ease: EASE }}
   483→      className="card-glow flex items-center justify-between gap-3 rounded-xl border bg-card p-4"
   484→    >
   485→      <div className="min-w-0">
   486→        <p className="text-xs font-medium text-muted-foreground">{label}</p>
   487→        {loading ? (
   488→          <Skeleton className="mt-1.5 h-7 w-14" />
   489→        ) : (
   490→          <p className="mt-1 text-xl font-semibold tracking-tight">
   491→            <AnimatedCounter value={value} />
   492→          </p>
   493→        )}
   494→      </div>
   495→      <div className={cn('shrink-0 rounded-lg p-2.5', tones[tone])}>{icon}</div>
   496→    </motion.div>
   497→  );
   498→}

   500→// ---------------------------------------------------------------------------
   501→// Camp card (grid tile)
   502→// ---------------------------------------------------------------------------
   503→function CampCard({
   504→  camp, index, busy, onToggle, onEdit, onDelete, onViewResidents,
   505→}: {
   506→  camp: CampWithCount;
   507→  index: number;
   508→  busy: boolean;
   509→  onToggle: (camp: CampWithCount) => void;
   510→  onEdit: (camp: CampWithCount) => void;
   511→  onDelete: (camp: CampWithCount) => void;
   512→  onViewResidents: (camp: CampWithCount) => void;
   513→}) {
   514→  return (
   515→    <motion.div
   516→      initial={{ opacity: 0, y: 8 }}
   517→      animate={{ opacity: 1, y: 0 }}
   518→      exit={{ opacity: 0, scale: 0.96, y: 8, transition: { duration: 0.18, ease: 'easeIn' } }}
   519→      transition={{ delay: Math.min(index * 0.05, 0.5), duration: 0.35, ease: EASE }}
   520→      whileHover={{ y: -4 }}
   521→      className={cn(
   522→        'card-glow group relative flex flex-col gap-4 rounded-xl border bg-card p-5',
   523→        !camp.is_active && 'opacity-80',
   524→      )}
   525→    >
   526→      {/* header: icon, name, per-card actions */}
   527→      <div className="flex items-start justify-between gap-2">
   528→        <div className="flex min-w-0 items-center gap-3">
   529→          <div
   530→            className={cn(
   531→              'flex size-10 shrink-0 items-center justify-center rounded-lg',
   532→              camp.is_active ? 'bg-brand/10 text-brand' : 'bg-zinc-500/10 text-zinc-400',
   533→            )}
   534→          >
   535→            <Tent size={20} />
   536→          </div>
   537→          <div className="min-w-0">
   538→            <h3 className="truncate text-base font-semibold" title={camp.name}>
   539→              {camp.name}
   540→            </h3>
   541→            <p className="mt-0.5 text-[11px] text-muted-foreground">
   542→              Created {fmtDate(camp.created_at)}
   543→            </p>
   544→          </div>
   545→        </div>
   546→        <div className="flex shrink-0 items-center gap-0.5">
   547→          <Button
   548→            variant="ghost"
   549→            size="icon"
   550→            disabled={busy}
   551→            onClick={() => onToggle(camp)}
   552→            title={camp.is_active ? 'Deactivate camp' : 'Activate camp'}
   553→            aria-label={camp.is_active ? `Deactivate camp ${camp.name}` : `Activate camp ${camp.name}`}
   554→            className={cn(
   555→              'size-8',
   556→              camp.is_active

   888→// ---------------------------------------------------------------------------

   891→function CampResidentsView({
   892→  camp, camps, employees, onBack,
   893→}: {
   894→  camp: CampWithCount;
   895→  camps: CampWithCount[];
   896→  employees: EmployeeVM[] | null;
   897→  onBack: () => void;
   898→}) {
   899→  const [selected, setSelected] = useState<Set<string>>(new Set());
   900→  const [pickerValue, setPickerValue] = useState<string | null>(null);
   901→  const [addingId, setAddingId] = useState<string | null>(null);
   902→
   903→  const [moveEmp, setMoveEmp] = useState<EmployeeVM | null>(null);
   904→  const [moveTarget, setMoveTarget] = useState('');
   905→  const [moving, setMoving] = useState(false);
   906→  const [bulkMoveOpen, setBulkMoveOpen] = useState(false);
   907→  const [bulkTarget, setBulkTarget] = useState('');
   908→  const [bulkMoving, setBulkMoving] = useState(false);
   909→  const [removeTarget, setRemoveTarget] = useState<{ ids: string[]; label: string } | null>(null);
   910→  const [removing, setRemoving] = useState(false);
   911→
   912→  const loading = employees === null;
   913→
   914→  const residents = useMemo(() => {
   915→    const list = (employees ?? []).filter((e) => e.camp_id === camp.id && e.status !== 'deleted');
   916→    return [...list].sort((a, b) => a.full_name.localeCompare(b.full_name));
   917→  }, [employees, camp.id]);
   918→
   919→  // every non-deleted employee who is not already a resident here (pick one to
   920→  // house here — instantly transferred out of any previous camp)
   921→  const addable = useMemo(
   922→    () => (employees ?? []).filter((e) => e.status !== 'deleted' && e.camp_id !== camp.id),
   923→    [employees, camp.id],
   924→  );
   925→
   926→  // move destinations: other ACTIVE camps only (inactive camps don't accept new
   927→  // residents — assignments to them are rejected by the data layer)
   928→  const destinations = useMemo(
   929→    () => camps.filter((c) => c.is_active && c.id !== camp.id),
   930→    [camps, camp.id],
   931→  );
   932→
   933→  // occupancy
   934→  const limit = camp.capacity !== null && camp.capacity > 0 ? camp.capacity : null;
   935→  const usagePct = limit !== null ? Math.min(100, (residents.length / limit) * 100) : 0;
   936→  const full = limit !== null && residents.length >= limit;
   937→  const nearlyFull = limit !== null && !full && usagePct > 85;
   938→
   939→  const allSelected = residents.length > 0 && residents.every((e) => selected.has(e.id));
   940→  const someSelected = !allSelected && residents.some((e) => selected.has(e.id));
   941→
   942→  const openMove = (emp: EmployeeVM) => {
   943→    setMoveTarget('');
   944→    setMoveEmp(emp);
   945→  };
   946→  const openBulkMove = () => {
   947→    setBulkTarget('');
   948→    setBulkMoveOpen(true);
   949→  };
   950→
   951→  const toggleOne = (id: string) => {
   952→    setSelected((prev) => {
   953→      const next = new Set(prev);
   954→      if (next.has(id)) next.delete(id);
   955→      else next.add(id);
   956→      return next;
   957→    });
   958→  };
   959→  const toggleAll = () => {
   960→    setSelected(allSelected ? new Set() : new Set(residents.map((e) => e.id)));
   961→  };
   962→
   963→  const deselect = (id: string) => {
   964→    setSelected((prev) => {
   965→      if (!prev.has(id)) return prev;
   966→      const next = new Set(prev);
   967→      next.delete(id);
   968→      return next;
   969→    });
   970→  };
   971→
   972→  // ---- add / transfer a resident into this camp -------------------------------
   973→  const handleAdd = async (id: string | null) => {
   974→    setPickerValue(null);
   975→    if (!id || addingId || !camp.is_active) return;
   976→    const emp = (employees ?? []).find((e) => e.id === id);
   977→    if (!emp) return;
   978→    setAddingId(id);
   979→    try {
   980→      await dataApi.assignEmployeeCamp(id, camp.id);
   981→      toast.success(`Moved to ${camp.name}`, {
   982→        description: emp.camp_name
   983→          ? `${emp.full_name} was transferred from "${emp.camp_name}" to "${camp.name}".`
   984→          : `${emp.full_name} now stays at "${camp.name}".`,
   985→      });
   986→    } catch (err) {
   987→      toast.error(err instanceof Error ? err.message : 'Failed to add the resident');
   988→    } finally {
   989→      setAddingId(null);
   990→    }
   991→  };
   992→
   993→  // ---- move one resident to another active camp (or no camp) ------------------
   994→  const handleMove = async (emp: EmployeeVM, campId: string | null) => {
   995→    if (moving) return;
   996→    setMoving(true);
   997→    try {
   998→      await dataApi.assignEmployeeCamp(emp.id, campId);
   999→      if (campId) {
  1000→        const dest = destinations.find((c) => c.id === campId);
  1001→        toast.success('Resident moved', {
  1002→          description: `${emp.full_name} was moved from "${camp.name}" to "${dest?.name ?? 'another camp'}".`,
  1003→        });
  1004→      } else {
  1005→        toast.success('Resident moved out', {
  1006→          description: `${emp.full_name} was removed from "${camp.name}" and now has no camp.`,
  1007→        });
  1008→      }
  1009→      deselect(emp.id);
  1010→      setMoveEmp(null);
  1011→    } catch (err) {
  1012→      toast.error(err instanceof Error ? err.message : 'Failed to move the resident');
  1013→    } finally {
  1014→      setMoving(false);
  1015→    }
  1016→  };
  1017→
  1018→  // ---- bulk move selected residents to an active camp -------------------------
  1019→  const handleBulkMove = async (campId: string) => {
  1020→    if (!campId || selected.size === 0 || bulkMoving) return;
  1021→    const dest = destinations.find((c) => c.id === campId);
  1022→    setBulkMoving(true);
  1023→    try {
  1024→      await Promise.all([...selected].map((id) => dataApi.assignEmployeeCamp(id, campId)));
  1025→      toast.success(`${selected.size} resident${selected.size === 1 ? '' : 's'} moved`, {
  1026→        description: `They were moved from "${camp.name}" to "${dest?.name ?? 'another camp'}".`,
  1027→      });
  1028→      setSelected(new Set());
  1029→      setBulkMoveOpen(false);
  1030→    } catch (err) {
  1031→      toast.error(err instanceof Error ? err.message : 'Failed to move the resident(s)');
  1032→    } finally {
  1033→      setBulkMoving(false);
  1034→    }
  1035→  };
  1036→
  1037→  // ---- remove resident(s) — set to "no camp", never delete --------------------
  1038→  const handleRemove = async () => {
  1039→    if (!removeTarget || removing) return;
  1040→    setRemoving(true);
  1041→    try {
  1042→      await Promise.all(removeTarget.ids.map((id) => dataApi.assignEmployeeCamp(id, null)));
  1043→      toast.success(`${removeTarget.ids.length} resident${removeTarget.ids.length === 1 ? '' : 's'} removed`, {
  1044→        description: `They were left without a camp — not deleted.`,
  1045→      });
  1046→      setSelected(new Set());
  1047→      setRemoveTarget(null);
  1048→    } catch (err) {
  1049→      toast.error(err instanceof Error ? err.message : 'Failed to remove the resident(s)');
  1050→    } finally {
  1051→      setRemoving(false);
  1052→    }
  1053→  };
  1054→
  1055→  const avatar = (emp: EmployeeVM, sizeClass: string, textClass: string) => (
  1056→    <span
  1057→      className={cn(
  1058→        'flex shrink-0 items-center justify-center overflow-hidden rounded-full bg-brand/15 font-bold text-brand',
  1059→        sizeClass,
  1060→        textClass,
  1061→      )}
  1062→    >
  1063→      {emp.photo ? (
  1064→        <img src={emp.photo} alt={emp.full_name} className="size-full object-cover" />
  1065→      ) : (
  1066→        initials(emp.full_name)
  1067→      )}
  1068→    </span>
  1069→  );
  1070→
  1071→  return (
  1072→    <div className="space-y-6">
  1073→      {/* ---- header: back, camp identity, occupancy ---- */}
  1074→      <div className="flex flex-col gap-4 lg:flex-row lg:items-start lg:justify-between">
  1075→        <div className="flex items-start gap-3">
  1076→          <Button
  1077→            variant="outline"
  1078→            size="icon"
  1079→            onClick={onBack}
  1080→            aria-label="Back to camps"
  1081→            className="size-10 shrink-0"
  1082→          >
  1083→            <ArrowLeft size={17} />
  1084→          </Button>
  1085→          <div
  1086→            className={cn(
  1087→              'flex size-10 shrink-0 items-center justify-center rounded-xl p-2.5',
  1088→              camp.is_active ? 'bg-brand/10 text-brand' : 'bg-zinc-500/10 text-zinc-400',
  1089→            )}
  1090→          >
  1091→            <Tent size={20} />
  1092→          </div>
  1093→          <div className="min-w-0">
  1094→            <div className="flex flex-wrap items-center gap-2">
  1095→              <h1 className="text-xl font-semibold tracking-tight md:text-2xl">{camp.name}</h1>
  1096→              <Badge variant="secondary" className="tabular-nums">
  1097→                {residents.length} {residents.length === 1 ? 'resident' : 'residents'}
  1098→              </Badge>
  1099→              {!camp.is_active && (
  1100→                <Badge variant="secondary" className="border-zinc-500/25 bg-zinc-500/10 text-zinc-400">
  1101→                  Inactive
  1102→                </Badge>
