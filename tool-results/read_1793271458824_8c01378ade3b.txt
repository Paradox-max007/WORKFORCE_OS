    88→    88→export function UniformRegistryPage() {
    89→    89→  const version = useRealtimeVersion(['uniform_registry', 'employee', 'site']);
    90→    90→
    91→    91→  const [entries, setEntries] = useState<UniformEntry[] | null>(null);
    92→    92→  const [employees, setEmployees] = useState<EmployeeVM[] | null>(null);
    93→    93→  const [sites, setSites] = useState<SiteWithCount[] | null>(null);
    94→    94→
    95→    95→  // filters
    96→    96→  const [search, setSearch] = useState('');
    97→    97→  const [debouncedSearch, setDebouncedSearch] = useState('');
    98→    98→  const [siteFilter, setSiteFilter] = useState('all');
    99→    99→  const [page, setPage] = useState(1);
   100→   100→
   101→   101→  // details drill-down
   102→   102→  const [detailsEmployeeId, setDetailsEmployeeId] = useState<string | null>(null);
   103→   103→
   104→   104→  // shared create/renew dialog
   105→   105→  const [dialogTarget, setDialogTarget] = useState<DialogTarget | null>(null);
   106→   106→  const [dialogOpen, setDialogOpen] = useState(false);
   107→   107→
   108→   108→  const openDialog = (t: DialogTarget) => {
   109→   109→    setDialogTarget(t);
   110→   110→    setDialogOpen(true);
   111→   111→  };
   112→   112→
   113→   113→  // ---- data loading ---------------------------------------------------------
   114→   114→  useEffect(() => {
   115→   115→    let cancelled = false;
   116→   116→    (async () => {
   117→   117→      const [es, emps, sts] = await Promise.all([
   118→   118→        dataApi.listUniformEntries(),
   119→   119→        dataApi.listEmployees(),
   120→   120→        dataApi.listSites(),
   121→   121→      ]);
   122→   122→      if (!cancelled) {
   123→   123→        setEntries(es);
   124→   124→        setEmployees(emps);
   125→   125→        setSites(sts);
   126→   126→      }
   127→   127→    })().catch(() => {
   128→   128→      if (!cancelled) {
   129→   129→        setEntries([]);
   130→   130→        setEmployees([]);
   131→   131→        setSites([]);
   132→   132→      }
   133→   133→    });
   134→   134→    return () => { cancelled = true; };
   135→   135→  }, [version]);
   136→   136→
   137→   137→  // 300ms search debounce
   138→   138→  useEffect(() => {
   139→   139→    const t = setTimeout(() => setDebouncedSearch(search), 300);
   140→   140→    return () => clearTimeout(t);
   141→   141→  }, [search]);
   142→   142→
   143→   143→  const employeeMap = useMemo(
   144→   144→    () => new Map((employees ?? []).map((e) => [e.id, e])),
   145→   145→    [employees],
   146→   146→  );
   147→   147→
   148→   148→  // ---- rows: one per employee (newest entry + record count) -------------------
   149→   149→  const rows = useMemo<RegistryRow[] | null>(() => {
   150→   150→    if (!entries) return null;
   151→   151→    const byEmp = new Map<string, RegistryRow>();
   152→   152→    // listUniformEntries is newest-first → first occurrence IS the newest entry
   153→   153→    for (const entry of entries) {
   154→   154→      const cur = byEmp.get(entry.employee_id);
   155→   155→      if (cur) cur.count += 1;
   156→   156→      else {
   157→   157→        byEmp.set(entry.employee_id, {
   158→   158→          entry,
   159→   159→          count: 1,
   160→   160→          employee: employeeMap.get(entry.employee_id) ?? null,
   161→   161→        });
   162→   162→      }
   163→   163→    }
   164→   164→    return Array.from(byEmp.values());
   165→   165→  }, [entries, employeeMap]);
   166→   166→
   167→   167→  // site filter options = distinct site names present across entries
   168→   168→  const siteOptions = useMemo(() => {
   169→   169→    const names = new Set<string>();
   170→   170→    (entries ?? []).forEach((e) => {
   171→   171→      if (e.site_name) names.add(e.site_name);
   172→   172→    });
   173→   173→    return Array.from(names).sort((a, b) => a.localeCompare(b));
   174→   174→  }, [entries]);
   175→   175→
   176→   176→  // ---- filtering + pagination -------------------------------------------------
   177→   177→  const filteredRows = useMemo(() => {
   178→   178→    if (!rows) return null;
   179→   179→    const q = debouncedSearch.trim().toLowerCase();
   180→   180→    return rows.filter((r) => {
   181→   181→      if (siteFilter !== 'all' && r.entry.site_name !== siteFilter) return false;
   182→   182→      if (!q) return true;
   183→   183→      return (
   184→   184→        r.entry.employee_name.toLowerCase().includes(q) ||
   185→   185→        String(r.entry.token_number).includes(q) ||
   186→   186→        String(r.entry.uniform_id).includes(q)
   187→   187→      );
   188→   188→    });
   189→   189→  }, [rows, debouncedSearch, siteFilter]);
   190→   190→
   191→   191→  const totalPages = filteredRows ? Math.max(1, Math.ceil(filteredRows.length / PAGE_SIZE)) : 1;
   192→   192→  const safePage = Math.min(page, totalPages);
   193→   193→  const pageRows = filteredRows
   194→   194→    ? filteredRows.slice((safePage - 1) * PAGE_SIZE, safePage * PAGE_SIZE)
   195→   195→    : null;
   196→   196→
   197→   197→  const loading = entries === null || employees === null || sites === null;
   198→   198→  const filtersActive = debouncedSearch.trim() !== '' || siteFilter !== 'all';
   199→   199→
   200→   200→  const resetFilters = () => {
   201→   201→    setSearch('');
   202→   202→    setDebouncedSearch('');
   203→   203→    setSiteFilter('all');
   204→   204→    setPage(1);
   205→   205→  };
   206→   206→
   207→   207→  const handleDeleteEntry = async (entry: UniformEntry) => {
   208→   208→    try {
   209→   209→      await dataApi.deleteUniformEntry(entry.id);
   210→   210→      toast.success('Uniform entry deleted');
   211→   211→    } catch (err) {
   212→   212→      toast.error(errMessage(err));
   213→   213→    }
   214→   214→  };
   215→   215→
   216→   216→  const openDetails = (employeeId: string) => setDetailsEmployeeId(employeeId);
   217→   217→
   218→   218→  const detailsEmployee = detailsEmployeeId && employees
   219→   219→    ? employees.find((e) => e.id === detailsEmployeeId) ?? null
   220→   220→    : null;
   221→   221→
   222→   222→  // ---- row action handlers ----------------------------------------------------
   223→   223→  const rowActions = (row: RegistryRow) => ({
   224→   224→    onAdd: () => openDialog({ mode: 'create', presetEmployeeId: row.entry.employee_id }),
   225→   225→    onView: () => {
   226→   226→      if (row.employee) openDetails(row.employee.id);
   227→   227→    },
   228→   228→    onRenew: () => openDialog({ mode: 'renew', entry: row.entry }),
   229→   229→    onDelete: handleDeleteEntry,
   230→   230→  });
   231→   231→
   232→   232→  return (
   233→   233→    <div className="mx-auto max-w-7xl">
   234→   234→      <AnimatePresence mode="wait" initial={false}>
   235→   235→        {detailsEmployee ? (
   236→   236→          <motion.div
   237→   237→            key={`details-${detailsEmployee.id}`}
   238→   238→            initial={{ opacity: 0, y: 14 }}
   239→   239→            animate={{ opacity: 1, y: 0 }}
   240→   240→            exit={{ opacity: 0, y: -10 }}
   241→   241→            transition={{ duration: 0.26, ease: [0.22, 1, 0.36, 1] }}
   242→   242→          >
   243→   243→            <UniformEntryDetails
   244→   244→              key={detailsEmployee.id}
   245→   245→              employee={detailsEmployee}
   246→   246→              onBack={() => setDetailsEmployeeId(null)}
   247→   247→              autoOpenAdd={false}
   248→   248→              renewEntryId={dialogTarget?.mode === 'renew' ? dialogTarget.entry.id : null}
   249→   249→              onRenewEntry={(entry) => openDialog({ mode: 'renew', entry })}
   250→   250→            />
   251→   251→          </motion.div>
   252→   252→        ) : (
   253→   253→          <motion.div
   254→   254→            key="registry-list"
   255→   255→            initial={{ opacity: 0, y: 14 }}
   256→   256→            animate={{ opacity: 1, y: 0 }}
   257→   257→            exit={{ opacity: 0, y: -10 }}
   258→   258→            transition={{ duration: 0.26, ease: [0.22, 1, 0.36, 1] }}
   259→   259→          >
   260→   260→            <PageHeader
   261→   261→              title="Uniform Registry"
   262→   262→              description="Track issued uniforms, safety gear and 6-month renewals"
   263→   263→              actions={
   264→   264→                <Button
   265→   265→                  className="brand-gradient rounded-lg text-brand-foreground shadow-lg shadow-brand/20 hover:opacity-90"
   266→   266→                  onClick={() => openDialog({ mode: 'create', presetEmployeeId: null })}
   267→   267→                >
   268→   268→                  <Plus size={16} />
   269→   269→                  New Entry
   270→   270→                </Button>
   271→   271→              }
   272→   272→            />
   273→   273→
   274→   274→            {loading ? (
   275→   275→              <div className="space-y-4">
   276→   276→                <div className="flex flex-col gap-2 sm:flex-row">
   277→   277→                  <Skeleton className="h-10 w-full max-w-sm" />
   278→   278→                  <Skeleton className="h-10 w-48" />
   279→   279→                </div>
   280→   280→                <SkeletonRows rows={6} />
   281→   281→              </div>
   282→   282→            ) : (
   283→   283→              <>
   284→   284→                {/* Filters */}
   285→   285→                <div className="mb-4 flex flex-col gap-2 sm:flex-row sm:items-center">
   286→   286→                  <div className="relative w-full sm:max-w-sm">
   287→   287→                    <Search size={15} className="pointer-events-none absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   288→   288→                    <Input
   289→   289→                      className="h-10 bg-card pl-9"
   290→   290→                      placeholder="Search employee, token # or uniform ID…"
   291→   291→                      value={search}
   292→   292→                      onChange={(e) => {
   293→   293→                        setSearch(e.target.value);
   294→   294→                        setPage(1);
   295→   295→                      }}
   296→   296→                      aria-label="Search uniform registry"
   297→   297→                    />
   298→   298→                  </div>
   299→   299→                  <Select
   300→   300→                    value={siteFilter}
   301→   301→                    onValueChange={(v) => {
   302→   302→                      setSiteFilter(v);
   303→   303→                      setPage(1);
   304→   304→                    }}
   305→   305→                  >
   306→   306→                    <SelectTrigger className="h-10 w-full bg-card sm:w-52" aria-label="Filter by site">
   307→   307→                      <SelectValue />
   308→   308→                    </SelectTrigger>
   309→   309→                    <SelectContent>
   310→   310→                      <SelectItem value="all">All Sites</SelectItem>
   311→   311→                      {siteOptions.map((name) => (
   312→   312→                        <SelectItem key={name} value={name}>{name}</SelectItem>
   313→   313→                      ))}
   314→   314→                    </SelectContent>
   315→   315→                  </Select>
   316→   316→                  {filtersActive && (
   317→   317→                    <Button variant="ghost" size="sm" className="h-10" onClick={resetFilters}>
   318→   318→                      <RotateCcw size={14} />
   319→   319→                      Reset
   320→   320→                    </Button>
   321→   321→                  )}
   322→   322→                </div>
   323→   323→
   324→   324→                {/* Content */}
   325→   325→                {filteredRows && filteredRows.length === 0 ? (
   326→   326→                  (entries?.length ?? 0) === 0 ? (
   327→   327→                    <EmptyState
   328→   328→                      icon={<Shirt size={22} />}
   329→   329→                      title="No uniform entries yet"
   330→   330→                      description="Start tracking issued uniforms, safety gear and their 6-month renewals by creating the first entry."
   331→   331→                      action={
   332→   332→                        <Button
   333→   333→                          className="brand-gradient rounded-lg text-brand-foreground hover:opacity-90"
   334→   334→                          onClick={() => openDialog({ mode: 'create', presetEmployeeId: null })}
   335→   335→                        >
   336→   336→                          <Plus size={15} />
   337→   337→                          New Entry
   338→   338→                        </Button>
   339→   339→                      }
   340→   340→                    />
   341→   341→                  ) : (
   342→   342→                    <EmptyState
   343→   343→                      icon={<Search size={22} />}
   344→   344→                      title="No entries match your filters"
   345→   345→                      description="Try a different employee name, token number or site."
   346→   346→                      action={
   347→   347→                        <Button variant="outline" onClick={resetFilters}>
   348→   348→                          <RotateCcw size={14} />
   349→   349→                          Reset filters
   350→   350→                        </Button>
   351→   351→                      }
   352→   352→                    />
   353→   353→                  )
   354→   354→                ) : (
   355→   355→                  <>
   356→   356→                    {/* Desktop table */}
   357→   357→                    <div className="card-glow hidden overflow-hidden rounded-xl border bg-card md:block">
   358→   358→                      <Table>
   359→   359→                        <TableHeader>
   360→   360→                          <tr className="hover:bg-transparent">
   361→   361→                            <TableHead className="pl-4 text-xs font-semibold uppercase tracking-wide text-muted-foreground">Token #</TableHead>
   362→   362→                            <TableHead className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Uniform ID</TableHead>
   363→   363→                            <TableHead className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Employee</TableHead>
   364→   364→                            <TableHead className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Document</TableHead>
   365→   365→                            <TableHead className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Items</TableHead>
   366→   366→                            <TableHead className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Site</TableHead>
   367→   367→                            <TableHead className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Team Leader</TableHead>

   369→   369→                            <TableHead className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Created</TableHead>

   378→   378→                              initial={{ opacity: 0, y: 12 }}

   381→   381→                              className="border-b border-border/40 transition-colors last:border-0 hover:bg-muted/40"
   382→   382→                            >
   383→   383→                              <TableCell className="pl-4">
   384→   384→                                <span className="font-mono text-sm font-semibold">#{row.entry.token_number}</span>

   386→   386→                              </TableCell>
   387→   387→                              <TableCell>
   388→   388→                                <span className="font-mono text-xs text-muted-foreground">{row.entry.uniform_id}</span>
   389→   389→                              </TableCell>
   390→   390→                              <TableCell>
   391→   391→                                <p className="font-medium">{row.entry.employee_name}</p>
   392→   392→                                {row.employee && (
   393→   393→                                  <p className="font-mono text-[11px] text-muted-foreground">{row.employee.employee_code}</p>
   394→   394→                                )}
   395→   395→                              </TableCell>
   396→   396→                              <TableCell>
   397→   397→                                <span className="flex items-center gap-1.5">
   398→   398→                                  <DocTypeBadge docType={row.entry.document_type} />
   399→   399→                                  <span className="font-mono text-xs">{row.entry.document_number}</span>
   400→   400→                                </span>
   401→   401→                              </TableCell>
   402→   402→                              <TableCell className="whitespace-normal">
   403→   403→                                <UniformItemBadges items={row.entry.items} />
   404→   404→                              </TableCell>
   405→   405→                              <TableCell>
   406→   406→                                <span className="text-xs">
   407→   407→                                  {row.entry.site_name ?? <span className="text-muted-foreground">—</span>}
   408→   408→                                </span>
   409→   409→                              </TableCell>
   410→   410→                              <TableCell>
   411→   411→                                {row.entry.team_leader_name ? (
   412→   412→                                  <span className="inline-flex items-center gap-1 text-xs">

   420→   420→                              <TableCell>

   425→   425→                              </TableCell>

   429→   429→                                </span>
   430→   430→                              </TableCell>
   431→   431→                              <TableCell>

   440→   440→                      </Table>
   441→   441→                    </div>
   442→   442→
   443→   443→                    {/* Mobile cards */}
   444→   444→                    <div className="space-y-3 md:hidden">
   445→   445→                      {(pageRows ?? []).map((row, i) => (

   648→   648→function ListPagination({
   649→   649→  page,
   650→   650→  totalPages,
   651→   651→  onPageChange,
   652→   652→}: {
   653→   653→  page: number;
   654→   654→  totalPages: number;
   655→   655→  onPageChange: (p: number) => void;
   656→   656→}) {
   657→   657→  if (totalPages <= 1) return null;
   658→   658→
   659→   659→  const pages: (number | 'ellipsis')[] = [];
   660→   660→  if (totalPages <= 7) {
   661→   661→    for (let p = 1; p <= totalPages; p++) pages.push(p);
   662→   662→  } else {
   663→   663→    pages.push(1);
   664→   664→    if (page > 3) pages.push('ellipsis');
   665→   665→    for (let p = Math.max(2, page - 1); p <= Math.min(totalPages - 1, page + 1); p++) pages.push(p);
   666→   666→    if (page < totalPages - 2) pages.push('ellipsis');
   667→   667→    pages.push(totalPages);
   668→   668→  }
   669→   669→
   670→   670→  return (
   671→   671→    <div className="flex items-center gap-1">
   672→   672→      <Button
   673→   673→        variant="outline"
   674→   674→        size="icon"
   675→   675→        className="size-8"
   676→   676→        disabled={page === 1}
   677→   677→        onClick={() => onPageChange(1)}
   678→   678→        aria-label="First page"
   679→   679→      >
   680→   680→        <ChevronsLeft size={15} />
   681→   681→      </Button>
   682→   682→      <Button
   683→   683→        variant="outline"
   684→   684→        size="icon"
   685→   685→        className="size-8"
   686→   686→        disabled={page === 1}
   687→   687→        onClick={() => onPageChange(page - 1)}
   688→   688→        aria-label="Previous page"
   689→   689→      >
   690→   690→        <ChevronLeft size={15} />
   691→   691→      </Button>
   692→   692→      {pages.map((p, idx) =>
   693→   693→        p === 'ellipsis' ? (
   694→   694→          <span key={`ellipsis-${idx}`} className="px-1 text-xs text-muted-foreground">
   695→   695→            …
   696→   696→          </span>
   697→   697→        ) : (
   698→   698→          <Button
   699→   699→            key={p}
   700→   700→            variant={p === page ? 'default' : 'outline'}
   701→   701→            size="icon"
   702→   702→            className={cn(
   703→   703→              'size-8 text-xs tabular-nums',
   704→   704→              p === page && 'brand-gradient border-transparent text-brand-foreground',
   705→   705→            )}
   706→   706→            onClick={() => onPageChange(p)}
   707→   707→            aria-label={`Page ${p}`}
   708→   708→            aria-current={p === page ? 'page' : undefined}
   709→   709→          >
   710→   710→            {p}
   711→   711→          </Button>
   712→   712→        ),
   713→   713→      )}
   714→   714→      <Button
   715→   715→        variant="outline"
   716→   716→        size="icon"
   717→   717→        className="size-8"
   718→   718→        disabled={page === totalPages}
   719→   719→        onClick={() => onPageChange(page + 1)}
   720→   720→        aria-label="Next page"
   721→   721→      >
   722→   722→        <ChevronRight size={15} />
   723→   723→      </Button>
   724→   724→      <Button
   725→   725→        variant="outline"
   726→   726→        size="icon"
   727→   727→        className="size-8"
   728→   728→        disabled={page === totalPages}
   729→   729→        onClick={() => onPageChange(totalPages)}
   730→   730→        aria-label="Last page"
   731→   731→      >
   732→   732→        <ChevronsRight size={15} />
   733→   733→      </Button>
   734→   734→    </div>
   735→   735→  );
   736→   736→}

   784→   784→function EntryForm({

   801→   801→

   803→   803→    if (mode === 'renew' && renewEntry) {
   804→   804→      const site = sites.find((s) => s.name === renewEntry.site_name) ?? null;
   805→   805→      const leader = site

   823→   823→    const doc = preset
   824→   824→      ? preset.passport_number
   825→   825→        ? { type: 'passport' as const, number: preset.passport_number }
   826→   826→        : preset.id_number

   835→   835→      siteName: null,
   836→   836→      leaderId: null,

   838→   838→      createdDate: todayYMD(),
   839→   839→    };

   843→   843→  const patch = (p: Partial<EntryFormState>) => setState((prev) => ({ ...prev, ...p }));
   844→   844→  const patchItem = (key: string, value: boolean) =>
   845→   845→    setState((prev) => ({ ...prev, items: { ...prev.items, [key]: value } }));
   846→   846→

   848→   848→  const siteObj = useMemo(
   849→   849→    () => (state.siteName ? sites.find((s) => s.name === state.siteName) ?? null : null),
   850→   850→    [sites, state.siteName],
   851→   851→  );
   852→   852→  const leaderPool = useMemo(() => {
   853→   853→    if (!state.siteName) return [];
   854→   854→    const pool = activeEmployees.filter((e) => e.site_name === state.siteName);
   855→   855→    return [...pool].sort((a, b) => Number(b.is_team_leader) - Number(a.is_team_leader));
   856→   856→  }, [activeEmployees, state.siteName]);
   857→   857→  const currentLeader = useMemo(
   858→   858→    () => (siteObj ? employees.find((e) => e.is_team_leader && e.team_leader_site_id === siteObj.id) ?? null : null),
   859→   859→    [employees, siteObj],
   860→   860→  );
   861→   861→  const expiryPreview = formatExpiryPreview(state.createdDate);
   862→   862→

   865→   865→    if (emp) {
   866→   866→      // auto-fill document: passport wins when both exist, else id, else blank
   867→   867→      if (emp.passport_number) {
   868→   868→        patch({ employeeId: id, docType: 'passport', docNumber: emp.passport_number });
   869→   869→      } else if (emp.id_number) {
   870→   870→        patch({ employeeId: id, docType: 'id', docNumber: emp.id_number });
   871→   871→      } else {
   872→   872→        patch({ employeeId: id, docNumber: '' });
   873→   873→      }
   874→   874→    } else {
   875→   875→      patch({ employeeId: null, docNumber: '' });
   876→   876→    }
   877→   877→  };
   878→   878→
   879→   879→  const handleSiteChange = (value: string) => {
   880→   880→    const next = value === NO_SITE ? null : value;
   881→   881→    patch({ siteName: next, leaderId: null });
   882→   882→  };
   883→   883→
   884→   884→  const handleSubmit = async () => {

   903→   903→      const leaderEmp = state.leaderId ? activeEmployees.find((e) => e.id === state.leaderId) ?? null : null;
   904→   904→      const teamLeaderName = siteObj
   905→   905→        ? leaderEmp?.full_name ?? null
   906→   906→        : state.leaderFreeText.trim() || null;
   907→   907→
   908→   908→      await dataApi.createUniformEntry({
   909→   909→        employee_id: state.employeeId,
